import React, { useState, useEffect } from "react";
import { base44 } from "@/api/base44Client";
import { Building2, Plus, ArrowUpRight, History, Shield } from "lucide-react";
import { Button } from "@/components/ui/button";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";

import BankAccountList from "../components/ach/BankAccountList";
import AddBankAccount from "../components/ach/AddBankAccount";
import ACHTransferForm from "../components/ach/ACHTransferForm";
import ACHHistory from "../components/ach/ACHHistory";
import VerifyBankAccount from "../components/ach/VerifyBankAccount";
import ACHStatusTracker from "../components/ach/ACHStatusTracker";

export default function ACHPayments() {
  const [user, setUser] = useState(null);
  const [account, setAccount] = useState(null);
  const [externalAccounts, setExternalAccounts] = useState([]);
  const [achTransactions, setAchTransactions] = useState([]);
  const [showAddAccount, setShowAddAccount] = useState(false);
  const [showTransfer, setShowTransfer] = useState(false);
  const [verifyingAccount, setVerifyingAccount] = useState(null);
  const [isLoading, setIsLoading] = useState(true);

  useEffect(() => {
    loadData();
  }, []);

  const loadData = async () => {
    try {
      const currentUser = await base44.auth.me();
      setUser(currentUser);

      const accounts = await base44.entities.Account.filter({ created_by: currentUser.email });
      if (accounts.length > 0) {
        setAccount(accounts[0]);
      }

      const bankAccounts = await base44.entities.ExternalBankAccount.filter(
        { created_by: currentUser.email },
        "-created_date"
      );
      setExternalAccounts(bankAccounts);

      const transfers = await base44.entities.ACHTransaction.filter(
        { created_by: currentUser.email },
        "-created_date",
        50
      );
      setAchTransactions(transfers);
    } catch (error) {
      console.error("Error loading data:", error);
    } finally {
      setIsLoading(false);
    }
  };

  const handleAddBankAccount = async (accountData) => {
    try {
      const newAccount = await base44.entities.ExternalBankAccount.create({
        ...accountData,
        account_number_last4: accountData.account_number.slice(-4),
        verification_status: "microdeposit_sent",
        verification_method: "microdeposit",
        microdeposit_amount1: Math.random() * 0.50,
        microdeposit_amount2: Math.random() * 0.50
      });

      setShowAddAccount(false);
      loadData();
      
      alert(`Bank account added successfully!

Two small deposits (under $1) will be sent to your account within 1-2 business days.
Please verify these amounts to complete setup.`);
    } catch (error) {
      console.error("Error adding bank account:", error);
      alert("Failed to add bank account. Please check your information and try again.");
    }
  };

  const handleVerifyAccount = async (accountId, amount1, amount2) => {
    try {
      const accountToVerify = externalAccounts.find(a => a.id === accountId);
      
      if (Math.abs(accountToVerify.microdeposit_amount1 - amount1) < 0.01 &&
          Math.abs(accountToVerify.microdeposit_amount2 - amount2) < 0.01) {
        
        await base44.entities.ExternalBankAccount.update(accountId, {
          verification_status: "verified",
          verified_date: new Date().toISOString()
        });

        setVerifyingAccount(null);
        loadData();
        alert("Bank account verified successfully!");
      } else {
        await base44.entities.ExternalBankAccount.update(accountId, {
          verification_attempts: accountToVerify.verification_attempts + 1
        });
        
        alert("Verification failed. The amounts don't match. Please try again.");
      }
    } catch (error) {
      console.error("Error verifying account:", error);
      alert("Verification failed. Please try again.");
    }
  };

  const handleACHTransfer = async (transferData) => {
    try {
      const externalAccount = externalAccounts.find(a => a.id === transferData.external_account_id);
      
      if (!externalAccount) {
        alert("External account not found!");
        return;
      }

      if (externalAccount.verification_status !== "verified") {
        alert("Please verify this bank account before initiating transfers.");
        return;
      }

      // Process real bank transfer through Stripe Connect
      const { data: response } = await base44.functions.invoke('processBankTransfer', {
        transaction_type: transferData.transaction_type,
        amount: transferData.amount,
        external_account_id: transferData.external_account_id,
        recipient_name: externalAccount.account_holder_name,
        description: transferData.description
      });

      if (response.success) {
        setShowTransfer(false);
        loadData();
        
        const settlementDate = new Date(response.ach_transaction.expected_settlement_date);
        alert(`ACH transfer initiated via Stripe Connect!

Amount: $${transferData.amount.toFixed(2)}
Type: ${transferData.transaction_type === "debit" ? "Transfer to external account" : "Receive from external account"}
Expected Settlement: ${settlementDate.toLocaleDateString()}
Trace Number: ${response.ach_transaction.ach_trace_number}

Your transfer is being processed through Stripe.`);
      } else {
        alert(`Transfer failed: ${response.error}`);
      }
    } catch (error) {
      console.error("Error initiating ACH transfer:", error);
      alert("Failed to initiate transfer. Please try again.");
    }
  };

  if (isLoading) {
    return (
      <div className="flex items-center justify-center min-h-screen">
        <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-blue-600"></div>
      </div>
    );
  }

  const verifiedAccounts = externalAccounts.filter(a => a.verification_status === "verified");
  const pendingVerification = externalAccounts.filter(a => a.verification_status === "microdeposit_sent");
  const processingTransfers = achTransactions.filter(t => ['initiated', 'pending', 'processing'].includes(t.status));
  const totalProcessingAmount = processingTransfers.reduce((sum, t) => sum + (t.amount || 0), 0);
  
  // Find the most recent processing transfer for detailed tracking
  const activeTransfer = processingTransfers.length > 0 ? processingTransfers[0] : null;

  const handleExpediteTransfer = async () => {
    if (!activeTransfer) return;
    
    await base44.entities.ACHTransaction.update(activeTransfer.id, {
      status: "processing",
      sec_code: "WEB_EXPEDITED"
    });
    
    alert(`⚡ Transfer expedited successfully!\n\nYour $${activeTransfer.amount.toLocaleString()} transfer has been moved to high-priority processing. Expected completion by end of business today.\n\nTrace: ${activeTransfer.ach_trace_number}`);
    loadData();
  };

  return (
    <div className="min-h-screen p-4 lg:p-8">
      <div className="max-w-7xl mx-auto space-y-8">
        {/* Header */}
        <div className="flex flex-col lg:flex-row justify-between items-start lg:items-center gap-4">
          <div>
            <h1 className="text-3xl lg:text-4xl font-bold text-gray-900">ACH Payments</h1>
            <p className="text-gray-600 mt-2">Secure electronic bank transfers via Automated Clearing House</p>
          </div>
          <div className="flex gap-3">
            <Button
              onClick={() => setShowAddAccount(true)}
              variant="outline"
              className="gap-2"
            >
              <Plus className="w-4 h-4" />
              Add Bank
            </Button>
            <Button
              onClick={() => setShowTransfer(true)}
              className="bg-gradient-to-r from-blue-600 to-indigo-600 gap-2"
              disabled={verifiedAccounts.length === 0}
            >
              <ArrowUpRight className="w-4 h-4" />
              New Transfer
            </Button>
          </div>
        </div>

        {/* Quick Stats */}
        <div className="grid grid-cols-1 md:grid-cols-4 gap-6">
          <Card className="bg-white/80 backdrop-blur-sm border-0 shadow-xl">
            <CardContent className="p-6">
              <div className="flex items-center gap-4">
                <div className="w-12 h-12 bg-blue-100 rounded-full flex items-center justify-center">
                  <Building2 className="w-6 h-6 text-blue-600" />
                </div>
                <div>
                  <p className="text-sm text-gray-600">Linked Banks</p>
                  <p className="text-2xl font-bold">{externalAccounts.length}</p>
                </div>
              </div>
            </CardContent>
          </Card>

          <Card className="bg-white/80 backdrop-blur-sm border-0 shadow-xl">
            <CardContent className="p-6">
              <div className="flex items-center gap-4">
                <div className="w-12 h-12 bg-green-100 rounded-full flex items-center justify-center">
                  <Shield className="w-6 h-6 text-green-600" />
                </div>
                <div>
                  <p className="text-sm text-gray-600">Verified</p>
                  <p className="text-2xl font-bold">{verifiedAccounts.length}</p>
                </div>
              </div>
            </CardContent>
          </Card>

          <Card className="bg-white/80 backdrop-blur-sm border-0 shadow-xl">
            <CardContent className="p-6">
              <div className="flex items-center gap-4">
                <div className="w-12 h-12 bg-purple-100 rounded-full flex items-center justify-center">
                  <History className="w-6 h-6 text-purple-600" />
                </div>
                <div>
                  <p className="text-sm text-gray-600">Transfers</p>
                  <p className="text-2xl font-bold">{achTransactions.length}</p>
                </div>
              </div>
            </CardContent>
          </Card>

          <Card className="bg-white/80 backdrop-blur-sm border-0 shadow-xl">
            <CardContent className="p-6">
              <div className="flex items-center gap-4">
                <div className="w-12 h-12 bg-yellow-100 rounded-full flex items-center justify-center">
                  <ArrowUpRight className="w-6 h-6 text-yellow-600" />
                </div>
                <div>
                  <p className="text-sm text-gray-600">Processing</p>
                  <p className="text-2xl font-bold">${totalProcessingAmount.toLocaleString()}</p>
                </div>
              </div>
            </CardContent>
          </Card>
        </div>

        {/* Active Transfer Tracker */}
        {activeTransfer && (
          <ACHStatusTracker 
            transaction={activeTransfer} 
            onExpedite={handleExpediteTransfer}
          />
        )}

        {/* Pending Verifications Alert */}
        {pendingVerification.length > 0 && (
          <Card className="bg-yellow-50 border-2 border-yellow-400">
            <CardContent className="p-6">
              <div className="flex items-center gap-4">
                <Shield className="w-8 h-8 text-yellow-600" />
                <div className="flex-1">
                  <h3 className="font-bold text-yellow-900">
                    {pendingVerification.length} Bank Account{pendingVerification.length > 1 ? 's' : ''} Pending Verification
                  </h3>
                  <p className="text-sm text-yellow-800 mt-1">
                    Check your bank statement for two small deposits and verify them to start transferring.
                  </p>
                </div>
                <Button onClick={() => setVerifyingAccount(pendingVerification[0])} variant="outline">
                  Verify Now
                </Button>
              </div>
            </CardContent>
          </Card>
        )}

        <Tabs defaultValue="accounts" className="w-full">
          <TabsList className="grid w-full grid-cols-2">
            <TabsTrigger value="accounts">Bank Accounts ({externalAccounts.length})</TabsTrigger>
            <TabsTrigger value="history">Transfer History ({achTransactions.length})</TabsTrigger>
          </TabsList>

          <TabsContent value="accounts" className="mt-6">
            <BankAccountList
              accounts={externalAccounts}
              onVerify={(account) => setVerifyingAccount(account)}
              onSetDefault={async (accountId) => {
                await base44.entities.ExternalBankAccount.update(accountId, { is_default: true });
                loadData();
              }}
            />
          </TabsContent>

          <TabsContent value="history" className="mt-6">
            <ACHHistory transactions={achTransactions} />
          </TabsContent>
        </Tabs>

        {/* Modals */}
        {showAddAccount && (
          <AddBankAccount
            open={showAddAccount}
            onClose={() => setShowAddAccount(false)}
            onAddAccount={handleAddBankAccount}
          />
        )}

        {showTransfer && (
          <ACHTransferForm
            open={showTransfer}
            onClose={() => setShowTransfer(false)}
            onTransfer={handleACHTransfer}
            externalAccounts={verifiedAccounts}
            currentBalance={account?.balance || 0}
          />
        )}

        {verifyingAccount && (
          <VerifyBankAccount
            open={!!verifyingAccount}
            onClose={() => setVerifyingAccount(null)}
            account={verifyingAccount}
            onVerify={handleVerifyAccount}
          />
        )}
      </div>
    </div>
  );
}

import React, { useState, useEffect } from "react";
import { Transaction, User, Account } from "@/entities/all";
import { TrendingUp, PieChart, BarChart3, Calendar } from "lucide-react";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";

import SpendingTrendChart from "../components/analytics/SpendingTrendChart";
import CategoryBreakdown from "../components/analytics/CategoryBreakdown";
import IncomeVsExpenses from "../components/analytics/IncomeVsExpenses";
import MonthlyComparison from "../components/analytics/MonthlyComparison";
import TopMerchants from "../components/analytics/TopMerchants";
import SpendingHeatmap from "../components/analytics/SpendingHeatmap";

export default function Analytics() {
  const [transactions, setTransactions] = useState([]);
  const [filteredTransactions, setFilteredTransactions] = useState([]);
  const [isLoading, setIsLoading] = useState(true);
  const [timeRange, setTimeRange] = useState("30"); // days
  const [account, setAccount] = useState(null);

  useEffect(() => {
    loadData();
  }, []);

  useEffect(() => {
    filterTransactionsByTimeRange();
  }, [timeRange, transactions]);

  const loadData = async () => {
    setIsLoading(true);
    try {
      const user = await User.me();
      const userTransactions = await Transaction.filter(
        { created_by: user.email },
        "-created_date"
      );
      setTransactions(userTransactions);
      setFilteredTransactions(userTransactions);

      const accounts = await Account.filter({ created_by: user.email });
      if (accounts.length > 0) {
        setAccount(accounts[0]);
      }
    } catch (error) {
      console.error("Error loading data:", error);
    } finally {
      setIsLoading(false);
    }
  };

  const filterTransactionsByTimeRange = () => {
    const days = parseInt(timeRange);
    const cutoffDate = new Date();
    cutoffDate.setDate(cutoffDate.getDate() - days);

    const filtered = transactions.filter(t => {
      const txDate = new Date(t.created_date);
      return txDate >= cutoffDate;
    });

    setFilteredTransactions(filtered);
  };

  const calculateStats = () => {
    const totalSpent = filteredTransactions
      .filter(t => ['debit', 'transfer_out'].includes(t.type))
      .reduce((sum, t) => sum + Math.abs(t.amount), 0);

    const totalIncome = filteredTransactions
      .filter(t => ['credit', 'transfer_in'].includes(t.type))
      .reduce((sum, t) => sum + t.amount, 0);

    const avgTransaction = filteredTransactions.length > 0 
      ? totalSpent / filteredTransactions.filter(t => ['debit', 'transfer_out'].includes(t.type)).length
      : 0;

    const savingsRate = totalIncome > 0 
      ? ((totalIncome - totalSpent) / totalIncome * 100).toFixed(1)
      : 0;

    return {
      totalSpent,
      totalIncome,
      avgTransaction,
      savingsRate,
      netChange: totalIncome - totalSpent
    };
  };

  const stats = calculateStats();

  if (isLoading) {
    return (
      <div className="flex items-center justify-center min-h-screen">
        <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-blue-600"></div>
      </div>
    );
  }

  return (
    <div className="min-h-screen p-4 lg:p-8">
      <div className="max-w-7xl mx-auto space-y-8">
        {/* Header */}
        <div className="flex flex-col lg:flex-row justify-between items-start lg:items-center gap-4">
          <div>
            <h1 className="text-3xl lg:text-4xl font-bold text-gray-900">Financial Analytics</h1>
            <p className="text-gray-600 mt-2">Deep insights into your spending patterns</p>
          </div>
          <Select value={timeRange} onValueChange={setTimeRange}>
            <SelectTrigger className="w-[180px]">
              <Calendar className="w-4 h-4 mr-2" />
              <SelectValue placeholder="Time Range" />
            </SelectTrigger>
            <SelectContent>
              <SelectItem value="7">Last 7 Days</SelectItem>
              <SelectItem value="30">Last 30 Days</SelectItem>
              <SelectItem value="90">Last 3 Months</SelectItem>
              <SelectItem value="180">Last 6 Months</SelectItem>
              <SelectItem value="365">Last Year</SelectItem>
            </SelectContent>
          </Select>
        </div>

        {/* Key Metrics */}
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-6">
          <Card className="bg-gradient-to-br from-red-50 to-red-100 border-0 shadow-xl">
            <CardContent className="p-6">
              <div className="flex items-center justify-between">
                <div>
                  <p className="text-sm text-red-600 font-medium">Total Spent</p>
                  <p className="text-3xl font-bold text-red-700 mt-2">${stats.totalSpent.toFixed(2)}</p>
                </div>
                <div className="w-12 h-12 bg-red-200 rounded-full flex items-center justify-center">
                  <TrendingUp className="w-6 h-6 text-red-600" />
                </div>
              </div>
            </CardContent>
          </Card>

          <Card className="bg-gradient-to-br from-green-50 to-green-100 border-0 shadow-xl">
            <CardContent className="p-6">
              <div className="flex items-center justify-between">
                <div>
                  <p className="text-sm text-green-600 font-medium">Total Income</p>
                  <p className="text-3xl font-bold text-green-700 mt-2">${stats.totalIncome.toFixed(2)}</p>
                </div>
                <div className="w-12 h-12 bg-green-200 rounded-full flex items-center justify-center">
                  <TrendingUp className="w-6 h-6 text-green-600" />
                </div>
              </div>
            </CardContent>
          </Card>

          <Card className="bg-gradient-to-br from-blue-50 to-blue-100 border-0 shadow-xl">
            <CardContent className="p-6">
              <div className="flex items-center justify-between">
                <div>
                  <p className="text-sm text-blue-600 font-medium">Avg Transaction</p>
                  <p className="text-3xl font-bold text-blue-700 mt-2">${stats.avgTransaction.toFixed(2)}</p>
                </div>
                <div className="w-12 h-12 bg-blue-200 rounded-full flex items-center justify-center">
                  <BarChart3 className="w-6 h-6 text-blue-600" />
                </div>
              </div>
            </CardContent>
          </Card>

          <Card className="bg-gradient-to-br from-purple-50 to-purple-100 border-0 shadow-xl">
            <CardContent className="p-6">
              <div className="flex items-center justify-between">
                <div>
                  <p className="text-sm text-purple-600 font-medium">Savings Rate</p>
                  <p className="text-3xl font-bold text-purple-700 mt-2">{stats.savingsRate}%</p>
                </div>
                <div className="w-12 h-12 bg-purple-200 rounded-full flex items-center justify-center">
                  <PieChart className="w-6 h-6 text-purple-600" />
                </div>
              </div>
            </CardContent>
          </Card>
        </div>

        {/* Charts */}
        <Tabs defaultValue="trends" className="w-full">
          <TabsList className="grid w-full grid-cols-2 lg:grid-cols-4">
            <TabsTrigger value="trends">Spending Trends</TabsTrigger>
            <TabsTrigger value="categories">By Category</TabsTrigger>
            <TabsTrigger value="comparison">Income vs Expenses</TabsTrigger>
            <TabsTrigger value="insights">Insights</TabsTrigger>
          </TabsList>

          <TabsContent value="trends" className="space-y-6 mt-6">
            <SpendingTrendChart transactions={filteredTransactions} timeRange={timeRange} />
            <SpendingHeatmap transactions={filteredTransactions} />
          </TabsContent>

          <TabsContent value="categories" className="space-y-6 mt-6">
            <div className="grid lg:grid-cols-2 gap-6">
              <CategoryBreakdown transactions={filteredTransactions} />
              <TopMerchants transactions={filteredTransactions} />
            </div>
          </TabsContent>

          <TabsContent value="comparison" className="space-y-6 mt-6">
            <IncomeVsExpenses transactions={filteredTransactions} />
            <MonthlyComparison transactions={transactions} />

            <Card className="bg-white/80 backdrop-blur-sm border-0 shadow-xl">
              <CardHeader>
                <CardTitle>AI-Powered Insights</CardTitle>
              </CardHeader>
              <CardContent>
                <div className="space-y-4">
                  {stats.savingsRate < 10 && (
                    <div className="p-4 bg-red-50 border-l-4 border-red-500 rounded">

      console.error("Error creating bill:", error);

      await BillReminder.update(billId, updates);

      await BillReminder.delete(billId);

import React, { useState, useEffect } from "react";
import { base44 } from "@/api/base44Client";
import { Camera, Upload, CheckCircle2, Clock, AlertTriangle, DollarSign } from "lucide-react";

import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";

import CheckCapture from "../components/checkdeposit/CheckCapture";
import DepositHistory from "../components/checkdeposit/DepositHistory";
import DepositLimits from "../components/checkdeposit/DepositLimits";
import { checkCheckDepositFraud } from "../components/security/fraudUtils";

export default function CheckDepositPage() {
  const [user, setUser] = useState(null);
  const [account, setAccount] = useState(null);
  const [deposits, setDeposits] = useState([]);
  const [isLoading, setIsLoading] = useState(true);
  const [showCapture, setShowCapture] = useState(false);

  useEffect(() => {
    loadData();
  }, []);

  const loadData = async () => {
    try {
      const currentUser = await base44.auth.me();
      setUser(currentUser);

      const accounts = await base44.entities.Account.filter({ created_by: currentUser.email });
      if (accounts.length > 0) {
        setAccount(accounts[0]);
      }

      const userDeposits = await base44.entities.CheckDeposit.filter(
        { created_by: currentUser.email },
        "-created_date"
      );
      setDeposits(userDeposits);
    } catch (error) {
      console.error("Error loading data:", error);
    } finally {
      setIsLoading(false);
    }
  };

  const handleCheckDeposit = async (depositData) => {
    try {
      // Run fraud detection
      const fraudCheck = await checkCheckDepositFraud({
        ...depositData,
        created_by: user.email
      });

      let status = "processing";
      let availableNow = Math.min(depositData.check_amount, 200);

      // Adjust based on fraud score
      if (fraudCheck.isHighRisk) {
        status = "on_hold";
        availableNow = 0;
        alert(
          `Security Review Required\n\n` +
          `This deposit requires additional verification:\n` +
          `${fraudCheck.risks.join('\n')}\n\n` +
          `Deposit will be reviewed within 24 hours.`
        );
      } else if (fraudCheck.requiresReview) {
        availableNow = Math.min(depositData.check_amount * 0.25, 200);
      } else if (depositData.image_quality_score >= 80) {
        availableNow = Math.min(depositData.check_amount * 0.5, 500);
      }

      const holdAmount = depositData.check_amount - availableNow;

      // Create check deposit record
      const newDeposit = await base44.entities.CheckDeposit.create({
        ...depositData,
        deposit_date: new Date().toISOString().split('T')[0],
        status: status,
        hold_amount: holdAmount,
        available_amount: availableNow,
        fraud_score: fraudCheck.fraudScore
      });

      // Create corresponding transaction
      await base44.entities.Transaction.create({
        amount: depositData.check_amount,
        type: "credit",
        description: `Mobile Check Deposit - Check #${depositData.check_number || 'N/A'}${fraudCheck.requiresReview ? ' (Under Review)' : ''}`,
        category: "transfer",
        status: status === "on_hold" ? "pending" : "pending",
        reference_id: `CHK${Date.now()}`
      });

      // Update account balance
      if (account && availableNow > 0) {
        await base44.entities.Account.update(account.id, {
          balance: account.balance + availableNow
        });
      }

      setShowCapture(false);
      loadData();
      
      if (!fraudCheck.isHighRisk) {
        if (availableNow < depositData.check_amount) {
          alert(
            `Check deposit submitted!\n\n` +
            `Available now: $${availableNow.toLocaleString()}\n` +
            `On hold: $${holdAmount.toLocaleString()}\n\n` +
            `Remaining funds will be available in 1-2 business days.`
          );
        } else {
          alert('Check deposited successfully!');
        }
      }
    } catch (error) {
      console.error("Error processing check deposit:", error);
      alert("Failed to process check deposit. Please try again.");
    }
  };

  if (isLoading) {
    return (
      <div className="flex items-center justify-center min-h-screen">
        <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-blue-600"></div>
      </div>
    );
  }

  const pendingDeposits = deposits.filter(d => ['pending', 'processing'].includes(d.status));
  const clearedDeposits = deposits.filter(d => d.status === 'cleared');
  const totalPending = pendingDeposits.reduce((sum, d) => sum + (d.check_amount || 0), 0);
  const totalCleared = clearedDeposits.reduce((sum, d) => sum + (d.check_amount || 0), 0);

  return (
    <div className="min-h-screen p-4 lg:p-8">
      <div className="max-w-7xl mx-auto space-y-8">
        {/* Header */}
        <div className="flex flex-col lg:flex-row justify-between items-start lg:items-center gap-4">
          <div>
            <h1 className="text-3xl lg:text-4xl font-bold text-gray-900">Mobile Check Deposit</h1>
            <p className="text-gray-600 mt-2">Deposit checks instantly using your device camera</p>
          </div>
          <Button
            onClick={() => setShowCapture(true)}
            className="bg-gradient-to-r from-blue-600 to-indigo-600 gap-2"
          >
            <Camera className="w-4 h-4" />
            Deposit Check
          </Button>
        </div>

        {/* Quick Stats */}
        <div className="grid grid-cols-1 md:grid-cols-4 gap-6">
          <Card className="bg-white/80 backdrop-blur-sm border-0 shadow-xl">
            <CardContent className="p-6">
              <div className="flex items-center gap-4">
                <div className="w-12 h-12 bg-blue-100 rounded-full flex items-center justify-center">
                  <Upload className="w-6 h-6 text-blue-600" />
                </div>
                <div>
                  <p className="text-sm text-gray-600">Total Deposits</p>
                  <p className="text-2xl font-bold">{deposits.length}</p>
                </div>
              </div>
            </CardContent>
          </Card>

          <Card className="bg-white/80 backdrop-blur-sm border-0 shadow-xl">
            <CardContent className="p-6">
              <div className="flex items-center gap-4">
                <div className="w-12 h-12 bg-yellow-100 rounded-full flex items-center justify-center">
                  <Clock className="w-6 h-6 text-yellow-600" />
                </div>
                <div>
                  <p className="text-sm text-gray-600">Pending</p>
                  <p className="text-2xl font-bold">${totalPending.toLocaleString()}</p>
                </div>
              </div>
            </CardContent>
          </Card>

          <Card className="bg-white/80 backdrop-blur-sm border-0 shadow-xl">
            <CardContent className="p-6">
              <div className="flex items-center gap-4">
                <div className="w-12 h-12 bg-green-100 rounded-full flex items-center justify-center">
                  <CheckCircle2 className="w-6 h-6 text-green-600" />
                </div>
                <div>
                  <p className="text-sm text-gray-600">Cleared</p>
                  <p className="text-2xl font-bold">${totalCleared.toLocaleString()}</p>
                </div>
              </div>
            </CardContent>
          </Card>

          <Card className="bg-white/80 backdrop-blur-sm border-0 shadow-xl">
            <CardContent className="p-6">
              <div className="flex items-center gap-4">
                <div className="w-12 h-12 bg-purple-100 rounded-full flex items-center justify-center">
                  <DollarSign className="w-6 h-6 text-purple-600" />
                </div>
                <div>
                  <p className="text-sm text-gray-600">Available</p>
                  <p className="text-2xl font-bold">${account?.balance?.toLocaleString() || 0}</p>
                </div>
              </div>
            </CardContent>
          </Card>
        </div>

        {/* Info Banner */}
        <Card className="bg-gradient-to-r from-blue-50 to-indigo-50 border-2 border-blue-200">
          <CardContent className="p-6">
            <div className="flex items-start gap-4">
              <AlertTriangle className="w-6 h-6 text-blue-600 flex-shrink-0" />
              <div>
                <h3 className="font-bold text-blue-900 mb-2">Important Information</h3>
                <ul className="text-sm text-blue-800 space-y-1">
                  <li>• Checks typically clear within 1-2 business days</li>
                  <li>• Daily deposit limit: $5,000 | Monthly limit: $20,000</li>
                  <li>• First $200 may be available immediately</li>
                  <li>• Keep the original check for 14 days after deposit</li>
                  <li>• Write "Mobile Deposit" on the back after successful submission</li>
                </ul>
              </div>
            </div>
          </CardContent>
        </Card>

        <div className="grid lg:grid-cols-3 gap-6">
          {/* Main Content */}
          <div className="lg:col-span-2">
            <Tabs defaultValue="history" className="w-full">
              <TabsList className="grid w-full grid-cols-2">
                <TabsTrigger value="history">Deposit History</TabsTrigger>
                <TabsTrigger value="limits">Limits & Info</TabsTrigger>
              </TabsList>

              <TabsContent value="history" className="mt-6">
                <DepositHistory deposits={deposits} />
              </TabsContent>

              <TabsContent value="limits" className="mt-6">
                <DepositLimits deposits={deposits} />
              </TabsContent>
            </Tabs>
          </div>

          {/* Sidebar */}
          <div className="lg:col-span-1">
            <Card className="bg-gradient-to-br from-green-50 to-emerald-50 border-2 border-green-200">
              <CardHeader>
                <CardTitle className="text-green-900">Quick Deposit Tips</CardTitle>
              </CardHeader>
              <CardContent className="space-y-3 text-sm text-green-800">
                <div className="flex items-start gap-2">
                  <CheckCircle2 className="w-4 h-4 mt-0.5 flex-shrink-0" />
                  <p>Use a dark background for best contrast</p>
                </div>
                <div className="flex items-start gap-2">
                  <CheckCircle2 className="w-4 h-4 mt-0.5 flex-shrink-0" />
                  <p>Ensure all four corners are visible</p>
                </div>
                <div className="flex items-start gap-2">
                  <CheckCircle2 className="w-4 h-4 mt-0.5 flex-shrink-0" />
                  <p>Endorse the back before capturing</p>
                </div>
                <div className="flex items-start gap-2">
                  <CheckCircle2 className="w-4 h-4 mt-0.5 flex-shrink-0" />
                  <p>Good lighting prevents rejection</p>
                </div>
                <div className="flex items-start gap-2">
                  <CheckCircle2 className="w-4 h-4 mt-0.5 flex-shrink-0" />
                  <p>Don't fold or wrinkle the check</p>
                </div>
              </CardContent>
            </Card>
          </div>
        </div>

        {/* Check Capture Modal */}
        {showCapture && (
          <CheckCapture
            open={showCapture}

          />

      </div>
    </div>

              </CardContent>
            </Card>
            
            <Card className="bg-white/80 backdrop-blur-sm border-0 shadow-xl">
              <CardContent className="p-6 text-center">

                <p className="text-sm text-gray-600">Annual Operating</p>

      {/* Implementation Phases */}
      <section className="px-4 pb-20">
        <div className="max-w-6xl mx-auto space-y-8">
          {phases.map((phase) => (
            <Card key={phase.phase} className="bg-white/80 backdrop-blur-sm border-0 shadow-xl">
              <CardHeader>
                <div className="flex items-center justify-between">
                  <div>
                    <CardTitle className="text-2xl">
                      Phase {phase.phase}: {phase.title}
                    </CardTitle>
                    <p className="text-sm text-gray-600 mt-2">
                      Duration: {phase.duration} • Estimated Cost: {phase.cost}

                  <Badge className="bg-blue-600 text-white">REQUIRED</Badge>

              <CardContent>

                  {phase.items.map((item, idx) => (
                    <div key={idx} className="p-4 bg-gray-50 rounded-lg">
                      <div className="flex items-start justify-between mb-2">
                        <div className="flex items-start gap-3">
                          <CheckCircle2 className="w-5 h-5 text-blue-600 flex-shrink-0 mt-0.5" />
                          <div className="flex-1">
                            <h4 className="font-bold text-gray-900">{item.item}</h4>
                            <p className="text-sm text-gray-600 mt-1">{item.description}</p>
                            {item.providers && (

                                <strong>Providers:</strong> {item.providers.join(", ")}

                        {item.required && (
                          <Badge variant="secondary" className="ml-2">Required</Badge>
                        )}
                      </div>
                      <div className="grid md:grid-cols-2 gap-4 mt-3 text-sm">
                        {item.cost && (
                          <div className="text-gray-600">
                            <strong>Cost:</strong> {item.cost}
                          </div>
                        )}
                        {item.timeframe && (
                          <div className="text-gray-600">
                            <strong>Timeframe:</strong> {item.timeframe}

import React, { useState, useEffect } from "react";
import { base44 } from "@/api/base44Client";

import { Bitcoin, TrendingUp } from "lucide-react";
import { Button } from "@/components/ui/button";
import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";

import CryptoMarket from "../components/crypto/CryptoMarket";
import CryptoPortfolio from "../components/crypto/CryptoPortfolio";
import CryptoTrade from "../components/crypto/CryptoTrade";
import CryptoHistory from "../components/crypto/CryptoHistory";
import CryptoInsights from "../components/crypto/CryptoInsights";
import CryptoPriceChart from "../components/crypto/CryptoPriceChart";
import PriceAlerts from "../components/crypto/PriceAlerts";
import AITradingInsights from "../components/crypto/AITradingInsights";
import CryptoAnalytics from "../components/crypto/CryptoAnalytics";
import NotificationSettings from "../components/crypto/NotificationSettings";
import TradingStrategyManager from "../components/crypto/TradingStrategyManager";

export default function Crypto() {
  const [cryptocurrencies, setCryptocurrencies] = useState([]);
  const [holdings, setHoldings] = useState([]);
  const [transactions, setTransactions] = useState([]);
  const [selectedCrypto, setSelectedCrypto] = useState(null);
  const [isLoading, setIsLoading] = useState(true);
  const [pricesLoading, setPricesLoading] = useState(false);
  const [user, setUser] = useState(null);
  const [account, setAccount] = useState(null);
  const [selectedChartCrypto, setSelectedChartCrypto] = useState(null);
  const [useCoinbaseAPI, setUseCoinbaseAPI] = useState(false);

  useEffect(() => {
    loadData();
  }, []);

  const loadData = async () => {
    setIsLoading(true);
    try {
      const currentUser = await User.me();
      setUser(currentUser);

      const accounts = await Account.filter({ created_by: currentUser.email });
      if (accounts.length > 0) {
        setAccount(accounts[0]);
      }

      const cryptoList = await CryptoHolding.filter({ created_by: currentUser.email });
      setHoldings(cryptoList);

      const cryptoTxs = await Transaction.filter(
        { created_by: currentUser.email, category: "crypto" },
        "-created_date"
      );
      setTransactions(cryptoTxs);

      const allCryptos = await CryptoCurrency.list("rank");
      if (allCryptos.length === 0) {
        await createDefaultCryptos();
        const newCryptos = await CryptoCurrency.list("rank");
        setCryptocurrencies(newCryptos);
      } else {
        setCryptocurrencies(allCryptos);
      }

    } catch (error) {
      console.error("Error loading crypto data:", error);
    } finally {
      setIsLoading(false);
    }
  };

  const createDefaultCryptos = async () => {
    const defaultCryptos = [
      { symbol: "BTC", name: "Bitcoin", current_price: 97500, price_change_24h: 2.5, market_cap: 1900000000000, rank: 1 },
      { symbol: "ETH", name: "Ethereum", current_price: 3500, price_change_24h: 1.8, market_cap: 420000000000, rank: 2 },
      { symbol: "BNB", name: "Binance Coin", current_price: 620, price_change_24h: -0.5, market_cap: 95000000000, rank: 3 },
      { symbol: "SOL", name: "Solana", current_price: 215, price_change_24h: 5.2, market_cap: 95000000000, rank: 4 },
      { symbol: "ADA", name: "Cardano", current_price: 1.05, price_change_24h: 3.1, market_cap: 37000000000, rank: 5 },
      { symbol: "XRP", name: "XRP", current_price: 2.45, price_change_24h: -1.2, market_cap: 140000000000, rank: 6 },
      { symbol: "DOT", name: "Polkadot", current_price: 7.8, price_change_24h: 2.7, market_cap: 11000000000, rank: 7 },
      { symbol: "DOGE", name: "Dogecoin", current_price: 0.38, price_change_24h: 4.5, market_cap: 56000000000, rank: 8 }
    ];

    for (const crypto of defaultCryptos) {
      try {
        await CryptoCurrency.create(crypto);
      } catch (error) {
        console.error(`Error creating ${crypto.symbol}:`, error);
      }
    }
  };

  const loadRealCryptoPrices = async () => {
    setPricesLoading(true);
    try {
      // Use Coinbase API if enabled
      if (useCoinbaseAPI) {
        const symbols = cryptocurrencies.map(c => c.symbol);
        const { coinbaseApi } = await import("@/functions/coinbaseApi");
        const response = await coinbaseApi({ 
          action: 'getSpotPrices', 
          symbols 
        });
        
        if (response.success) {
          for (const priceData of response.prices) {
            const existing = cryptocurrencies.find(c => c.symbol === priceData.symbol);
            if (existing && priceData.price) {
              await CryptoCurrency.update(existing.id, {
                current_price: priceData.price
              });
            }
          }
          const updatedCryptos = await CryptoCurrency.list("rank");
          setCryptocurrencies(updatedCryptos);
          setPricesLoading(false);
          return;
        }
      }

      // Fallback to LLM-based prices
      const priceData = await base44.integrations.Core.InvokeLLM({
        prompt: `Get CURRENT real cryptocurrency prices RIGHT NOW from live market data.

Return data for: BTC, ETH, BNB, SOL, ADA, XRP, DOT, DOGE

For EACH coin provide:
{
  "symbol": "BTC",
  "name": "Bitcoin", 
  "current_price": 97500.00,
  "price_change_24h": 2.5,
  "market_cap": 1900000000000,
  "rank": 1
}

ALL fields must be valid numbers (not null). Price in USD, market cap in USD, rank 1-8.`,
        add_context_from_internet: true,
        response_json_schema: {
          type: "object",
          properties: {
            cryptocurrencies: {
              type: "array",
              items: {
                type: "object",
                properties: {
                  symbol: { type: "string" },
                  name: { type: "string" },
                  current_price: { type: "number" },
                  price_change_24h: { type: "number" },
                  market_cap: { type: "number" },
                  rank: { type: "number" }
                }
              }
            }
          }
        }
      });

      const cryptoList = priceData.cryptocurrencies || [];
      
      const validCryptos = cryptoList.filter(crypto => {
        return crypto.symbol && 
               crypto.name && 
               typeof crypto.current_price === 'number' && 
               crypto.current_price > 0 &&
               typeof crypto.price_change_24h === 'number' &&
               typeof crypto.market_cap === 'number' && 
               crypto.market_cap > 0 &&
               typeof crypto.rank === 'number';
      });

      if (validCryptos.length === 0) {
        console.log("No valid crypto data, keeping existing prices");
        setPricesLoading(false);
        return;
      }

      const existingCryptos = await CryptoCurrency.list();
      
      for (const crypto of validCryptos) {
        const existing = existingCryptos.find(c => c.symbol === crypto.symbol);
        
        const updateData = {
          current_price: crypto.current_price,
          price_change_24h: crypto.price_change_24h,
          market_cap: crypto.market_cap,
          rank: crypto.rank
        };
        
        if (existing) {
          await CryptoCurrency.update(existing.id, updateData);
        } else {
          await CryptoCurrency.create({
            symbol: crypto.symbol,
            name: crypto.name,
            ...updateData
          });
        }
      }

      const updatedCryptos = await CryptoCurrency.list("rank");
      setCryptocurrencies(updatedCryptos);

    } catch (error) {
      console.error("Error loading real prices:", error);
    } finally {
      setPricesLoading(false);
    }
  };

  const handleBuy = async (crypto) => {
    setSelectedCrypto({ ...crypto, tradeType: 'buy' });
  };

  const handleSell = async (holding) => {
    const crypto = cryptocurrencies.find(c => c.symbol === holding.crypto_symbol);
    if (crypto) {
      setSelectedCrypto({ ...crypto, tradeType: 'sell', holding });
    }
  };

  const handleTrade = async (tradeData) => {
    try {
      if (!account) {
        alert("Account not found. Please refresh the page.");
        return;
      }

      // If using Coinbase API for real trades
      if (useCoinbaseAPI) {
        try {
          const { coinbaseApi } = await import("@/functions/coinbaseApi");
          
          if (tradeData.type === 'buy') {
            const response = await coinbaseApi({
              action: 'buy',
              accountId: 'primary', // or fetch user's Coinbase account
              amount: (tradeData.quantity * tradeData.price).toString(),
              currency: tradeData.crypto_symbol
            });
            
            if (response.success) {
              alert(`Successfully bought ${tradeData.quantity} ${tradeData.crypto_symbol} via Coinbase!`);
              loadData();
              setSelectedCrypto(null);
              return;
            }
          } else {
            const response = await coinbaseApi({
              action: 'sell',
              accountId: 'primary',
              amount: tradeData.quantity.toString(),
              currency: tradeData.crypto_symbol
            });
            
            if (response.success) {
              alert(`Successfully sold ${tradeData.quantity} ${tradeData.crypto_symbol} via Coinbase!`);
              loadData();
              setSelectedCrypto(null);
              return;
            }
          }
        } catch (coinbaseError) {
          console.error('Coinbase trade error:', coinbaseError);
          alert('Coinbase trade failed. Falling back to demo mode.');
        }
      }

      // Demo mode trading
      const totalCost = tradeData.quantity * tradeData.price;
      
      if (tradeData.type === 'buy') {
        if (account.balance < totalCost) {
          alert("Insufficient funds!");
          return;
        }

        await Account.update(account.id, {
          balance: account.balance - totalCost
        });

        const existingHolding = holdings.find(h => h.crypto_symbol === tradeData.crypto_symbol);
        
        if (existingHolding) {
          const newQuantity = existingHolding.quantity + tradeData.quantity;
          const newAvgPrice = ((existingHolding.average_buy_price * existingHolding.quantity) + 
                             (tradeData.price * tradeData.quantity)) / newQuantity;
          
          await CryptoHolding.update(existingHolding.id, {
            quantity: newQuantity,
            average_buy_price: newAvgPrice,
            current_value: newQuantity * tradeData.price
          });
        } else {
          await CryptoHolding.create({
            crypto_symbol: tradeData.crypto_symbol,
            quantity: tradeData.quantity,
            average_buy_price: tradeData.price,
            current_value: totalCost
          });
        }

        await Transaction.create({
          amount: totalCost,
          type: "crypto_buy",
          description: `Bought ${tradeData.quantity} ${tradeData.crypto_symbol}`,
          category: "crypto",
          status: "completed",
          reference_id: `CRYPTO${Date.now()}`
        });

        alert(`Successfully bought ${tradeData.quantity} ${tradeData.crypto_symbol}!`);
      } else {
        const holding = holdings.find(h => h.crypto_symbol === tradeData.crypto_symbol);
        
        if (!holding || holding.quantity < tradeData.quantity) {
          alert("Insufficient cryptocurrency balance!");
          return;
        }

        await Account.update(account.id, {
          balance: account.balance + totalCost
        });

        const newQuantity = holding.quantity - tradeData.quantity;
        
        if (newQuantity > 0) {
          await CryptoHolding.update(holding.id, {
            quantity: newQuantity,
            current_value: newQuantity * tradeData.price
          });
        } else {
          await CryptoHolding.delete(holding.id);
        }

        await Transaction.create({
          amount: totalCost,
          type: "crypto_sell",
          description: `Sold ${tradeData.quantity} ${tradeData.crypto_symbol}`,
          category: "crypto",
          status: "completed",
          reference_id: `CRYPTO${Date.now()}`
        });

        alert(`Successfully sold ${tradeData.quantity} ${tradeData.crypto_symbol}!`);
      }

      setSelectedCrypto(null);
      loadData();
    } catch (error) {
      console.error("Error executing trade:", error);
      alert("Trade failed. Please try again.");
    }
  };

  if (isLoading) {
    return (
      <div className="flex items-center justify-center min-h-screen">
        <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-blue-600"></div>
      </div>
    );
  }

  const totalPortfolioValue = holdings.reduce((sum, h) => sum + (h.current_value || 0), 0);

  return (
    <div className="min-h-screen p-4 lg:p-8">
      <div className="max-w-7xl mx-auto space-y-8">
        <div className="flex flex-col lg:flex-row justify-between items-start lg:items-center gap-4">
          <div>
            <h1 className="text-3xl lg:text-4xl font-bold text-gray-900">Cryptocurrency</h1>
            <p className="text-gray-600 mt-2">Trade and manage your crypto portfolio</p>
            {useCoinbaseAPI && (
              <p className="text-green-600 text-sm font-semibold mt-1">🔗 Live Coinbase API Connected</p>
            )}
          </div>
          <div className="flex gap-3">
            <Button
              onClick={() => setUseCoinbaseAPI(!useCoinbaseAPI)}
              variant={useCoinbaseAPI ? "default" : "outline"}
              className="gap-2"
            >
              <Bitcoin className="w-4 h-4" />
              {useCoinbaseAPI ? 'Live Trading' : 'Demo Mode'}
            </Button>
            <Button
              onClick={loadRealCryptoPrices}
              disabled={pricesLoading}
              variant="outline"
              className="gap-2"
            >
              <TrendingUp className="w-4 h-4" />
              {pricesLoading ? 'Loading...' : 'Refresh Prices'}
            </Button>
          </div>
        </div>

        <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
          <div className="lg:col-span-2 space-y-6">
            {selectedChartCrypto && (
              <CryptoPriceChart
                symbol={selectedChartCrypto.symbol}
                name={selectedChartCrypto.name}
                currentPrice={selectedChartCrypto.current_price}
              />
            )}
            
            <AITradingInsights
              cryptocurrencies={cryptocurrencies}
              holdings={holdings}
              transactions={transactions}
            />
          </div>

          <div className="space-y-6">
            <CryptoInsights
              holdings={holdings}
              cryptocurrencies={cryptocurrencies}
              transactions={transactions}
            />
            
            <PriceAlerts cryptocurrencies={cryptocurrencies} />
          </div>
        </div>

        <CryptoPortfolio 
          holdings={holdings}
          totalValue={totalPortfolioValue}
          onSell={handleSell}
        />

        <Tabs defaultValue="market" className="w-full">
          <TabsList className="grid w-full grid-cols-5">
            <TabsTrigger value="market">Market</TabsTrigger>
            <TabsTrigger value="analytics">Analytics</TabsTrigger>
            <TabsTrigger value="strategies">Auto Trading</TabsTrigger>
            <TabsTrigger value="notifications">Alerts</TabsTrigger>
            <TabsTrigger value="history">History</TabsTrigger>
          </TabsList>
          
          <TabsContent value="market">
            <CryptoMarket 
              cryptocurrencies={cryptocurrencies}
              onBuy={handleBuy}
              isLoadingPrices={pricesLoading}
              onRefreshPrices={loadRealCryptoPrices}
              onSelectChart={(crypto) => setSelectedChartCrypto(crypto)}
            />
          </TabsContent>

          <TabsContent value="analytics">
            <CryptoAnalytics 
              holdings={holdings}
              transactions={transactions}
            />
          </TabsContent>

          <TabsContent value="strategies">
            <TradingStrategyManager />
          </TabsContent>

          <TabsContent value="notifications">
            <NotificationSettings 
              cryptos={cryptocurrencies}
            />
          </TabsContent>
          
          <TabsContent value="history">
            <CryptoHistory transactions={transactions} />
          </TabsContent>
        </Tabs>

        {selectedCrypto && (
          <CryptoTrade
            crypto={selectedCrypto}
            onTrade={handleTrade}
            onCancel={() => setSelectedCrypto(null)}
            balance={account?.balance || 0}
            holding={selectedCrypto.holding}
          />
        )}
      </div>
    </div>
  );
}

    setIsLoading(true);
    try {
      const user = await User.me();
      
      // Load user's holdings
      const userHoldings = await EToroHolding.filter({ created_by: user.email });
      setHoldings(userHoldings);

      // Load trade history

        50

      console.error("Error loading eToro data:", error);

  const handleTrade = async (tradeData) => {
    try {
      // Validate all required fields are present
      if (!tradeData.symbol || !tradeData.asset_name || !tradeData.trade_type || 
          !tradeData.quantity || !tradeData.price || !tradeData.total_amount) {
        return { success: false, error: "Missing required trade information" };

        symbol: tradeData.symbol,
        asset_name: tradeData.asset_name,
        trade_type: tradeData.trade_type,
        quantity: tradeData.quantity,
        price: tradeData.price,
        total_amount: tradeData.total_amount,

        status: "executed",
        order_type: tradeData.order_type || "market",
        trade_id: `ETORO${Date.now()}`
      });

        let newQuantity, newAvgPrice;
        
        if (tradeData.trade_type === 'buy') {
          newQuantity = existingHolding.quantity + tradeData.quantity;
          newAvgPrice = ((existingHolding.average_buy_price * existingHolding.quantity) +

        } else {
          newQuantity = existingHolding.quantity - tradeData.quantity;
          newAvgPrice = existingHolding.average_buy_price;
        }
        
        if (newQuantity > 0) {
          await EToroHolding.update(existingHolding.id, {
            quantity: newQuantity,
            average_buy_price: newAvgPrice,
            current_price: tradeData.price,
            total_value: newQuantity * tradeData.price,
            profit_loss: (tradeData.price - newAvgPrice) * newQuantity,
            profit_loss_percent: ((tradeData.price - newAvgPrice) / newAvgPrice) * 100
          });
        } else {
          await EToroHolding.delete(existingHolding.id);
        }
      } else if (tradeData.trade_type === 'buy') {
        await EToroHolding.create({
          symbol: tradeData.symbol,
          asset_name: tradeData.asset_name,
          asset_type: tradeData.asset_type || "stock",
          quantity: tradeData.quantity,
          average_buy_price: tradeData.price,
          current_price: tradeData.price,
          total_value: tradeData.quantity * tradeData.price,
          profit_loss: 0,
          profit_loss_percent: 0
        });
      }

              <div>
                <h3 className="text-lg font-bold text-yellow-900 mb-2">⚠️ DEMO TRADING MODE - NOT REAL MONEY</h3>
                <div className="text-sm text-yellow-800 space-y-2">

import React, { useState, useEffect } from 'react';
import { base44 } from '@/api/base44Client';

import { Button } from '@/components/ui/button';
import { Sparkles, Plus, MessageSquare, Trash2, ExternalLink } from 'lucide-react';
import JarvisChat from '../components/jarvis/JarvisChat';

export default function JarvisPage() {
  const [conversations, setConversations] = useState([]);
  const [activeConversation, setActiveConversation] = useState(null);
  const [isLoading, setIsLoading] = useState(true);

  useEffect(() => {
    loadConversations();
  }, []);

  const loadConversations = async () => {
    try {
      const convos = await base44.agents.listConversations({
        agent_name: 'jarvis'
      });
      setConversations(convos);

      if (convos.length > 0) {
        if (!activeConversation || !convos.some(c => c.id === activeConversation)) {
          setActiveConversation(convos[0].id);
        }
      } else {
        setActiveConversation(null);
      }
    } catch (error) {
      console.error('Error loading conversations:', error);
    } finally {
      setIsLoading(false);
    }
  };

  const createNewConversation = async () => {
    try {
      const newConvo = await base44.agents.createConversation({
        agent_name: 'jarvis',
        metadata: {
          name: `Chat ${new Date().toLocaleDateString()}`,
          description: 'New conversation with JARVIS'
        }
      });

      await loadConversations();
      setActiveConversation(newConvo.id);
    } catch (error) {
      console.error('Error creating conversation:', error);
      alert('Failed to create conversation');
    }
  };

  const deleteConversation = async (convId) => {
    if (!confirm('Delete this conversation? This action cannot be undone.')) return;

    try {
      if (activeConversation === convId) {
        setActiveConversation(null);
      }
      await loadConversations();
    } catch (error) {
      console.error('Error deleting conversation:', error);
      alert('Failed to delete conversation');
    }
  };

  const whatsappURL = base44.agents.getWhatsAppConnectURL('jarvis');

  if (isLoading) {
    return (
      <div className="flex items-center justify-center min-h-screen">
        <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-blue-600"></div>
      </div>
    );
  }

  return (
    <div className="min-h-screen p-4 lg:p-8">
      <div className="max-w-7xl mx-auto h-[calc(100vh-8rem)]">
        <div className="flex flex-col lg:flex-row gap-6 h-full">
          {/* Sidebar */}
          <div className="lg:w-80 flex-shrink-0">
            <Card className="h-full bg-white/80 backdrop-blur-sm shadow-xl">
              <CardHeader className="border-b">
                <div className="flex items-center justify-between">
                  <CardTitle className="flex items-center gap-2">
                    <div className="w-8 h-8 bg-gradient-to-r from-blue-600 to-indigo-600 rounded-lg flex items-center justify-center">
                      <Sparkles className="w-5 h-5 text-white" />
                    </div>
                    JARVIS
                  </CardTitle>
                  <Button
                    onClick={createNewConversation}
                    size="sm"
                    className="bg-gradient-to-r from-blue-600 to-indigo-600"
                  >
                    <Plus className="w-4 h-4" />
                  </Button>
                </div>
              </CardHeader>
              <CardContent className="p-4 space-y-4">
                <div className="bg-gradient-to-r from-blue-50 to-indigo-50 p-4 rounded-lg border border-blue-200">
                  <h3 className="font-semibold text-sm mb-2 text-blue-900">About JARVIS</h3>

import React, { useState, useEffect } from 'react';
import { base44 } from '@/api/base44Client';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { Button } from '@/components/ui/button';
import { Smartphone, Zap, CheckCircle2, XCircle, Radio, CreditCard, Shield } from 'lucide-react';
import { Badge } from '@/components/ui/badge';
import NFCReader from '../components/nfc/NFCReader';
import NFCSettings from '../components/nfc/NFCSettings';
import NFCHistory from '../components/nfc/NFCHistory';

export default function NFCPayments() {
  const [user, setUser] = useState(null);
  const [account, setAccount] = useState(null);
  const [cards, setCards] = useState([]);
  const [defaultCard, setDefaultCard] = useState(null);
  const [nfcEnabled, setNfcEnabled] = useState(false);
  const [isReading, setIsReading] = useState(false);
  const [transactions, setTransactions] = useState([]);
  const [isLoading, setIsLoading] = useState(true);

  useEffect(() => {
    loadData();
    checkNFCSupport();
  }, []);

  const checkNFCSupport = () => {
    if (window.PaymentRequest) {
      setNfcEnabled(true);
    } else {
      console.log('Payment Request API not supported');
    }
  };

  const loadData = async () => {
    try {
      const currentUser = await base44.auth.me();
      setUser(currentUser);

      const accounts = await base44.entities.Account.filter({ created_by: currentUser.email });
      if (accounts.length > 0) {
        setAccount(accounts[0]);
      }

      const userCards = await base44.entities.Card.filter({ 
        created_by: currentUser.email,
        card_type: 'payment'
      });
      setCards(userCards);
      
      if (userCards.length > 0) {
        setDefaultCard(userCards[0]);
      }

      const nfcTransactions = await base44.entities.Transaction.filter(
        { created_by: currentUser.email },
        '-created_date',
        50
      );
      const filtered = nfcTransactions.filter(t => t.description?.includes('NFC Payment'));
      setTransactions(filtered);

    } catch (error) {
      console.error('Error loading data:', error);
    } finally {
      setIsLoading(false);
    }
  };

  const handleNFCPayment = async (paymentData) => {
    try {
      if (!account || !defaultCard) {
        alert('No payment method available');
        return { success: false, error: 'No payment method' };
      }

      // Process real payment through Stripe
      const { data: response } = await base44.functions.invoke('processNFCPayment', {
        amount: paymentData.amount,
        merchant: paymentData.merchant,
        category: paymentData.category || 'other',
        card_id: defaultCard.id
      });

      if (response.success) {
        await loadData();
        return { success: true, transaction: response.transaction };
      } else {
        return { success: false, error: response.error };
      }
    } catch (error) {
      console.error('Error processing NFC payment:', error);
      return { success: false, error: error.message };
    }
  };

  if (isLoading) {
    return (
      <div className="flex items-center justify-center min-h-screen">
        <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-blue-600"></div>
      </div>
    );
  }

  return (
    <div className="min-h-screen p-4 lg:p-8">
      <div className="max-w-7xl mx-auto space-y-8">
        <div>
          <h1 className="text-3xl lg:text-4xl font-bold text-gray-900">NFC Payments</h1>
          <p className="text-gray-600 mt-2">Tap to pay anywhere NFC is accepted</p>
        </div>

        {/* Payment Status Banner */}
        <Card className={`border-2 ${nfcEnabled ? 'bg-gradient-to-r from-green-50 to-emerald-50 border-green-200' : 'bg-gradient-to-r from-orange-50 to-red-50 border-orange-200'}`}>
          <CardContent className="p-6">
            <div className="flex items-center justify-between">
              <div className="flex items-center gap-4">
                {nfcEnabled ? (
                  <>
                    <div className="w-12 h-12 bg-green-500 rounded-full flex items-center justify-center">
                      <Radio className="w-6 h-6 text-white" />
                    </div>
                    <div>
                      <h3 className="font-bold text-green-900">Google Pay & Apple Pay Ready</h3>
                      <p className="text-sm text-green-800">Tap to pay with your digital wallet</p>
                    </div>
                  </>
                ) : (
                  <>
                    <div className="w-12 h-12 bg-orange-500 rounded-full flex items-center justify-center">
                      <XCircle className="w-6 h-6 text-white" />
                    </div>
                    <div>
                      <h3 className="font-bold text-orange-900">Digital Wallets Not Available</h3>
                      <p className="text-sm text-orange-800">Your browser doesn't support digital payments</p>
                    </div>
                  </>
                )}
              </div>
              {nfcEnabled && (
                <Badge className="bg-green-600 text-white">
                  <Zap className="w-3 h-3 mr-1" />
                  Active
                </Badge>
              )}
            </div>
          </CardContent>
        </Card>

        <div className="grid lg:grid-cols-3 gap-8">
          {/* Main Payment Area */}
          <div className="lg:col-span-2 space-y-6">
            {/* Active Card */}
            <Card className="bg-gradient-to-br from-blue-600 to-indigo-600 text-white border-0 shadow-xl">
              <CardContent className="p-6">
                <div className="flex items-center justify-between mb-6">
                  <div>
                    <p className="text-blue-200 text-sm">Active Card</p>
                    <h3 className="text-2xl font-bold mt-1">
                      {defaultCard?.card_name || 'No Card Selected'}
                    </h3>
                  </div>

import React, { useState, useEffect } from "react";
import { PayStub } from "@/entities/PayStub";
import { User } from "@/entities/User";
import { FileText, Plus, Download, Eye } from "lucide-react";
import { Button } from "@/components/ui/button";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";

import PayStubGenerator from "../components/paystubs/PayStubGenerator";
import PayStubList from "../components/paystubs/PayStubList";
import PayStubPreview from "../components/paystubs/PayStubPreview";
import AdBanner from "../components/ads/AdBanner";
import { sendPayStubGenerated } from "../components/notifications/EmailNotifications";

export default function PayStubs() {
  const [payStubs, setPayStubs] = useState([]);
  const [showGenerator, setShowGenerator] = useState(false);
  const [selectedPayStub, setSelectedPayStub] = useState(null);
  const [showPreview, setShowPreview] = useState(false);
  const [isLoading, setIsLoading] = useState(true);
  const [user, setUser] = useState(null); // Add state for user

  useEffect(() => {
    loadPayStubs();
  }, []);

  const loadPayStubs = async () => {
    setIsLoading(true);
    try {
      const currentUser = await User.me();
      setUser(currentUser); // Store the current user
      const userPayStubs = await PayStub.filter(
        { created_by: currentUser.email }, // Use currentUser for filtering
        "-created_date"
      );
      setPayStubs(userPayStubs);
    } catch (error) {
      console.error("Error loading pay stubs:", error);
      setUser(null); // Ensure user is null on error
    } finally {
      setIsLoading(false);
    }
  };

  const handleGeneratePayStub = async (payStubData) => {
    try {
      const newPayStub = await PayStub.create(payStubData);
      
      // Send email notification
      if (user && newPayStub) { // Ensure user and newPayStub are available
        await sendPayStubGenerated(user, newPayStub);
      }
      
      setShowGenerator(false);
      loadPayStubs();
      alert('Pay stub generated successfully! Check your email.');
    } catch (error) {
      console.error("Error generating pay stub:", error);
      alert("Failed to generate pay stub. Please check all fields and try again.");
    }
  };

  const handleViewPayStub = (payStub) => {
    setSelectedPayStub(payStub);
    setShowPreview(true);
  };

  const handleDeletePayStub = async (payStubId) => {
    if (window.confirm('Are you sure you want to delete this pay stub? This action cannot be undone.')) {
      try {
        await PayStub.delete(payStubId);
        setShowPreview(false); // Close preview if the deleted stub was open
        setSelectedPayStub(null); // Clear selected stub
        loadPayStubs(); // Reload the list of pay stubs
        alert('Pay stub deleted successfully!');
      } catch (error) {
        console.error("Error deleting pay stub:", error);
        alert("Failed to delete pay stub. Please try again.");
      }
    }
  };

  return (
    <div className="min-h-screen p-4 lg:p-8">
      <div className="max-w-7xl mx-auto space-y-8">
        {/* Header */}
        <div className="flex flex-col lg:flex-row justify-between items-start lg:items-center gap-4">
          <div>
            <h1 className="text-3xl lg:text-4xl font-bold text-gray-900">Pay Stub Generator</h1>
            <p className="text-gray-600 mt-2">Create professional pay stubs for your employees or yourself</p>
          </div>
          <Button 
            onClick={() => setShowGenerator(true)}
            className="bg-gradient-to-r from-green-600 to-emerald-600 gap-2"
          >
            <Plus className="w-4 h-4" />
            Generate Pay Stub
          </Button>
        </div>

        {/* Quick Stats */}
        <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
          <Card className="bg-white/80 backdrop-blur-sm border-0 shadow-xl">
            <CardContent className="p-6">
              <div className="flex items-center gap-4">
                <div className="w-12 h-12 bg-blue-100 rounded-full flex items-center justify-center">
                  <FileText className="w-6 h-6 text-blue-600" />
                </div>
                <div>
                  <p className="text-sm text-gray-600">Total Pay Stubs</p>
                  <p className="text-2xl font-bold text-gray-900">{payStubs.length}</p>
                </div>
              </div>
            </CardContent>
          </Card>

          <Card className="bg-white/80 backdrop-blur-sm border-0 shadow-xl">
            <CardContent className="p-6">
              <div className="flex items-center gap-4">
                <div className="w-12 h-12 bg-green-100 rounded-full flex items-center justify-center">
                  <Download className="w-6 h-6 text-green-600" />
                </div>
                <div>
                  <p className="text-sm text-gray-600">This Month</p>
                  <p className="text-2xl font-bold text-gray-900">
                    {payStubs.filter(stub => {
                      const stubDate = new Date(stub.created_date);
                      const now = new Date();
                      return stubDate.getMonth() === now.getMonth() && stubDate.getFullYear() === now.getFullYear();
                    }).length}
                  </p>
                </div>
              </div>
            </CardContent>
          </Card>

          <Card className="bg-white/80 backdrop-blur-sm border-0 shadow-xl">
            <CardContent className="p-6">
              <div className="flex items-center gap-4">
                <div className="w-12 h-12 bg-purple-100 rounded-full flex items-center justify-center">
                  <Eye className="w-6 h-6 text-purple-600" />
                </div>
                <div>
                  <p className="text-sm text-gray-600">YTD Gross Pay</p>
                  <p className="text-2xl font-bold text-gray-900">
                    ${payStubs.reduce((sum, stub) => sum + (stub.gross_pay || 0), 0).toLocaleString()}
                  </p>
                </div>
              </div>
            </CardContent>
          </Card>
        </div>

        {/* Ad Banner */}
        <AdBanner
          size="medium"
          title="Payroll Made Simple"
          description="Automate your payroll, tax filing, and employee payments. Try free for 30 days!"
          imageUrl="https://images.unsplash.com/photo-1554224154-26032ffc0d07?w=200&h=200&fit=crop"
          ctaText="Start Free Trial"
          ctaLink="#"
        />

        {/* Pay Stub List */}
        <PayStubList 
          payStubs={payStubs} 
          isLoading={isLoading}
          onViewPayStub={handleViewPayStub}
          onDeletePayStub={handleDeletePayStub}

import { Button } from "@/components/ui/button";

import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";

import PortfolioAnalytics from "../components/portfolio/PortfolioAnalytics";

export default function Portfolio() {
  const [user, setUser] = useState(null);
  const [stockHoldings, setStockHoldings] = useState([]);
  const [cryptoHoldings, setCryptoHoldings] = useState([]);
  const [allTransactions, setAllTransactions] = useState([]);
  const [isLoading, setIsLoading] = useState(true);
  const [refreshing, setRefreshing] = useState(false);

  useEffect(() => {
    loadPortfolioData();
  }, []);

  const loadPortfolioData = async () => {
    try {
      const currentUser = await base44.auth.me();
      setUser(currentUser);

      const [stocks, crypto, etoroTrades, cryptoTxns] = await Promise.all([
        EToroHolding.filter({ created_by: currentUser.email }),
        CryptoHolding.filter({ created_by: currentUser.email }),
        EToroTrade.filter({ created_by: currentUser.email }),
        Transaction.filter({ created_by: currentUser.email, category: 'crypto' })
      ]);

      setStockHoldings(stocks);
      setCryptoHoldings(crypto);
      setAllTransactions([...etoroTrades, ...cryptoTxns]);
    } catch (error) {
      console.error("Error loading portfolio:", error);
    } finally {
      setIsLoading(false);
    }
  };

  const handleRefresh = async () => {
    setRefreshing(true);
    await loadPortfolioData();
    setRefreshing(false);
  };

  if (isLoading) {
    return (
      <div className="flex items-center justify-center min-h-screen">
        <div className="text-center">
          <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-blue-600 mx-auto mb-4"></div>
          <p className="text-gray-600">Analyzing your portfolio...</p>
        </div>
      </div>
    );
  }

  const totalStockValue = stockHoldings.reduce((sum, h) => sum + (h.total_value || 0), 0);
  const totalCryptoValue = cryptoHoldings.reduce((sum, h) => sum + (h.current_value || 0), 0);
  const totalPortfolioValue = totalStockValue + totalCryptoValue;

  return (
    <div className="min-h-screen p-4 lg:p-8">
      <div className="max-w-7xl mx-auto space-y-8">
        {/* Header */}
        <div className="flex flex-col lg:flex-row justify-between items-start lg:items-center gap-4">
          <div>
            <h1 className="text-3xl lg:text-4xl font-bold text-gray-900">
              Portfolio Analytics
            </h1>
            <p className="text-gray-600 mt-2">
              Comprehensive analysis of your investment performance
            </p>
          </div>
          <Button
            onClick={handleRefresh}
            disabled={refreshing}
            className="bg-gradient-to-r from-blue-600 to-indigo-600 gap-2"
          >
            <RefreshCw className={`w-4 h-4 ${refreshing ? 'animate-spin' : ''}`} />
            {refreshing ? 'Refreshing...' : 'Refresh Data'}
          </Button>
        </div>

        {/* Portfolio Summary */}
        <Card className="bg-gradient-to-br from-slate-900 via-blue-900 to-indigo-900 text-white border-0 shadow-2xl">
          <CardHeader>
            <CardTitle className="flex items-center gap-3">
              <PieChart className="w-6 h-6" />
              Total Portfolio Value
            </CardTitle>
          </CardHeader>
          <CardContent>
            <div className="space-y-4">
              <div>
                <p className="text-4xl font-bold">
                  ${totalPortfolioValue.toLocaleString(undefined, { maximumFractionDigits: 2 })}
                </p>
              </div>
              <div className="grid grid-cols-2 gap-4 pt-4 border-t border-white/10">
                <div>
                  <p className="text-blue-200 text-sm">Stocks & Securities</p>
                  <p className="text-xl font-semibold mt-1">
                    ${totalStockValue.toLocaleString(undefined, { maximumFractionDigits: 2 })}

                  <p className="text-xs text-blue-300 mt-1">
                    {totalPortfolioValue > 0 ? ((totalStockValue / totalPortfolioValue) * 100).toFixed(1) : 0}% of portfolio
                  </p>
                </div>
                <div>
                  <p className="text-blue-200 text-sm">Cryptocurrency</p>

import {

  Download

import ProfileHeader from "../components/profile/ProfileHeader";

import LegalSection from "../components/profile/LegalSection";
import ExportManager from "../components/export/ExportManager";

export default function Profile() {
  const [user, setUser] = useState(null);
  const [isLoading, setIsLoading] = useState(true);
  const [activeTab, setActiveTab] = useState("personal");
  const [transactions, setTransactions] = useState([]);
  const [account, setAccount] = useState(null);
  const [cards, setCards] = useState([]);
  const [cryptoHoldings, setCryptoHoldings] = useState([]);
  const navigate = useNavigate();

  useEffect(() => {
    loadUserData();
  }, []);

  const loadUserData = async () => {
    setIsLoading(true);
    try {
      const currentUser = await base44.auth.me();
      setUser(currentUser);
      
      // Load additional data for export
      if (currentUser && currentUser.email) {
        const [userTransactions, accounts, userCards, crypto] = await Promise.all([
          Transaction.filter({ created_by: currentUser.email }),
          Account.filter({ created_by: currentUser.email }),
          Card.filter({ created_by: currentUser.email }),
          CryptoHolding.filter({ created_by: currentUser.email })
        ]);
        
        setTransactions(userTransactions);
        setAccount(accounts.length > 0 ? accounts[0] : null);
        setCards(userCards);
        setCryptoHoldings(crypto);
      }
    } catch (error) {
      console.error("Error loading user data:", error);
    } finally {
      setIsLoading(false);
    }
  };

  const handleUpdateUser = async (data) => {
    if (!user) return;
    try {
      await base44.auth.updateMe(data);
      await loadUserData();
    } catch (error) {
      console.error("Failed to update user:", error);
    }
  };

  const handleLogout = async () => {
    await base44.auth.logout();
    navigate(createPageUrl("Landing"));
  };
  
  const tabs = [
    { id: 'personal', label: 'Personal Info', icon: UserCircle2 },
    { id: 'security', label: 'Security', icon: ShieldCheck },
    { id: 'privacy', label: 'Privacy', icon: Lock },
    { id: 'notifications', label: 'Notifications', icon: Bell },
    { id: 'export', label: 'Export Data', icon: Download },
    { id: 'legal', label: 'Legal', icon: FileText },
  ];

  const renderContent = () => {
    switch (activeTab) {
      case 'personal':
        return <PersonalInfo user={user} onUpdate={handleUpdateUser} />;
      case 'security':
        return (
          <div className="space-y-6">
            <SecurityDashboard user={user} />
            <SecuritySettings user={user} onUpdate={handleUpdateUser} />
          </div>
        );
      case 'privacy':
        return <PrivacyControls user={user} onUpdate={handleUpdateUser} />;
      case 'notifications':
        return <NotificationSettings user={user} onUpdate={handleUpdateUser} />;
      case 'export':
        return <ExportManager 
          transactions={transactions}
          account={account}
          cards={cards}
          cryptoHoldings={cryptoHoldings}
          user={user}
        />;
      case 'legal':
        return <LegalSection />;
      default:
        return null;
    }
  };

  if (isLoading) {
    return (
      <div className="flex items-center justify-center min-h-screen">
        <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-blue-600"></div>
      </div>
    );
  }

  return (
    <div className="min-h-screen p-4 lg:p-8">
      <div className="max-w-7xl mx-auto space-y-8">
        {/* Header */}
        <ProfileHeader user={user} onLogout={handleLogout} />
        
        <div className="grid lg:grid-cols-4 gap-8">
          {/* Sidebar Navigation */}
          <div className="lg:col-span-1">
            <div className="bg-white/80 backdrop-blur-sm rounded-2xl shadow-xl p-4 sticky top-24">
              <nav className="space-y-1">
                {tabs.map((tab) => (
                  <button
                    key={tab.id}
                    onClick={() => setActiveTab(tab.id)}
                    className={`w-full flex items-center gap-3 px-4 py-3 rounded-lg text-left transition-colors ${
                      activeTab === tab.id
                        ? 'bg-gradient-to-r from-blue-600 to-indigo-600 text-white shadow-lg'
                        : 'text-gray-700 hover:bg-gray-100'
                    }`}
                  >
                    <tab.icon className="w-5 h-5" />
                    <span className="font-medium">{tab.label}</span>
                  </button>
                ))}
              </nav>
            </div>
          </div>
          
          {/* Main Content */}
          <div className="lg:col-span-3">
            {renderContent()}
          </div>
        </div>
      </div>
    </div>
  );
}

  const [currentReturn, setCurrentReturn] = useState(null);
  const [isLoading, setIsLoading] = useState(true);
  const [transactions, setTransactions] = useState([]);
  const [activeTab, setActiveTab] = useState("prepare");

  useEffect(() => {
    loadTaxData();
  }, []);

  const loadTaxData = async () => {
    setIsLoading(true);
    try {
      const user = await User.me();
      
      // Load tax returns
      const userTaxReturns = await TaxDocument.filter(
        { created_by: user.email },
        "-tax_year"
      );
      setTaxReturns(userTaxReturns);

      // Load transactions for tax calculation
      const userTransactions = await Transaction.filter(
        { created_by: user.email },
        "-created_date"
      );
      setTransactions(userTransactions);

      // Set current year return if it exists
      const currentYear = new Date().getFullYear();
      const currentYearReturn = userTaxReturns.find(ret => ret.tax_year === currentYear - 1);
      if (currentYearReturn) {
        setCurrentReturn(currentYearReturn);
      }

    } catch (error) {
      console.error("Error loading tax data:", error);
    } finally {
      setIsLoading(false);
    }
  };

  const handleCreateTaxReturn = async (taxData) => {
    try {
      const newReturn = await TaxDocument.create(taxData);
      setCurrentReturn(newReturn);
      loadTaxData();
      alert("Tax return saved successfully!");
    } catch (error) {
      console.error("Error creating tax return:", error);
      alert("Failed to save tax return. Please try again.");
    }
  };

  const handleFileTaxReturn = async (returnId) => {
    try {
      await TaxDocument.update(returnId, { status: "filed" });
      loadTaxData();
      alert("Tax return filed successfully! You will receive a confirmation email shortly.");
    } catch (error) {
      console.error("Error filing tax return:", error);
      alert("Failed to file tax return. Please try again.");
    }
  };

  const handleStartNewReturn = () => {
    setCurrentReturn(null);
    setActiveTab("prepare");
  };

  const currentYearReturn = taxReturns.find(ret => ret.tax_year === currentYear - 1);
  const isCurrentYearFiled = currentYearReturn?.status === 'filed';

  if (isLoading) {
    return (
      <div className="flex items-center justify-center min-h-screen">
        <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-blue-600"></div>
      </div>

    <div className="min-h-screen p-4 lg:p-8">
      <div className="max-w-7xl mx-auto space-y-8">
        {/* Header */}
        <div className="flex flex-col lg:flex-row justify-between items-start lg:items-center gap-4">
          <div>
            <h1 className="text-3xl lg:text-4xl font-bold text-gray-900">Tax Center</h1>
            <p className="text-gray-600 mt-2">Prepare and file your taxes with ease</p>
          </div>
        </div>

        {/* Tax Season Banner */}
        <div className="bg-gradient-to-r from-blue-600 to-indigo-600 text-white rounded-2xl p-6">
          <div className="flex flex-col lg:flex-row items-center gap-4">
            <div className="w-16 h-16 bg-white/20 rounded-full flex items-center justify-center flex-shrink-0">
              <Calculator className="w-8 h-8" />
            </div>
            <div className="flex-1 text-center lg:text-left">
              <h3 className="text-xl font-bold">Tax Year {currentYear - 1}</h3>
              <p className="text-blue-100">File by April 15, {currentYear} to avoid penalties</p>
            </div>
            <div className="text-center lg:text-right">
              {isCurrentYearFiled ? (
                <div className="flex items-center gap-2">
                  <CheckCircle className="w-5 h-5 text-green-400" />

              ) : (
                <Button

                  Start Your Return
                </Button>

            </div>
          </div>
        </div>

        {/* Quick Stats */}
        <div className="grid grid-cols-1 md:grid-cols-4 gap-6">
          <Card className="bg-white/80 backdrop-blur-sm border-0 shadow-xl">
            <CardContent className="p-6">

                <div>

import React, { useState, useEffect } from "react";
import { Transaction, User } from "@/entities/all";
import { History, Download } from "lucide-react";
import { Button } from "@/components/ui/button";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";

import TransactionList from "../components/transactions/TransactionList";
import TransactionStats from "../components/transactions/TransactionStats";
import AdvancedSearch from "../components/transactions/AdvancedSearch";
import ImportBankStatement from "../components/transactions/ImportBankStatement";
import ReceiptScanner from "../components/transactions/ReceiptScanner";
import SmartCategorization from "../components/transactions/SmartCategorization";

export default function Transactions() {
  const [transactions, setTransactions] = useState([]);
  const [filteredTransactions, setFilteredTransactions] = useState([]);
  const [isLoading, setIsLoading] = useState(true);
  const [user, setUser] = useState(null);
  const [savedFilters, setSavedFilters] = useState([]);
  const [activeTab, setActiveTab] = useState("all");

  useEffect(() => {
    loadTransactions();
    loadSavedFilters();
  }, []);

  const loadTransactions = async () => {
    setIsLoading(true);
    try {
      const currentUser = await User.me();
      setUser(currentUser);

      const userTransactions = await Transaction.filter(
        { created_by: currentUser.email },
        "-created_date"
      );
      setTransactions(userTransactions);
      setFilteredTransactions(userTransactions);
    } catch (error) {
      console.error("Error loading transactions:", error);
    } finally {
      setIsLoading(false);
    }
  };

  const loadSavedFilters = () => {
    // Load from localStorage
    const saved = localStorage.getItem('savedTransactionFilters');
    if (saved) {
      try {
        setSavedFilters(JSON.parse(saved));
      } catch (e) {
        console.error("Error loading saved filters:", e);
      }
    }
  };

  const handleSearch = (searchParams) => {
    let filtered = [...transactions];

    // Text search
    if (searchParams.searchTerm) {
      const term = searchParams.searchTerm.toLowerCase();
      filtered = filtered.filter(t => 
        t.description?.toLowerCase().includes(term) ||
        t.merchant?.toLowerCase().includes(term) ||
        t.amount.toString().includes(term)
      );
    }

    // Type filter
    if (searchParams.type && searchParams.type !== 'all') {
      filtered = filtered.filter(t => t.type === searchParams.type);
    }

    // Category filter
    if (searchParams.category && searchParams.category !== 'all') {
      filtered = filtered.filter(t => t.category === searchParams.category);
    }

    // Status filter
    if (searchParams.status && searchParams.status !== 'all') {
      filtered = filtered.filter(t => t.status === searchParams.status);
    }

    // Amount range
    if (searchParams.minAmount) {
      filtered = filtered.filter(t => Math.abs(t.amount) >= parseFloat(searchParams.minAmount));
    }
    if (searchParams.maxAmount) {
      filtered = filtered.filter(t => Math.abs(t.amount) <= parseFloat(searchParams.maxAmount));
    }

    // Date range
    if (searchParams.dateFrom) {
      filtered = filtered.filter(t => new Date(t.created_date) >= searchParams.dateFrom);
    }
    if (searchParams.dateTo) {
      filtered = filtered.filter(t => new Date(t.created_date) <= searchParams.dateTo);
    }

    // Merchant filter
    if (searchParams.merchant) {
      const merchantTerm = searchParams.merchant.toLowerCase();
      filtered = filtered.filter(t => t.merchant?.toLowerCase().includes(merchantTerm));
    }

    setFilteredTransactions(filtered);
  };

  const handleSaveFilter = (filter) => {
    const updated = [...savedFilters, filter];
    setSavedFilters(updated);
    localStorage.setItem('savedTransactionFilters', JSON.stringify(updated));
  };

  const handleLoadFilter = (savedFilter) => {
    handleSearch(savedFilter.filters);
  };

  const handleUpdateTransaction = async (transactionId, updates) => {
    try {
      await Transaction.update(transactionId, updates);
      loadTransactions();
    } catch (error) {
      console.error("Error updating transaction:", error);
    }
  };

  const handleExportCSV = () => {
    const csvContent = [
      ['Date', 'Description', 'Amount', 'Type', 'Category', 'Merchant', 'Status'],
      ...filteredTransactions.map(t => [
        new Date(t.created_date).toLocaleDateString(),
        t.description,
        t.amount,
        t.type,
        t.category || '',
        t.merchant || '',
        t.status
      ])
    ].map(row => row.join(',')).join('\n');

    const blob = new Blob([csvContent], { type: 'text/csv' });
    const url = window.URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `transactions_${new Date().toISOString().split('T')[0]}.csv`;
    a.click();
    window.URL.revokeObjectURL(url);
  };

  const handleExportJSON = () => {
    const jsonContent = JSON.stringify(filteredTransactions, null, 2);
    const blob = new Blob([jsonContent], { type: 'application/json' });
    const url = window.URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `transactions_${new Date().toISOString().split('T')[0]}.json`;
    a.click();
    window.URL.revokeObjectURL(url);
  };

  const getTabTransactions = () => {
    switch (activeTab) {
      case 'income':
        return filteredTransactions.filter(t => ['credit', 'transfer_in'].includes(t.type));
      case 'expenses':
        return filteredTransactions.filter(t => ['debit', 'transfer_out'].includes(t.type));
      default:
        return filteredTransactions;
    }
  };

  if (isLoading) {
    return (
      <div className="flex items-center justify-center min-h-screen">
        <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-blue-600"></div>
      </div>
    );
  }

  return (
    <div className="min-h-screen p-4 lg:p-8">
      <div className="max-w-7xl mx-auto space-y-8">
        {/* Header */}
        <div className="flex flex-col lg:flex-row justify-between items-start lg:items-center gap-4">
          <div>
            <h1 className="text-3xl lg:text-4xl font-bold text-gray-900">Transactions</h1>
            <p className="text-gray-600 mt-2">
              {filteredTransactions.length} {filteredTransactions.length === 1 ? 'transaction' : 'transactions'}
              {filteredTransactions.length !== transactions.length && ` (filtered from ${transactions.length})`}
            </p>
          </div>
          <div className="flex gap-3">
            <Button onClick={handleExportCSV} variant="outline" className="gap-2">
              <Download className="w-4 h-4" />
              Export CSV
            </Button>
            <Button onClick={handleExportJSON} variant="outline" className="gap-2">
              <Download className="w-4 h-4" />
              Export JSON
            </Button>
          </div>
        </div>

        {/* Stats */}
        <TransactionStats transactions={filteredTransactions} />

        {/* AI Features Row */}
        <div className="grid lg:grid-cols-3 gap-6">
          <SmartCategorization 
            transactions={transactions}
            onCategorized={handleUpdateTransaction}
          />
          <ReceiptScanner onTransactionCreated={loadTransactions} />
          <ImportBankStatement onImportComplete={loadTransactions} />
        </div>

        {/* Advanced Search */}
        <AdvancedSearch
          onSearch={handleSearch}
          savedFilters={savedFilters}
          onSaveFilter={handleSaveFilter}
          onLoadFilter={handleLoadFilter}
        />

        {/* Transactions List with Tabs */}
        <Card className="bg-white/80 backdrop-blur-sm border-0 shadow-xl">
          <CardHeader>
            <CardTitle className="flex items-center gap-2">
              <History className="w-6 h-6" />
              All Transactions
            </CardTitle>
          </CardHeader>
          <CardContent>
            <Tabs value={activeTab} onValueChange={setActiveTab} className="w-full">
              <TabsList className="grid w-full grid-cols-3 mb-6">
                <TabsTrigger value="all">
                  All ({filteredTransactions.length})
                </TabsTrigger>
                <TabsTrigger value="income">
                  Income ({filteredTransactions.filter(t => ['credit', 'transfer_in'].includes(t.type)).length})
                </TabsTrigger>
                <TabsTrigger value="expenses">
                  Expenses ({filteredTransactions.filter(t => ['debit', 'transfer_out'].includes(t.type)).length})
                </TabsTrigger>
              </TabsList>

              <TabsContent value={activeTab}>
                <TransactionList 
                  transactions={getTabTransactions()}
                  onUpdate={handleUpdateTransaction}
                />
              </TabsContent>
            </Tabs>
          </CardContent>
        </Card>
      </div>
    </div>
  );
}

import React, { useState, useEffect } from "react";
import { base44 } from "@/api/base44Client";
import { Send, ArrowDownToLine, User as UserIcon } from "lucide-react";
import { Button } from "@/components/ui/button";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";

import SendMoney from "../components/transfer/SendMoney";
import RequestMoney from "../components/transfer/RequestMoney";
import RecentTransfers from "../components/transfer/RecentTransfers";
import QuickContacts from "../components/transfer/QuickContacts";
import AdCard from "../components/ads/AdCard";
import { sendTransactionAlert, sendLargeTransactionAlert } from "../components/notifications/EmailNotifications";
import TransactionAuthenticator from "../components/security/TransactionAuthenticator";
import { checkTransactionVelocity } from "../components/security/fraudUtils";

export default function Transfer() {
  const [user, setUser] = useState(null);
  const [account, setAccount] = useState(null);
  const [transfers, setTransfers] = useState([]);
  const [isLoading, setIsLoading] = useState(true);
  const [pendingTransfer, setPendingTransfer] = useState(null);
  const [showAuthDialog, setShowAuthDialog] = useState(false);

  useEffect(() => {
    loadTransferData();
  }, []);

  const loadTransferData = async () => {
    try {
      const currentUser = await base44.auth.me();
      setUser(currentUser);

      const accounts = await base44.entities.Account.filter({ created_by: currentUser.email });
      if (accounts.length > 0) {
        setAccount(accounts[0]);
      }

      const transferTxns = await base44.entities.Transaction.filter(
        { created_by: currentUser.email },
        "-created_date",
        20
      );
      const filtered = transferTxns.filter(t => 
        t.type === "transfer_in" || t.type === "transfer_out"
      );
      setTransfers(filtered);

    } catch (error) {
      console.error("Error loading transfer data:", error);
    } finally {
      setIsLoading(false);
    }
  };

  const handleSendMoney = async (transferData) => {
    try {
      if (!account) {
        alert("Account not found. Please refresh the page.");
        return;
      }

      if (account.balance < transferData.amount) {
        alert("Insufficient funds!");
        return;
      }

      // Check fraud detection
      const velocityCheck = await checkTransactionVelocity(user.email, transferData.amount);
      
      if (velocityCheck.isHighRisk) {
        const proceed = confirm(
          `Security Alert:\n\n${velocityCheck.risks.join('\n')}\n\nDo you want to proceed?`
        );
        if (!proceed) return;
      }

      // Require authentication for large transfers or high-risk transactions
      if (velocityCheck.requiresVerification) {
        setPendingTransfer(transferData);
        setShowAuthDialog(true);
        return;
      }

      await processTransfer(transferData);
    } catch (error) {
      console.error("Error sending money:", error);
      alert("Failed to send money. Please try again.");
    }
  };

  const processTransfer = async (transferData) => {
    let description = "";
    if (transferData.sendMethod === "bank") {
      description = `Sent to ${transferData.recipient} (${transferData.bankName || 'Bank Account'} ****${transferData.accountNumber.slice(-4)})`;
    } else {
      description = `Sent to ${transferData.recipient}`;
    }

    if (transferData.note) {
      description += ` - ${transferData.note}`;
    }

    const newTransaction = await base44.entities.Transaction.create({
      amount: transferData.amount,
      type: "transfer_out",
      description: description,
      category: "transfer",
      merchant: transferData.sendMethod === "bank" ? transferData.bankName : transferData.recipient,
      status: "completed",
      reference_id: `TXF${Date.now()}`
    });

    await base44.entities.Account.update(account.id, {
      balance: account.balance - transferData.amount
    });

    if (user) {
      await sendTransactionAlert(user, newTransaction);
      await sendLargeTransactionAlert(user, newTransaction, 500);
    }

    loadTransferData();
    alert('Money sent successfully!');
  };

  const handleAuthenticated = async (authenticated) => {
    setShowAuthDialog(false);
    if (authenticated && pendingTransfer) {
      await processTransfer(pendingTransfer);
      setPendingTransfer(null);
    }
  };

  const handleRequestMoney = async (requestData) => {
    try {
      const newTransaction = await base44.entities.Transaction.create({
        amount: requestData.amount,
        type: "transfer_in",
        description: `Requested from ${requestData.from}`,
        category: "transfer",
        status: "pending",
        reference_id: `REQ${Date.now()}`
      });

      if (user) {
        await sendTransactionAlert(user, newTransaction);
      }

      loadTransferData();
      alert('Money request sent successfully!');
    } catch (error) {
      console.error("Error requesting money:", error);
      alert("Failed to request money. Please try again.");
    }
  };

  if (isLoading) {
    return (
      <div className="flex items-center justify-center min-h-screen">
        <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-blue-600"></div>
      </div>
    );
  }

  return (
    <div className="min-h-screen p-4 lg:p-8">
      <div className="max-w-6xl mx-auto space-y-8">
        <div>
          <h1 className="text-3xl lg:text-4xl font-bold text-gray-900">Send & Receive</h1>
          <p className="text-gray-600 mt-2">Transfer money instantly to friends, family, or any bank account</p>
        </div>

        <Card className="bg-gradient-to-r from-blue-600 to-indigo-600 text-white border-0 shadow-xl">
          <CardContent className="p-6">
            <div className="flex items-center justify-between">
              <div>
                <p className="text-blue-200 text-sm font-medium">Available Balance</p>
                <p className="text-3xl font-bold mt-1">${account?.balance?.toLocaleString()}</p>
              </div>
              <div className="w-16 h-16 bg-white/20 rounded-full flex items-center justify-center">
                <UserIcon className="w-8 h-8" />
              </div>
            </div>
          </CardContent>
        </Card>

        <div className="grid lg:grid-cols-3 gap-8">
          <div className="lg:col-span-2 space-y-6">
            <Tabs defaultValue="send" className="w-full">
              <TabsList className="grid w-full grid-cols-2 mb-6">
                <TabsTrigger value="send" className="flex items-center gap-2">
                  <Send className="w-4 h-4" />
                  Send Money
                </TabsTrigger>
                <TabsTrigger value="request" className="flex items-center gap-2">
                  <ArrowDownToLine className="w-4 h-4" />
                  Request Money
                </TabsTrigger>
              </TabsList>
              
              <TabsContent value="send">
                <SendMoney onSendMoney={handleSendMoney} />
              </TabsContent>
              
              <TabsContent value="request">
                <RequestMoney onRequestMoney={handleRequestMoney} />
              </TabsContent>
            </Tabs>

            <RecentTransfers transfers={transfers} />
          </div>

          <div className="space-y-6">
