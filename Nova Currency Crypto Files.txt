import React, { useState, useEffect } from "react";
import { base44 } from "@/api/base44Client";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { Badge } from "@/components/ui/badge";
import { Sparkles, TrendingUp, TrendingDown, RefreshCw, AlertCircle } from "lucide-react";

export default function AITradingInsights({ cryptocurrencies, holdings, transactions }) {
  const [insights, setInsights] = useState([]);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    generateInsights();
  }, [cryptocurrencies]);

  const generateInsights = async () => {
    setLoading(true);
    try {
      const topCryptos = cryptocurrencies.slice(0, 8);
      const holdingSymbols = holdings.map(h => h.crypto_symbol);
      
      const response = await base44.integrations.Core.InvokeLLM({
        prompt: `As an AI financial analyst, analyze these cryptocurrency market trends and provide trading insights:

Current Market Data:
${topCryptos.map(c => `- ${c.name} (${c.symbol}): $${c.current_price} (${c.price_change_24h > 0 ? '+' : ''}${c.price_change_24h}%)`).join('\n')}

User Holdings: ${holdingSymbols.length > 0 ? holdingSymbols.join(', ') : 'None'}
Recent Activity: ${transactions.length} transactions

Provide 5 actionable insights in this format:
{
  "insights": [
    {
      "title": "Short insight title",
      "description": "Detailed analysis (2-3 sentences)",
      "symbol": "BTC",
      "action": "buy|sell|hold",
      "confidence": "high|medium|low",
      "reasoning": "Brief reason for recommendation"
    }
  ]
}

Focus on:
1. Market momentum and trends
2. Potential buying opportunities
3. Risk warnings
4. Portfolio diversification
5. Technical indicators`,
        add_context_from_internet: true,
        response_json_schema: {
          type: "object",
          properties: {
            insights: {
              type: "array",
              items: {
                type: "object",
                properties: {
                  title: { type: "string" },
                  description: { type: "string" },
                  symbol: { type: "string" },
                  action: { type: "string" },
                  confidence: { type: "string" },
                  reasoning: { type: "string" }
                }
              }
            }
          }
        }
      });

      setInsights(response.insights || []);
    } catch (error) {
      console.error("Error generating insights:", error);
      setInsights([]);
    } finally {
      setLoading(false);
    }
  };

  const getActionColor = (action) => {
    switch (action?.toLowerCase()) {
      case 'buy': return 'bg-green-100 text-green-800';
      case 'sell': return 'bg-red-100 text-red-800';
      case 'hold': return 'bg-yellow-100 text-yellow-800';
      default: return 'bg-gray-100 text-gray-800';
    }
  };

  const getConfidenceBadge = (confidence) => {
    switch (confidence?.toLowerCase()) {
      case 'high': return 'bg-blue-600 text-white';
      case 'medium': return 'bg-blue-400 text-white';
      case 'low': return 'bg-blue-200 text-blue-800';
      default: return 'bg-gray-400 text-white';
    }
  };

  return (
    <Card className="shadow-lg">
      <CardHeader>
        <div className="flex items-center justify-between">
          <CardTitle className="flex items-center gap-2">
            <Sparkles className="w-5 h-5 text-purple-600" />
            AI Trading Insights
          </CardTitle>
          <Button 
            size="sm" 
            variant="outline" 
            onClick={generateInsights}
            disabled={loading}
          >
            <RefreshCw className={`w-4 h-4 ${loading ? 'animate-spin' : ''}`} />
          </Button>
        </div>
      </CardHeader>
      <CardContent>
        {loading ? (
          <div className="flex items-center justify-center py-12">
            <div className="text-center">
              <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-purple-600 mx-auto mb-3"></div>
              <p className="text-sm text-gray-600">Analyzing market trends...</p>
            </div>
          </div>
        ) : insights.length === 0 ? (
          <div className="text-center py-8">
            <AlertCircle className="w-12 h-12 text-gray-400 mx-auto mb-3" />
            <p className="text-gray-600">No insights available</p>
            <Button onClick={generateInsights} className="mt-4" size="sm">
              Generate Insights
            </Button>
          </div>
        ) : (
          <div className="space-y-4">
            {insights.map((insight, index) => (
              <div key={index} className="p-4 rounded-lg border bg-gradient-to-r from-white to-purple-50">
                <div className="flex items-start justify-between mb-2">
                  <div className="flex items-center gap-2">
                    <h4 className="font-semibold text-gray-900">{insight.title}</h4>
                    <Badge className={getActionColor(insight.action)}>
                      {insight.action?.toUpperCase()}
                    </Badge>
                  </div>
                  <Badge className={getConfidenceBadge(insight.confidence)}>
                    {insight.confidence}
                  </Badge>
                </div>
                
                <p className="text-sm text-gray-700 mb-2">{insight.description}</p>
                
                <div className="flex items-center gap-2 text-sm">
                  <span className="font-medium text-purple-600">{insight.symbol}</span>
                  <span className="text-gray-500">•</span>
                  <span className="text-gray-600">{insight.reasoning}</span>
                </div>
              </div>
            ))}
          </div>
        )}
        
        <div className="mt-4 p-3 bg-blue-50 rounded-lg border border-blue-200">
          <div className="flex items-start gap-2">
            <AlertCircle className="w-4 h-4 text-blue-600 mt-0.5 flex-shrink-0" />
            <p className="text-xs text-blue-800">
              AI insights are generated based on current market data and trends. 
              This is not financial advice. Always conduct your own research before trading.
            </p>
          </div>
        </div>
      </CardContent>
    </Card>
  );
}

import React, { useState } from "react";
import { base44 } from "@/api/base44Client";
import { Dialog, DialogContent, DialogHeader, DialogTitle } from "@/components/ui/dialog";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";

import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";
import { toast } from "sonner";

export default function CreateStrategyModal({ onClose, onSuccess }) {
  const [formData, setFormData] = useState({
    strategy_name: "",
    crypto_symbol: "BTC",
    strategy_type: "price_drop",
    action: "buy",
    use_ai_predictions: false,
    max_trades_per_day: 5
  });
  const [buyConditions, setBuyConditions] = useState({
    price_threshold: "",
    amount_usd: ""
  });
  const [sellConditions, setSellConditions] = useState({
    price_threshold: "",
    take_profit_percentage: "",
    stop_loss_percentage: "",
    quantity: ""
  });
  const [loading, setLoading] = useState(false);

  const handleSubmit = async (e) => {
    e.preventDefault();
    
    if (!formData.strategy_name || !formData.crypto_symbol) {
      toast.error("Please fill in all required fields");
      return;
    }

    setLoading(true);
    try {
      const strategyData = {
        ...formData,
        buy_conditions: formData.action === "buy" || formData.action === "both" ? {
          price_threshold: buyConditions.price_threshold ? parseFloat(buyConditions.price_threshold) : undefined,
          amount_usd: buyConditions.amount_usd ? parseFloat(buyConditions.amount_usd) : undefined
        } : undefined,
        sell_conditions: formData.action === "sell" || formData.action === "both" ? {
          price_threshold: sellConditions.price_threshold ? parseFloat(sellConditions.price_threshold) : undefined,
          take_profit_percentage: sellConditions.take_profit_percentage ? parseFloat(sellConditions.take_profit_percentage) : undefined,
          stop_loss_percentage: sellConditions.stop_loss_percentage ? parseFloat(sellConditions.stop_loss_percentage) : undefined,
          quantity: sellConditions.quantity ? parseFloat(sellConditions.quantity) : undefined
        } : undefined,
        is_active: true,
        trades_today: 0,
        total_executions: 0,
        total_profit_loss: 0,
        status: "active"
      };

      await base44.entities.TradingStrategy.create(strategyData);
      toast.success("Trading strategy created!");
      onSuccess();
    } catch (error) {
      console.error("Error creating strategy:", error);
      toast.error("Failed to create strategy");
    } finally {
      setLoading(false);
    }
  };

  return (
    <Dialog open onOpenChange={onClose}>
      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
        <DialogHeader>
          <DialogTitle>Create Trading Strategy</DialogTitle>
        </DialogHeader>
        
        <form onSubmit={handleSubmit} className="space-y-4 mt-4">
          <div>
            <Label>Strategy Name *</Label>
            <Input
              placeholder="e.g., BTC Dip Buyer"
              value={formData.strategy_name}
              onChange={(e) => setFormData({ ...formData, strategy_name: e.target.value })}
              required
            />
          </div>

          <div className="grid grid-cols-2 gap-4">
            <div>
              <Label>Cryptocurrency *</Label>
              <Select value={formData.crypto_symbol} onValueChange={(value) => setFormData({ ...formData, crypto_symbol: value })}>
                <SelectTrigger>
                  <SelectValue />
                </SelectTrigger>
                <SelectContent>
                  <SelectItem value="BTC">Bitcoin (BTC)</SelectItem>
                  <SelectItem value="ETH">Ethereum (ETH)</SelectItem>
                  <SelectItem value="SOL">Solana (SOL)</SelectItem>
                  <SelectItem value="ADA">Cardano (ADA)</SelectItem>
                  <SelectItem value="DOT">Polkadot (DOT)</SelectItem>
                  <SelectItem value="MATIC">Polygon (MATIC)</SelectItem>
                </SelectContent>
              </Select>
            </div>

            <div>
              <Label>Strategy Type</Label>
              <Select value={formData.strategy_type} onValueChange={(value) => setFormData({ ...formData, strategy_type: value })}>
                <SelectTrigger>
                  <SelectValue />
                </SelectTrigger>
                <SelectContent>
                  <SelectItem value="price_drop">Price Drop</SelectItem>
                  <SelectItem value="price_rise">Price Rise</SelectItem>
                  <SelectItem value="dca">Dollar Cost Average</SelectItem>
                  <SelectItem value="ai_driven">AI Driven</SelectItem>
                  <SelectItem value="rsi">RSI Based</SelectItem>
                  <SelectItem value="moving_average">Moving Average</SelectItem>
                </SelectContent>
              </Select>
            </div>
          </div>

          <div className="grid grid-cols-2 gap-4">
            <div>
              <Label>Action *</Label>
              <Select value={formData.action} onValueChange={(value) => setFormData({ ...formData, action: value })}>
                <SelectTrigger>
                  <SelectValue />
                </SelectTrigger>
                <SelectContent>
                  <SelectItem value="buy">Buy Only</SelectItem>
                  <SelectItem value="sell">Sell Only</SelectItem>
                  <SelectItem value="both">Buy & Sell</SelectItem>
                </SelectContent>
              </Select>
            </div>

            <div>

            </div>

          <div className="flex items-center space-x-2">

import React, { useState, useEffect } from "react";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { LineChart, Line, AreaChart, Area, BarChart, Bar, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer, PieChart, Pie, Cell } from "recharts";
import { TrendingUp, TrendingDown, Activity, Target, Award, Clock } from "lucide-react";
import { Badge } from "@/components/ui/badge";

export default function CryptoAnalytics({ holdings, transactions }) {
  const [portfolioHistory, setPortfolioHistory] = useState([]);
  const [metrics, setMetrics] = useState({
    totalPL: 0,
    totalPLPercent: 0,
    winRate: 0,
    avgTradeDuration: 0,
    bestPerformer: null,
    worstPerformer: null
  });

  useEffect(() => {
    calculateMetrics();
    generatePortfolioHistory();
  }, [holdings, transactions]);

  const calculateMetrics = () => {
    if (!holdings?.length || !transactions?.length) return;

    // Calculate total P/L
    let totalPL = 0;
    let totalInvested = 0;
    let bestPerformer = { symbol: '', pl: -Infinity };
    let worstPerformer = { symbol: '', pl: Infinity };

    holdings.forEach(holding => {
      const invested = holding.quantity * (holding.average_buy_price || 0);
      const current = holding.current_value || 0;
      const pl = current - invested;
      const plPercent = invested > 0 ? ((pl / invested) * 100) : 0;

      totalPL += pl;
      totalInvested += invested;

      if (plPercent > bestPerformer.pl) {
        bestPerformer = { symbol: holding.crypto_symbol, pl: plPercent };
      }
      if (plPercent < worstPerformer.pl) {
        worstPerformer = { symbol: holding.crypto_symbol, pl: plPercent };
      }
    });

    const totalPLPercent = totalInvested > 0 ? ((totalPL / totalInvested) * 100) : 0;

    // Calculate win rate
    const buyTxs = transactions.filter(t => t.type === 'crypto_buy');
    const sellTxs = transactions.filter(t => t.type === 'crypto_sell');
    const completedTrades = Math.min(buyTxs.length, sellTxs.length);
    
    let wins = 0;
    sellTxs.forEach(sell => {
      const correspondingBuy = buyTxs.find(buy => 
        sell.description?.includes(buy.description?.split(' ')[1])
      );
      if (correspondingBuy && sell.amount > correspondingBuy.amount) {
        wins++;
      }
    });

    const winRate = completedTrades > 0 ? (wins / completedTrades) * 100 : 0;

    // Calculate average trade duration
    const tradeDurations = sellTxs.map(sell => {
      const correspondingBuy = buyTxs.find(buy => 
        sell.description?.includes(buy.description?.split(' ')[1])
      );
      if (correspondingBuy) {
        const buyDate = new Date(correspondingBuy.created_date);
        const sellDate = new Date(sell.created_date);
        return (sellDate - buyDate) / (1000 * 60 * 60 * 24); // days
      }
      return 0;
    }).filter(d => d > 0);

    const avgTradeDuration = tradeDurations.length > 0 
      ? tradeDurations.reduce((a, b) => a + b, 0) / tradeDurations.length 
      : 0;

    setMetrics({
      totalPL,
      totalPLPercent,
      winRate,
      avgTradeDuration,
      bestPerformer: bestPerformer.pl > -Infinity ? bestPerformer : null,
      worstPerformer: worstPerformer.pl < Infinity ? worstPerformer : null
    });
  };

  const generatePortfolioHistory = () => {
    if (!transactions?.length) return;

    // Generate portfolio value over time
    const sortedTxs = [...transactions]
      .filter(t => t.type === 'crypto_buy' || t.type === 'crypto_sell')
      .sort((a, b) => new Date(a.created_date) - new Date(b.created_date));

    let runningValue = 0;
    const history = sortedTxs.map((tx, index) => {
      if (tx.type === 'crypto_buy') {
        runningValue += tx.amount;
      } else if (tx.type === 'crypto_sell') {
        runningValue += tx.amount;
      }

      return {
        date: new Date(tx.created_date).toLocaleDateString('en-US', { month: 'short', day: 'numeric' }),
        value: runningValue,
        type: tx.type
      };
    });

    // Add current portfolio value
    const currentValue = holdings?.reduce((sum, h) => sum + (h.current_value || 0), 0) || 0;
    history.push({
      date: 'Now',
      value: currentValue,
      type: 'current'
    });

    setPortfolioHistory(history);
  };

  const getAssetDistribution = () => {
    if (!holdings?.length) return [];
    
    return holdings.map(h => ({
      name: h.crypto_symbol,
      value: h.current_value || 0
    }));
  };

  const COLORS = ['#3b82f6', '#8b5cf6', '#ec4899', '#f59e0b', '#10b981', '#6366f1'];

  return (
    <div className="space-y-6">
      {/* Performance Metrics */}
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
        <Card className="bg-white/80 backdrop-blur-sm">
          <CardContent className="p-6">
            <div className="flex items-center justify-between">
              <div>
                <p className="text-sm text-gray-600 mb-1">Total P/L</p>
                <p className={`text-2xl font-bold ${metrics.totalPL >= 0 ? 'text-green-600' : 'text-red-600'}`}>
                  ${Math.abs(metrics.totalPL).toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 })}
                </p>
                <p className={`text-sm ${metrics.totalPLPercent >= 0 ? 'text-green-600' : 'text-red-600'}`}>
                  {metrics.totalPLPercent >= 0 ? '+' : ''}{metrics.totalPLPercent.toFixed(2)}%
                </p>
              </div>
              {metrics.totalPL >= 0 ? (
                <TrendingUp className="w-8 h-8 text-green-600" />
              ) : (
                <TrendingDown className="w-8 h-8 text-red-600" />
              )}
            </div>
          </CardContent>
        </Card>

        <Card className="bg-white/80 backdrop-blur-sm">
          <CardContent className="p-6">
            <div className="flex items-center justify-between">
              <div>
                <p className="text-sm text-gray-600 mb-1">Win Rate</p>
                <p className="text-2xl font-bold text-blue-600">{metrics.winRate.toFixed(1)}%</p>
                <p className="text-sm text-gray-500">Success rate</p>
              </div>
              <Target className="w-8 h-8 text-blue-600" />
            </div>
          </CardContent>
        </Card>

        <Card className="bg-white/80 backdrop-blur-sm">
          <CardContent className="p-6">
            <div className="flex items-center justify-between">
              <div>
                <p className="text-sm text-gray-600 mb-1">Avg Hold Time</p>
                <p className="text-2xl font-bold text-purple-600">{metrics.avgTradeDuration.toFixed(1)}</p>
                <p className="text-sm text-gray-500">days</p>
              </div>
              <Clock className="w-8 h-8 text-purple-600" />
            </div>
          </CardContent>
        </Card>

        <Card className="bg-white/80 backdrop-blur-sm">
          <CardContent className="p-6">
            <div className="flex items-center justify-between">
              <div>
                <p className="text-sm text-gray-600 mb-1">Top Performer</p>
                {metrics.bestPerformer ? (
                  <>
                    <p className="text-2xl font-bold text-green-600">{metrics.bestPerformer.symbol}</p>
                    <p className="text-sm text-green-600">+{metrics.bestPerformer.pl.toFixed(2)}%</p>
                  </>
                ) : (
                  <p className="text-sm text-gray-400">No data</p>
                )}
              </div>
              <Award className="w-8 h-8 text-green-600" />
            </div>
          </CardContent>
        </Card>
      </div>

      {/* Portfolio Value Over Time */}
      <Card className="bg-white/80 backdrop-blur-sm">
        <CardHeader>
          <CardTitle className="flex items-center gap-2">
            <Activity className="w-5 h-5 text-blue-600" />
            Portfolio Value Over Time
          </CardTitle>
        </CardHeader>
        <CardContent>
          {portfolioHistory.length > 0 ? (
            <ResponsiveContainer width="100%" height={300}>
              <AreaChart data={portfolioHistory}>
                <defs>
                  <linearGradient id="colorValue" x1="0" y1="0" x2="0" y2="1">
                    <stop offset="5%" stopColor="#3b82f6" stopOpacity={0.8}/>
                    <stop offset="95%" stopColor="#3b82f6" stopOpacity={0}/>
                  </linearGradient>
                </defs>
                <CartesianGrid strokeDasharray="3 3" stroke="#e5e7eb" />
                <XAxis dataKey="date" stroke="#6b7280" />
                <YAxis stroke="#6b7280" />
                <Tooltip 
                  contentStyle={{ backgroundColor: 'white', border: '1px solid #e5e7eb', borderRadius: '8px' }}
                  formatter={(value) => `$${value.toLocaleString()}`}
                />
                <Area 
                  type="monotone" 
                  dataKey="value" 
                  stroke="#3b82f6" 
                  strokeWidth={2}
                  fillOpacity={1} 
                  fill="url(#colorValue)" 
                />
              </AreaChart>
            </ResponsiveContainer>
          ) : (
            <div className="text-center py-12 text-gray-500">
              <Activity className="w-12 h-12 mx-auto mb-3 opacity-50" />
              <p>No trading history yet</p>
            </div>
          )}
        </CardContent>
      </Card>

      {/* Asset Distribution */}
      <div className="grid lg:grid-cols-2 gap-6">
        <Card className="bg-white/80 backdrop-blur-sm">
          <CardHeader>
            <CardTitle>Asset Distribution</CardTitle>
          </CardHeader>
          <CardContent>
            {getAssetDistribution().length > 0 ? (
              <ResponsiveContainer width="100%" height={300}>
                <PieChart>
                  <Pie
                    data={getAssetDistribution()}
                    cx="50%"
                    cy="50%"
                    labelLine={false}

export default function CryptoPriceChart({ symbol, name, currentPrice }) {
  const [historicalData, setHistoricalData] = useState([]);
  const [timeframe, setTimeframe] = useState("7d");
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    loadHistoricalData();
  }, [symbol, timeframe]);

  const loadHistoricalData = async () => {
    setLoading(true);
    try {
      const response = await base44.integrations.Core.InvokeLLM({
        prompt: `Generate realistic historical price data for ${name} (${symbol}) cryptocurrency for the past ${timeframe === '24h' ? '24 hours' : timeframe === '7d' ? '7 days' : timeframe === '30d' ? '30 days' : '90 days'}.

Return ${timeframe === '24h' ? 24 : timeframe === '7d' ? 7 : timeframe === '30d' ? 30 : 90} data points showing realistic price fluctuations.
Current price is $${currentPrice}. Generate realistic historical data leading up to this price.

Return format:
{
  "data": [
    { "time": "2024-01-01", "price": 42000.50 },
    ...
  ]
}`,
        add_context_from_internet: true,
        response_json_schema: {
          type: "object",
          properties: {
            data: {
              type: "array",
              items: {
                type: "object",
                properties: {
                  time: { type: "string" },
                  price: { type: "number" }
                }
              }
            }
          }
        }
      });

      setHistoricalData(response.data || []);
    } catch (error) {
      console.error("Error loading historical data:", error);
      // Generate fallback data
      generateFallbackData();
    } finally {
      setLoading(false);
    }
  };

  const generateFallbackData = () => {
    const points = timeframe === '24h' ? 24 : timeframe === '7d' ? 7 : timeframe === '30d' ? 30 : 90;
    const data = [];
    let price = currentPrice * 0.95;
    
    for (let i = 0; i < points; i++) {
      const change = (Math.random() - 0.48) * (currentPrice * 0.03);
      price = Math.max(price + change, currentPrice * 0.8);
      
      const date = new Date();
      date.setDate(date.getDate() - (points - i - 1));
      
      data.push({
        time: date.toLocaleDateString(),
        price: parseFloat(price.toFixed(2))
      });
    }
    
    data[data.length - 1].price = currentPrice;
    setHistoricalData(data);
  };

  const priceChange = historicalData.length > 1 
    ? ((historicalData[historicalData.length - 1]?.price - historicalData[0]?.price) / historicalData[0]?.price) * 100 
    : 0;

  const isPositive = priceChange >= 0;

  return (
    <Card className="shadow-lg">
      <CardHeader>
        <div className="flex items-center justify-between">
          <CardTitle className="flex items-center gap-2">
            {symbol} Price Chart
            {isPositive ? (
              <TrendingUp className="w-5 h-5 text-green-600" />
            ) : (
              <TrendingDown className="w-5 h-5 text-red-600" />
            )}
          </CardTitle>
          <Button 
            size="sm" 
            variant="outline" 
            onClick={loadHistoricalData}

      amount,
      total: calculateTotal(numAmount)

export default function PriceAlerts({ cryptocurrencies }) {
  const [alerts, setAlerts] = useState([]);
  const [showDialog, setShowDialog] = useState(false);
  const [newAlert, setNewAlert] = useState({
    symbol: '',
    target_price: '',
    condition: 'above',
    active: true
  });

  useEffect(() => {
    loadAlerts();
  }, []);

  const loadAlerts = async () => {
    try {
      const user = await base44.auth.me();
      const savedAlerts = user.price_alerts || [];
      setAlerts(savedAlerts);
    } catch (error) {
      console.error("Error loading alerts:", error);
    }
  };

  const createAlert = async () => {
    if (!newAlert.symbol || !newAlert.target_price) {
      alert("Please fill in all fields");
      return;
    }

    try {
      const user = await base44.auth.me();
      const crypto = cryptocurrencies.find(c => c.symbol === newAlert.symbol);
      
      const alertData = {
        ...newAlert,
        target_price: parseFloat(newAlert.target_price),
        crypto_name: crypto?.name || newAlert.symbol,
        current_price: crypto?.current_price || 0,
        created_at: new Date().toISOString()
      };

      const updatedAlerts = [...alerts, alertData];
      await base44.auth.updateMe({ price_alerts: updatedAlerts });
      
      setAlerts(updatedAlerts);
      setShowDialog(false);
      setNewAlert({ symbol: '', target_price: '', condition: 'above', active: true });
    } catch (error) {
      console.error("Error creating alert:", error);
      alert("Failed to create alert");
    }
  };

  const deleteAlert = async (index) => {
    try {
      const updatedAlerts = alerts.filter((_, i) => i !== index);
      await base44.auth.updateMe({ price_alerts: updatedAlerts });
      setAlerts(updatedAlerts);
    } catch (error) {
      console.error("Error deleting alert:", error);
    }
  };

  const checkAlerts = async () => {
    const user = await base44.auth.me();
    const triggeredAlerts = [];

    for (const alert of alerts) {
      if (!alert.active) continue;

      const crypto = cryptocurrencies.find(c => c.symbol === alert.symbol);
      if (!crypto) continue;

      const currentPrice = crypto.current_price;
      const triggered = 
        (alert.condition === 'above' && currentPrice >= alert.target_price) ||
        (alert.condition === 'below' && currentPrice <= alert.target_price);

      if (triggered) {
        triggeredAlerts.push(alert);
        
        // Send email notification
        await base44.integrations.Core.SendEmail({
          to: user.email,
          subject: `🔔 Price Alert: ${alert.symbol} ${alert.condition} $${alert.target_price}`,
          body: `Your price alert has been triggered!\n\n${alert.crypto_name} (${alert.symbol})\nCurrent Price: $${currentPrice}\nTarget Price: $${alert.target_price}\nCondition: ${alert.condition}\n\nView your portfolio: ${window.location.origin}`
        });
      }
    }

    if (triggeredAlerts.length > 0) {
      // Deactivate triggered alerts
      const updatedAlerts = alerts.map(alert => {
        const wasTriggered = triggeredAlerts.some(t => 
          t.symbol === alert.symbol && t.target_price === alert.target_price
        );
        return wasTriggered ? { ...alert, active: false } : alert;
      });
      
      await base44.auth.updateMe({ price_alerts: updatedAlerts });
      setAlerts(updatedAlerts);
    }
  };

  useEffect(() => {
    const interval = setInterval(() => {
      if (alerts.length > 0 && cryptocurrencies.length > 0) {
        checkAlerts();
      }
    }, 60000); // Check every minute

    return () => clearInterval(interval);
  }, [alerts, cryptocurrencies]);

  return (
    <Card className="shadow-lg">
      <CardHeader>
        <div className="flex items-center justify-between">
          <CardTitle className="flex items-center gap-2">
            <Bell className="w-5 h-5" />
            Price Alerts
          </CardTitle>
          <Dialog open={showDialog} onOpenChange={setShowDialog}>
            <DialogTrigger asChild>
              <Button size="sm">
                <Plus className="w-4 h-4 mr-2" />
                New Alert
              </Button>
            </DialogTrigger>
            <DialogContent>
              <DialogHeader>
                <DialogTitle>Create Price Alert</DialogTitle>
              </DialogHeader>
              <div className="space-y-4 mt-4">
                <div>
                  <Label>Cryptocurrency</Label>
                  <Select value={newAlert.symbol} onValueChange={(value) => setNewAlert({ ...newAlert, symbol: value })}>
                    <SelectTrigger>
                      <SelectValue placeholder="Select crypto" />
                    </SelectTrigger>
                    <SelectContent>
                      {cryptocurrencies.map(crypto => (
                        <SelectItem key={crypto.symbol} value={crypto.symbol}>
                          {crypto.name} ({crypto.symbol})
                        </SelectItem>
                      ))}
                    </SelectContent>
                  </Select>
                </div>

                <div>
                  <Label>Condition</Label>
                  <Select value={newAlert.condition} onValueChange={(value) => setNewAlert({ ...newAlert, condition: value })}>
                    <SelectTrigger>
                      <SelectValue />
                    </SelectTrigger>
                    <SelectContent>
                      <SelectItem value="above">Price goes above</SelectItem>
                      <SelectItem value="below">Price goes below</SelectItem>
                    </SelectContent>
                  </Select>
                </div>

                <div>
                  <Label>Target Price</Label>
                  <Input
                    type="number"
                    placeholder="0.00"
                    value={newAlert.target_price}
                    onChange={(e) => setNewAlert({ ...newAlert, target_price: e.target.value })}
                  />
                </div>

                <Button onClick={createAlert} className="w-full">
                  Create Alert
                </Button>
              </div>
            </DialogContent>
          </Dialog>
        </div>
      </CardHeader>
      <CardContent>
        {alerts.length === 0 ? (
          <p className="text-gray-500 text-center py-8">No price alerts set</p>
        ) : (
          <div className="space-y-3">
            {alerts.map((alert, index) => (
              <div key={index} className={`p-4 rounded-lg border ${alert.active ? 'bg-white' : 'bg-gray-50 opacity-60'}`}>
                <div className="flex items-center justify-between">
                  <div className="flex items-center gap-3">
                    {alert.condition === 'above' ? (
                      <TrendingUp className="w-5 h-5 text-green-600" />
                    ) : (
                      <TrendingDown className="w-5 h-5 text-red-600" />
                    )}
                    <div>
                      <p className="font-semibold">{alert.crypto_name} ({alert.symbol})</p>
                      <p className="text-sm text-gray-600">
                        Alert when {alert.condition} ${alert.target_price.toLocaleString()}
                      </p>
                      {!alert.active && (
                        <span className="text-xs text-yellow-600 font-medium">Triggered</span>
                      )}
                    </div>
                  </div>
                  <Button
                    size="sm"
                    variant="ghost"
                    onClick={() => deleteAlert(index)}
                  >
                    <Trash2 className="w-4 h-4 text-red-600" />

  const toggleStrategy = async (strategy) => {
    try {
      await base44.entities.TradingStrategy.update(strategy.id, {
        is_active: !strategy.is_active
      });
      
      setStrategies(strategies.map(s => 
        s.id === strategy.id ? { ...s, is_active: !s.is_active } : s
      ));
      
      toast.success(`Strategy ${!strategy.is_active ? 'activated' : 'paused'}`);
    } catch (error) {
      console.error("Error toggling strategy:", error);
      toast.error("Failed to update strategy");
    }
  };

  const deleteStrategy = async (strategyId) => {
    if (!confirm("Are you sure you want to delete this strategy?")) return;
    
    try {
      await base44.entities.TradingStrategy.delete(strategyId);
      setStrategies(strategies.filter(s => s.id !== strategyId));
      toast.success("Strategy deleted");
    } catch (error) {
      console.error("Error deleting strategy:", error);
      toast.error("Failed to delete strategy");
    }
  };

  const getActionIcon = (action) => {
    switch (action) {
      case "buy": return <TrendingUp className="w-4 h-4 text-green-600" />;
      case "sell": return <TrendingDown className="w-4 h-4 text-red-600" />;
      default: return <Zap className="w-4 h-4 text-blue-600" />;
    }
  };

  return (
    <>
      <Card className="shadow-lg">
        <CardHeader>
          <div className="flex items-center justify-between">
            <CardTitle className="flex items-center gap-2">
              <Zap className="w-6 h-6 text-blue-600" />
              Automated Trading Strategies
            </CardTitle>
            <Button onClick={() => setShowCreateModal(true)} size="sm">
              <Plus className="w-4 h-4 mr-2" />
              New Strategy
            </Button>
          </div>
        </CardHeader>
        <CardContent>
          {loading ? (
            <div className="flex items-center justify-center py-8">
              <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-blue-600"></div>
            </div>
          ) : strategies.length === 0 ? (
            <div className="text-center py-8">
              <Zap className="w-12 h-12 text-gray-400 mx-auto mb-3" />
              <p className="text-gray-600 mb-4">No trading strategies yet</p>
              <Button onClick={() => setShowCreateModal(true)} size="sm">
                Create Your First Strategy
              </Button>
            </div>
          ) : (
            <div className="space-y-4">
              {strategies.map((strategy) => (
                <div
                  key={strategy.id}
                  className={`p-4 rounded-lg border ${
                    strategy.is_active
                      ? "bg-gradient-to-r from-white to-green-50 border-green-200"
                      : "bg-gray-50 border-gray-200"
                  }`}
                >
                  <div className="flex items-start justify-between mb-3">
                    <div className="flex-1">
                      <div className="flex items-center gap-2 mb-1">
                        {getActionIcon(strategy.action)}
                        <h4 className="font-semibold text-gray-900">{strategy.strategy_name}</h4>
                        <Badge variant={strategy.is_active ? "default" : "outline"}>
                          {strategy.is_active ? "Active" : "Paused"}
                        </Badge>
                        {strategy.use_ai_predictions && (
                          <Badge className="bg-purple-600">AI-Powered</Badge>
                        )}
                      </div>
                      <div className="flex items-center gap-4 text-sm text-gray-600">
                        <span className="font-semibold">{strategy.crypto_symbol}</span>
                        <span>•</span>
                        <span className="capitalize">{strategy.action}</span>
                        <span>•</span>
                        <span className="capitalize">{strategy.strategy_type.replace('_', ' ')}</span>
                      </div>
                    </div>
                    <div className="flex items-center gap-2">
                      <Button
                        size="sm"
                        variant={strategy.is_active ? "outline" : "default"}
                        onClick={() => toggleStrategy(strategy)}
                      >
                        {strategy.is_active ? "Pause" : "Activate"}
                      </Button>
                      <Button
                        size="sm"
                        variant="ghost"
                        onClick={() => deleteStrategy(strategy.id)}
                        className="text-red-600"
                      >
                        <Trash2 className="w-4 h-4" />
                      </Button>
                    </div>
                  </div>

                  <div className="grid grid-cols-2 md:grid-cols-4 gap-3 text-sm">
                    <div className="p-2 bg-white rounded">
                      <p className="text-xs text-gray-500">Executions</p>
                      <p className="font-semibold">{strategy.total_executions || 0}</p>
                    </div>
                    <div className="p-2 bg-white rounded">
                      <p className="text-xs text-gray-500">Today</p>
                      <p className="font-semibold">
                        {strategy.trades_today || 0}/{strategy.max_trades_per_day}
                      </p>
                    </div>
                    <div className="p-2 bg-white rounded">
                      <p className="text-xs text-gray-500">P/L</p>
                      <p className={`font-semibold ${
                        (strategy.total_profit_loss || 0) >= 0 ? "text-green-600" : "text-red-600"
                      }`}>
                        ${(strategy.total_profit_loss || 0).toFixed(2)}
                      </p>
                    </div>
                    <div className="p-2 bg-white rounded">
                      <p className="text-xs text-gray-500">Last Run</p>
                      <p className="font-semibold text-xs">
                        {strategy.last_executed
                          ? new Date(strategy.last_executed).toLocaleDateString()
                          : "Never"}
                      </p>
                    </div>
                  </div>

                  {/* Conditions */}
                  <div className="mt-3 p-3 bg-blue-50 rounded-lg">
                    <p className="text-xs font-semibold text-blue-900 mb-2">Conditions:</p>
                    <div className="space-y-1 text-xs text-blue-800">
                      {strategy.buy_conditions && (
                        <div>
                          <span className="font-semibold">Buy:</span>{" "}
                          {strategy.buy_conditions.price_threshold && 
                            `Price ≤ $${strategy.buy_conditions.price_threshold}`}
                          {strategy.buy_conditions.amount_usd && 
                            ` • Amount: $${strategy.buy_conditions.amount_usd}`}
                        </div>
                      )}
                      {strategy.sell_conditions && (
                        <div>
                          <span className="font-semibold">Sell:</span>{" "}
                          {strategy.sell_conditions.price_threshold && 
                            `Price ≥ $${strategy.sell_conditions.price_threshold}`}
                          {strategy.sell_conditions.take_profit_percentage && 
                            ` • Take Profit: ${strategy.sell_conditions.take_profit_percentage}%`}
                          {strategy.sell_conditions.stop_loss_percentage && 
                            ` • Stop Loss: ${strategy.sell_conditions.stop_loss_percentage}%`}
                        </div>
                      )}
                    </div>
                  </div>
                </div>
              ))}
            </div>
          )}

          <div className="mt-4 p-3 bg-yellow-50 rounded-lg border border-yellow-200">
            <p className="text-xs text-yellow-800">
              <Zap className="w-4 h-4 inline mr-1" />
              Automated strategies run every 5 minutes. Ensure you have sufficient balance and holdings for execution.
            </p>
          </div>
        </CardContent>
      </Card>

      {showCreateModal && (
        <CreateStrategyModal
          onClose={() => setShowCreateModal(false)}
          onSuccess={() => {
            loadStrategies();
            setShowCreateModal(false);
          }}
        />
      )}
    </>
  );
}
