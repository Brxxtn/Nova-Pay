
import React, { useState } from "react";
import { Dialog, DialogContent, DialogHeader, DialogTitle } from "@/components/ui/dialog";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import { Textarea } from "@/components/ui/textarea";
import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";
import { AlertCircle } from "lucide-react";

const MAX_GIFT_CARD_BALANCE = 9998.99;

export default function AddCardModal({ open, onOpenChange, onAddCard }) {
  const [cardType, setCardType] = useState("payment");
  const [validationErrors, setValidationErrors] = useState({}); // State for all validation errors

  const [formData, setFormData] = useState({
    card_type: "payment",
    card_name: "",
    card_number: "",
    cvv: "", // New field
    zip_code: "", // New field
    payment_network: "visa",
    card_variant: "physical",
    status: "active",
    expiry_date: "",
    balance: 0,
    issuer: "",
    barcode_data: "",
    barcode_type: "none",
    member_id: "",
    points_balance: 0,
    id_number: "",
    issue_date: "",
    country: "",
    color_scheme: "blue",
    notes: ""
  });

  const validateCVV = (cvv) => {
    const cvvRegex = /^\d{3,4}$/;
    return cvvRegex.test(cvv);
  };

  const validateExpiryDate = (expiry) => {
    // Format: MM/YY
    const expiryRegex = /^(0[1-9]|1[0-2])\/\d{2}$/;
    if (!expiryRegex.test(expiry)) return false;
    
    const [monthStr, yearStr] = expiry.split('/');
    const month = parseInt(monthStr, 10);
    const year = parseInt(yearStr, 10); // e.g., '25' for 2025
    
    const currentDate = new Date();
    const currentFullYear = currentDate.getFullYear(); // e.g., 2023
    const currentMonth = currentDate.getMonth() + 1; // getMonth() is 0-indexed
    
    // Convert two-digit year to four-digit year for comparison, assuming 20xx
    const fullExpYear = 2000 + year;
    const fullCurrentYear = currentFullYear; // Use full year for comparison

    // Check if card is expired
    if (fullExpYear < fullCurrentYear) return false;
    if (fullExpYear === fullCurrentYear && month < currentMonth) return false;
    
    return true;
  };

  const validateZipCode = (zip) => {
    // US zip code: 5 digits or 5+4 format
    const zipRegex = /^\d{5}(-\d{4})?$/;
    return zipRegex.test(zip);
  };

  const handleBalanceChange = (value) => {
    const numValue = parseFloat(value);
    
    let balanceErrorMsg = "";

    if (isNaN(numValue) || numValue < 0) { // Handle non-numeric or negative input
        balanceErrorMsg = "Balance must be a positive number.";
        setFormData({...formData, balance: 0}); // Set to 0 for display if invalid
    } else if (numValue > MAX_GIFT_CARD_BALANCE) {
        balanceErrorMsg = `Maximum balance is $${MAX_GIFT_CARD_BALANCE.toFixed(2)}`;
        setFormData({...formData, balance: numValue}); // Store the value as typed, even if over max, for handleSubmit to catch
    } else {
        setFormData({...formData, balance: numValue});
    }

    setValidationErrors(prev => {
      const newErrors = {...prev};
      if (balanceErrorMsg) {
        newErrors.balance = balanceErrorMsg;
      } else {
        delete newErrors.balance;
      }
      return newErrors;
    });
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    let errors = {}; // Use `let` so we can modify it

    // Payment card validations
    if (cardType === "payment") {
      if (formData.cvv && !validateCVV(formData.cvv)) {
        errors.cvv = "CVV must be 3 or 4 digits";
      }
      
      if (formData.expiry_date && !validateExpiryDate(formData.expiry_date)) {
        errors.expiry = "Invalid or expired date (MM/YY format)";
      }
      
      if (formData.zip_code && !validateZipCode(formData.zip_code)) {
        errors.zip = "Invalid ZIP code (5 digits or 5+4 format)";
      }
    }
    
    // Gift card balance validation
    if (cardType === "gift") {
      if (isNaN(formData.balance) || formData.balance < 0) {
        errors.balance = "Balance must be a positive number.";
      } else if (formData.balance > MAX_GIFT_CARD_BALANCE) {
        errors.balance = `Maximum balance is $${MAX_GIFT_CARD_BALANCE.toFixed(2)}`;
      }
    }
    
    setValidationErrors(errors); // Update validationErrors state

    if (Object.keys(errors).length > 0) {
      return; // Prevent form submission if there are errors
    }
    
    // Prepare final form data, including capping balance if it exceeded max
    const finalFormData = { ...formData, card_type: cardType };
    if (cardType === "gift") {
      if (finalFormData.balance > MAX_GIFT_CARD_BALANCE) {
        finalFormData.balance = MAX_GIFT_CARD_BALANCE; // Cap the value for submission if it's over
      }
      if (isNaN(finalFormData.balance) || finalFormData.balance < 0) {
        finalFormData.balance = 0; // Ensure it's 0 if invalid
      }
    }

    onAddCard(finalFormData);
    resetForm();
  };

  const resetForm = () => {
    setFormData({
      card_type: "payment",
      card_name: "",
      card_number: "",
      cvv: "", // Reset new field
      zip_code: "", // Reset new field
      payment_network: "visa",
      card_variant: "physical",
      status: "active",
      expiry_date: "",
      balance: 0,
      issuer: "",
      barcode_data: "",
      barcode_type: "none",
      member_id: "",
      points_balance: 0,
      id_number: "",
      issue_date: "",
      country: "",
      color_scheme: "blue",
      notes: ""
    });
    setValidationErrors({}); // Clear all validation errors on reset
  };

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
        <DialogHeader>
          <DialogTitle>Add New Card</DialogTitle>
        </DialogHeader>
        
        <Tabs value={cardType} onValueChange={setCardType}>
          <TabsList className="grid w-full grid-cols-5">
            <TabsTrigger value="payment">Payment</TabsTrigger>
            <TabsTrigger value="gift">Gift</TabsTrigger>
            <TabsTrigger value="loyalty">Loyalty</TabsTrigger>
            <TabsTrigger value="id">ID</TabsTrigger>
            <TabsTrigger value="passport">Passport</TabsTrigger>
          </TabsList>
          
          <form onSubmit={handleSubmit}>
            <TabsContent value="payment" className="space-y-4">
              <div className="space-y-2">
                <Label htmlFor="card_name">Card Name</Label>
                <Input
                  id="card_name"
                  placeholder="My Visa Card"
                  value={formData.card_name}
                  onChange={(e) => setFormData({...formData, card_name: e.target.value})}
                  required
                />
              </div>
              
              <div className="grid grid-cols-2 gap-4">
                <div className="space-y-2">
                  <Label>Payment Network</Label>
                  <Select
                    value={formData.payment_network}
                    onValueChange={(value) => setFormData({...formData, payment_network: value})}
                  >
                    <SelectTrigger>
                      <SelectValue />
                    </SelectTrigger>
                    <SelectContent>
                      <SelectItem value="visa">Visa</SelectItem>
                      <SelectItem value="mastercard">Mastercard</SelectItem>
                      <SelectItem value="american_express">American Express</SelectItem>
                      <SelectItem value="discover">Discover</SelectItem>
                    </SelectContent>
                  </Select>
                </div>
                
                <div className="space-y-2">
                  <Label>Card Type</Label>
                  <Select
                    value={formData.card_variant}
                    onValueChange={(value) => setFormData({...formData, card_variant: value})}
                  >
                    <SelectTrigger>
                      <SelectValue />
                    </SelectTrigger>
                    <SelectContent>
                      <SelectItem value="physical">Physical</SelectItem>
                      <SelectItem value="virtual">Virtual</SelectItem>
                    </SelectContent>
                  </Select>
                </div>
              </div>
              
              <div className="grid grid-cols-2 gap-4">
                <div className="space-y-2">
                  <Label htmlFor="card_number">Card Number (Last 4 digits)</Label>
                  <Input
                    id="card_number"
                    placeholder="1234"
                    maxLength={4}
                    value={formData.card_number}
                    onChange={(e) => setFormData({...formData, card_number: e.target.value})}
                  />
                </div>
                
                <div className="space-y-2">
                  <Label htmlFor="expiry_date">Expiry Date (MM/YY)</Label>
                  <Input
                    id="expiry_date"
                    placeholder="12/25"
                    maxLength={5}
                    value={formData.expiry_date}
                    onChange={(e) => {
                      setFormData({...formData, expiry_date: e.target.value});
                      setValidationErrors(prev => { // Clear error on change
                        const newErrors = {...prev};
                        delete newErrors.expiry;
                        return newErrors;
                      });
                    }}
                  />
                  {validationErrors.expiry && (
                    <div className="flex items-center gap-2 text-red-600 text-sm mt-1">
                      <AlertCircle className="w-4 h-4" />
                      <span>{validationErrors.expiry}</span>
                    </div>
                  )}
                </div>
              </div>
              
              <div className="grid grid-cols-2 gap-4">
                <div className="space-y-2">
                  <Label htmlFor="cvv">CVV</Label>
                  <Input
                    id="cvv"
                    type="password"
                    placeholder="123"
                    maxLength={4}
                    value={formData.cvv}
                    onChange={(e) => {
                      setFormData({...formData, cvv: e.target.value});
                      setValidationErrors(prev => { // Clear error on change
                        const newErrors = {...prev};
                        delete newErrors.cvv;
                        return newErrors;
                      });
                    }}
                  />
                  {validationErrors.cvv && (
                    <div className="flex items-center gap-2 text-red-600 text-sm mt-1">
                      <AlertCircle className="w-4 h-4" />
                      <span>{validationErrors.cvv}</span>
                    </div>
                  )}
                </div>
                
                <div className="space-y-2">
                  <Label htmlFor="zip_code">ZIP Code</Label>
                  <Input
                    id="zip_code"
                    placeholder="12345"
                    maxLength={10}
                    value={formData.zip_code}
                    onChange={(e) => {
                      setFormData({...formData, zip_code: e.target.value});
                      setValidationErrors(prev => { // Clear error on change
                        const newErrors = {...prev};
                        delete newErrors.zip;
                        return newErrors;
                      });
                    }}
                  />
                  {validationErrors.zip && (
                    <div className="flex items-center gap-2 text-red-600 text-sm mt-1">
                      <AlertCircle className="w-4 h-4" />
                      <span>{validationErrors.zip}</span>
                    </div>
                  )}
                </div>
              </div>
            </TabsContent>
            
            <TabsContent value="gift" className="space-y-4">
              <div className="space-y-2">
                <Label htmlFor="gift_name">Gift Card Name</Label>
                <Input
                  id="gift_name"
                  placeholder="Amazon Gift Card"
                  value={formData.card_name}
                  onChange={(e) => setFormData({...formData, card_name: e.target.value})}
                  required
                />
              </div>
              
              <div className="grid grid-cols-2 gap-4">
                <div className="space-y-2">
                  <Label htmlFor="issuer">Issuer/Brand</Label>
                  <Input
                    id="issuer"
                    placeholder="Amazon"
                    value={formData.issuer}
                    onChange={(e) => setFormData({...formData, issuer: e.target.value})}
                  />
                </div>
                
                <div className="space-y-2">
                  <Label htmlFor="balance">Balance ($)</Label>
                  <Input
                    id="balance"
                    type="number"
                    step="0.01"
                    min="0"
                    max={MAX_GIFT_CARD_BALANCE}
                    placeholder="50.00"
                    value={formData.balance || ""} // Shows empty string if 0 or NaN
                    onChange={(e) => handleBalanceChange(e.target.value)}
                    required
                  />
                  {validationErrors.balance && ( // Use validationErrors.balance
                    <div className="flex items-center gap-2 text-red-600 text-sm mt-1">
                      <AlertCircle className="w-4 h-4" />
                      <span>{validationErrors.balance}</span>
                    </div>
                  )}
                  <p className="text-xs text-gray-500 mt-1">
                    Maximum balance: ${MAX_GIFT_CARD_BALANCE.toFixed(2)}
                  </p>
                </div>
              </div>
              
              <div className="space-y-2">
                <Label htmlFor="gift_number">Card Number</Label>
                <Input
                  id="gift_number"
                  placeholder="1234-5678-9012-3456"
                  value={formData.card_number}
                  onChange={(e) => setFormData({...formData, card_number: e.target.value})}
                />
              </div>
              
              <div className="space-y-2">
                <Label>Color Scheme</Label>
                <Select
                  value={formData.color_scheme}
                  onValueChange={(value) => setFormData({...formData, color_scheme: value})}
                >
                  <SelectTrigger>
                    <SelectValue />
                  </SelectTrigger>
                  <SelectContent>
                    <SelectItem value="blue">Blue</SelectItem>
                    <SelectItem value="purple">Purple</SelectItem>
                    <SelectItem value="green">Green</SelectItem>
                    <SelectItem value="orange">Orange</SelectItem>
                    <SelectItem value="red">Red</SelectItem>
                    <SelectItem value="gold">Gold</SelectItem>
                    <SelectItem value="silver">Silver</SelectItem>
                    <SelectItem value="black">Black</SelectItem>
                  </SelectContent>
                </Select>
              </div>
            </TabsContent>
            
            <TabsContent value="loyalty" className="space-y-4">
              <div className="space-y-2">
                <Label htmlFor="loyalty_name">Program Name</Label>
                <Input
                  id="loyalty_name"
                  placeholder="Starbucks Rewards"
                  value={formData.card_name}
                  onChange={(e) => setFormData({...formData, card_name: e.target.value})}
                  required
                />
              </div>
              
              <div className="grid grid-cols-2 gap-4">
                <div className="space-y-2">
                  <Label htmlFor="loyalty_issuer">Company</Label>
                  <Input
                    id="loyalty_issuer"
                    placeholder="Starbucks"
                    value={formData.issuer}
                    onChange={(e) => setFormData({...formData, issuer: e.target.value})}
                  />
                </div>
                
                <div className="space-y-2">
                  <Label htmlFor="member_id">Member ID</Label>
                  <Input
                    id="member_id"
                    placeholder="123456789"
                    value={formData.member_id}
                    onChange={(e) => setFormData({...formData, member_id: e.target.value})}
                  />
                </div>
              </div>
              
              <div className="space-y-2">
                <Label htmlFor="points">Points Balance</Label>
                <Input
                  id="points"
                  type="number"
                  placeholder="1250"
                  value={formData.points_balance}
                  onChange={(e) => setFormData({...formData, points_balance: parseInt(e.target.value) || 0})}
                />
              </div>
              
              <div className="grid grid-cols-2 gap-4">
                <div className="space-y-2">
                  <Label>Barcode Type</Label>
                  <Select
                    value={formData.barcode_type}
                    onValueChange={(value) => setFormData({...formData, barcode_type: value})}
                  >
                    <SelectTrigger>
                      <SelectValue />
                    </SelectTrigger>
                    <SelectContent>
                      <SelectItem value="none">None</SelectItem>
                      <SelectItem value="qr">QR Code</SelectItem>
                      <SelectItem value="code128">Barcode (Code 128)</SelectItem>
                      <SelectItem value="ean13">Barcode (EAN-13)</SelectItem>
                    </SelectContent>
                  </Select>
                </div>
                
                <div className="space-y-2">
                  <Label>Color Scheme</Label>
                  <Select
                    value={formData.color_scheme}
                    onValueChange={(value) => setFormData({...formData, color_scheme: value})}
                  >
                    <SelectTrigger>
                      <SelectValue />
                    </SelectTrigger>
                    <SelectContent>
                      <SelectItem value="blue">Blue</SelectItem>
                      <SelectItem value="purple">Purple</SelectItem>
                      <SelectItem value="green">Green</SelectItem>
                      <SelectItem value="orange">Orange</SelectItem>
                      <SelectItem value="red">Red</SelectItem>
                    </SelectContent>
                  </Select>
                </div>
              </div>
              
              {formData.barcode_type !== "none" && (
                <div className="space-y-2">
                  <Label htmlFor="barcode_data">Barcode Data</Label>
                  <Input
                    id="barcode_data"
                    placeholder="1234567890123"
                    value={formData.barcode_data}
                    onChange={(e) => setFormData({...formData, barcode_data: e.target.value})}
                  />
                </div>
              )}
            </TabsContent>
            
            <TabsContent value="id" className="space-y-4">
              <div className="space-y-2">
                <Label htmlFor="id_name">ID Type</Label>
                <Input
                  id="id_name"
                  placeholder="Driver's License"
                  value={formData.card_name}
                  onChange={(e) => setFormData({...formData, card_name: e.target.value})}
                  required
                />
              </div>
              
              <div className="grid grid-cols-2 gap-4">
                <div className="space-y-2">
                  <Label htmlFor="id_number">ID Number</Label>
                  <Input
                    id="id_number"
                    placeholder="DL123456"
                    value={formData.id_number}
                    onChange={(e) => setFormData({...formData, id_number: e.target.value})}
                  />
                </div>
                
                <div className="space-y-2">
                  <Label htmlFor="id_country">Country/State</Label>
                  <Input
                    id="id_country"
                    placeholder="California"
                    value={formData.country}
                    onChange={(e) => setFormData({...formData, country: e.target.value})}
                  />
                </div>
              </div>
              
              <div className="grid grid-cols-2 gap-4">
                <div className="space-y-2">
                  <Label htmlFor="id_issue">Issue Date</Label>
                  <Input
                    id="id_issue"
                    type="date"
                    value={formData.issue_date}
                    onChange={(e) => setFormData({...formData, issue_date: e.target.value})}
                  />
                </div>
                
                <div className="space-y-2">
                  <Label htmlFor="id_expiry">Expiry Date</Label>
                  <Input
                    id="id_expiry"
                    placeholder="12/2028"
                    value={formData.expiry_date}
                    onChange={(e) => setFormData({...formData, expiry_date: e.target.value})}
                  />
                </div>
              </div>
            </TabsContent>
            
            <TabsContent value="passport" className="space-y-4">
              <div className="space-y-2">
                <Label htmlFor="passport_name">Passport Name</Label>
                <Input
                  id="passport_name"
                  placeholder="US Passport"
                  value={formData.card_name}
                  onChange={(e) => setFormData({...formData, card_name: e.target.value})}
                  required
                />
              </div>
              
              <div className="grid grid-cols-2 gap-4">
                <div className="space-y-2">
                  <Label htmlFor="passport_number">Passport Number</Label>
                  <Input
                    id="passport_number"
                    placeholder="P12345678"
                    value={formData.id_number}
                    onChange={(e) => setFormData({...formData, id_number: e.target.value})}
                  />
                </div>
                
                <div className="space-y-2">
                  <Label htmlFor="passport_country">Country</Label>
                  <Input
                    id="passport_country"
                    placeholder="United States"
                    value={formData.country}
                    onChange={(e) => setFormData({...formData, country: e.target.value})}
                  />
                </div>
              </div>
              
              <div className="grid grid-cols-2 gap-4">
                <div className="space-y-2">
                  <Label htmlFor="passport_issue">Issue Date</Label>
                  <Input
                    id="passport_issue"
                    type="date"
                    value={formData.issue_date}
                    onChange={(e) => setFormData({...formData, issue_date: e.target.value})}
                  />
                </div>
                
                <div className="space-y-2">
                  <Label htmlFor="passport_expiry">Expiry Date</Label>
                  <Input
                    id="passport_expiry"
                    type="date"
                    value={formData.expiry_date}
                    onChange={(e) => setFormData({...formData, expiry_date: e.target.value})}
                  />
                </div>
              </div>
            </TabsContent>
            
            <div className="space-y-2 mt-4">
              <Label htmlFor="notes">Notes (Optional)</Label>
              <Textarea
                id="notes"
                placeholder="Additional information..."
                value={formData.notes}
                onChange={(e) => setFormData({...formData, notes: e.target.value})}
              />
            </div>
            
            <div className="flex justify-end gap-3 mt-6">
              <Button type="button" variant="outline" onClick={() => {
                onOpenChange(false);
                resetForm();
              }}>
                Cancel
              </Button>
              <Button type="submit" className="bg-gradient-to-r from-blue-600 to-indigo-600">
                Add Card
              </Button>
            </div>
          </form>
        </Tabs>
      </DialogContent>
    </Dialog>
  );
}

import React from "react";
import { Dialog, DialogContent, DialogHeader, DialogTitle } from "@/components/ui/dialog";
import { Badge } from "@/components/ui/badge";
import { Button } from "@/components/ui/button";
import { Copy, Settings, Trash2 } from "lucide-react";

const cardGradients = {
  visa: "from-blue-600 to-blue-800",
  mastercard: "from-red-500 to-red-700", 
  american_express: "from-green-600 to-green-800"
};

const cardLogos = {
  visa: "VISA",
  mastercard: "Mastercard",
  american_express: "AMEX"
};

export default function CardDetails({ card, open, onOpenChange }) {
  if (!card) return null;

  const copyToClipboard = (text) => {
    navigator.clipboard.writeText(text);
  };

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent className="sm:max-w-lg">
        <DialogHeader>
          <DialogTitle>Card Details</DialogTitle>
        </DialogHeader>
        
        <div className="space-y-6">
          {/* Card Visual */}
          <div className={`h-56 bg-gradient-to-br ${cardGradients[card.card_type]} p-6 text-white relative overflow-hidden rounded-xl`}>
            <div className="absolute top-0 right-0 w-32 h-32 bg-white/10 rounded-full -translate-y-16 translate-x-16" />
            <div className="absolute bottom-0 left-0 w-20 h-20 bg-white/10 rounded-full translate-y-10 -translate-x-10" />
            
            <div className="relative z-10 h-full flex flex-col justify-between">
              <div className="flex justify-between items-start">
                <div className="text-sm font-medium opacity-80">
                  {card.card_variant === 'virtual' ? 'VIRTUAL' : 'PHYSICAL'}
                </div>
                <div className="text-right">
                  <div className="text-xl font-bold">{cardLogos[card.card_type]}</div>
                  <div className="text-sm opacity-80">
                    {card.expiry_month}/{card.expiry_year}
                  </div>
                </div>
              </div>
              
              <div>
                <div className="text-xl font-mono tracking-wider mb-3">
                  •••• •••• •••• {card.card_number?.slice(-4)}
                </div>
                <div className="text-sm opacity-90">
                  {card.nickname || `${card.card_type.charAt(0).toUpperCase() + card.card_type.slice(1)} Card`}
                </div>
              </div>
            </div>
          </div>

          {/* Card Info */}
          <div className="space-y-4">
            <div className="grid grid-cols-2 gap-4">
              <div>
                <p className="text-sm text-gray-500">Status</p>
                <Badge variant={card.status === 'active' ? 'default' : card.status === 'frozen' ? 'secondary' : 'destructive'}>
                  {card.status}
                </Badge>
              </div>
              <div>
                <p className="text-sm text-gray-500">Daily Limit</p>
                <p className="font-semibold">${card.spending_limit?.toLocaleString()}</p>
              </div>
            </div>

            <div>
              <p className="text-sm text-gray-500 mb-2">Card Number</p>
              <div className="flex items-center gap-2">
                <code className="bg-gray-100 px-3 py-2 rounded flex-1 font-mono">
                  •••• •••• •••• {card.card_number?.slice(-4)}
                </code>
                <Button size="icon" variant="outline" onClick={() => copyToClipboard(card.card_number)}>
                  <Copy className="w-4 h-4" />
                </Button>
              </div>
            </div>

            <div className="grid grid-cols-2 gap-4">
              <div>

import React, { useState } from "react";
import { Dialog, DialogContent, DialogHeader, DialogTitle } from "@/components/ui/dialog";

import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import { RadioGroup, RadioGroupItem } from "@/components/ui/radio-group";

export default function CreateCardModal({ open, onOpenChange, onCreateCard }) {
  const [formData, setFormData] = useState({
    card_type: 'visa',
    card_variant: 'virtual',
    nickname: '',
    spending_limit: 5000
  });
  const [isCreating, setIsCreating] = useState(false);

  const generateCardNumber = () => {
    const lastFour = Math.floor(1000 + Math.random() * 9000);
    return `****-****-****-${lastFour}`;
  };

  const getCurrentYear = () => new Date().getFullYear();
  const getCurrentMonth = () => (new Date().getMonth() + 1).toString().padStart(2, '0');

  const handleSubmit = async (e) => {
    e.preventDefault();
    setIsCreating(true);

    try {
      const cardData = {
        ...formData,
        card_number: generateCardNumber(),
        expiry_month: getCurrentMonth(),
        expiry_year: (getCurrentYear() + 4).toString(),
        status: 'active'
      };

      await onCreateCard(cardData);
      setFormData({
        card_type: 'visa',
        card_variant: 'virtual',
        nickname: '',
        spending_limit: 5000
      });
    } finally {
      setIsCreating(false);
    }
  };

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent className="sm:max-w-md">
        <DialogHeader>
          <DialogTitle>Create New Card</DialogTitle>
        </DialogHeader>
        
        <form onSubmit={handleSubmit} className="space-y-6">
          {/* Card Type */}
          <div className="space-y-3">
            <Label>Card Type</Label>
            <Select
              value={formData.card_type}
              onValueChange={(value) => setFormData({...formData, card_type: value})}
            >
              <SelectTrigger>
                <SelectValue />
              </SelectTrigger>
              <SelectContent>
                <SelectItem value="visa">Visa</SelectItem>
                <SelectItem value="mastercard">Mastercard</SelectItem>
                <SelectItem value="american_express">American Express</SelectItem>
              </SelectContent>
            </Select>
          </div>

          {/* Card Variant */}
          <div className="space-y-3">
            <Label>Card Type</Label>
            <RadioGroup
              value={formData.card_variant}
              onValueChange={(value) => setFormData({...formData, card_variant: value})}
              className="flex gap-6"
            >
              <div className="flex items-center space-x-2">
                <RadioGroupItem value="virtual" id="virtual" />
                <Label htmlFor="virtual">Virtual Card</Label>
              </div>
              <div className="flex items-center space-x-2">
                <RadioGroupItem value="physical" id="physical" />
                <Label htmlFor="physical">Physical Card</Label>
              </div>
            </RadioGroup>
          </div>

          {/* Nickname */}
          <div className="space-y-2">
            <Label htmlFor="nickname">Card Nickname (Optional)</Label>
            <Input
              id="nickname"
              placeholder="My Shopping Card"
              value={formData.nickname}
              onChange={(e) => setFormData({...formData, nickname: e.target.value})}
            />
          </div>

          {/* Spending Limit */}
          <div className="space-y-2">
            <Label htmlFor="limit">Daily Spending Limit</Label>
            <Input
              id="limit"
              type="number"
              min="100"
              max="25000"
