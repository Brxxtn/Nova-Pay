import React, { useState } from "react";
import { base44 } from "@/api/base44Client";
import { Dialog, DialogContent, DialogHeader, DialogTitle } from "@/components/ui/dialog";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import { Textarea } from "@/components/ui/textarea";
import { toast } from "sonner";

    goal_name: "",
    goal_type: "retirement",
    target_amount: "",
    target_date: "",
    priority: "medium",
    monthly_contribution: "",
    notes: ""

              <SelectTrigger>
                <SelectValue />
              </SelectTrigger>

                onChange={(e) => setFormData({ ...formData, target_date: e.target.value })}

import React, { useState } from "react";
import { Dialog, DialogContent, DialogHeader, DialogTitle } from "@/components/ui/dialog";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Progress } from "@/components/ui/progress";
import { Badge } from "@/components/ui/badge";
import { Plus, Minus, Trash2, Check } from "lucide-react";
import { format } from "date-fns";

export default function GoalDetails({ goal, onClose, onUpdate, onDelete }) {
  const [addAmount, setAddAmount] = useState("");
  const [withdrawAmount, setWithdrawAmount] = useState("");

  const progress = (goal.current_amount || 0) / goal.target_amount * 100;
  const remaining = goal.target_amount - (goal.current_amount || 0);
  const daysLeft = Math.ceil((new Date(goal.target_date) - new Date()) / (1000 * 60 * 60 * 24));

  const handleAddFunds = async () => {
    if (!addAmount || isNaN(addAmount)) return;
    const newAmount = (goal.current_amount || 0) + parseFloat(addAmount);
    await onUpdate(goal.id, { 
      current_amount: newAmount,
      status: newAmount >= goal.target_amount ? 'completed' : goal.status
    });
    setAddAmount("");
  };

  const handleWithdraw = async () => {
    if (!withdrawAmount || isNaN(withdrawAmount)) return;
    const newAmount = Math.max(0, (goal.current_amount || 0) - parseFloat(withdrawAmount));
    await onUpdate(goal.id, { 
      current_amount: newAmount,
      status: 'active'
    });
    setWithdrawAmount("");
  };

  const handleMarkComplete = async () => {
    await onUpdate(goal.id, { status: 'completed' });
  };

  return (
    <Dialog open={true} onOpenChange={onClose}>
      <DialogContent className="max-w-2xl">
        <DialogHeader>
          <DialogTitle className="flex items-center justify-between">
            <span>{goal.goal_name}</span>
            <Badge variant={goal.status === 'completed' ? 'default' : 'secondary'} className="capitalize">
              {goal.status}
            </Badge>
          </DialogTitle>
        </DialogHeader>

        <div className="space-y-6">
          {/* Progress Section */}
          <div className="bg-gradient-to-r from-green-50 to-emerald-50 rounded-xl p-6">
            <div className="flex items-center justify-between mb-4">
              <span className="text-sm text-gray-600">Progress</span>
              <span className="text-2xl font-bold text-green-700">{progress.toFixed(0)}%</span>
            </div>
            <Progress value={progress} className="h-3 mb-4" />
            <div className="grid grid-cols-2 gap-4 text-center">
              <div>
                <p className="text-sm text-gray-600">Current</p>
                <p className="text-2xl font-bold text-gray-900">${(goal.current_amount || 0).toFixed(2)}</p>
              </div>
              <div>
                <p className="text-sm text-gray-600">Target</p>
                <p className="text-2xl font-bold text-gray-900">${goal.target_amount.toFixed(2)}</p>
              </div>
            </div>
          </div>

          {/* Stats */}
          <div className="grid grid-cols-3 gap-4">
            <div className="text-center p-4 bg-gray-50 rounded-lg">
              <p className="text-sm text-gray-600">Remaining</p>
              <p className="text-xl font-bold text-gray-900">${remaining.toFixed(2)}</p>
            </div>
            <div className="text-center p-4 bg-gray-50 rounded-lg">
              <p className="text-sm text-gray-600">Days Left</p>
              <p className="text-xl font-bold text-gray-900">{daysLeft > 0 ? daysLeft : 'Overdue'}</p>
            </div>
            <div className="text-center p-4 bg-gray-50 rounded-lg">
              <p className="text-sm text-gray-600">Target Date</p>
              <p className="text-sm font-bold text-gray-900">{format(new Date(goal.target_date), 'MMM d, yyyy')}</p>
            </div>
          </div>

          {/* Add/Withdraw Funds */}
          {goal.status !== 'completed' && (
            <div className="grid grid-cols-2 gap-4">
              <div className="space-y-2">
                <label className="text-sm font-medium">Add Funds</label>
                <div className="flex gap-2">
                  <Input
                    type="number"
                    step="0.01"
                    placeholder="Amount"
                    value={addAmount}
                    onChange={(e) => setAddAmount(e.target.value)}
                  />
                  <Button onClick={handleAddFunds} className="bg-green-600 hover:bg-green-700">
                    <Plus className="w-4 h-4" />
                  </Button>
                </div>
              </div>
              <div className="space-y-2">
                <label className="text-sm font-medium">Withdraw</label>
                <div className="flex gap-2">
                  <Input
                    type="number"
                    step="0.01"
                    placeholder="Amount"
                    value={withdrawAmount}
                    onChange={(e) => setWithdrawAmount(e.target.value)}
                  />
                  <Button onClick={handleWithdraw} variant="outline">
                    <Minus className="w-4 h-4" />
                  </Button>
                </div>
              </div>
            </div>
          )}

          {/* Notes */}
          {goal.notes && (
            <div className="bg-gray-50 rounded-lg p-4">
              <p className="text-sm font-medium text-gray-600 mb-2">Notes</p>
              <p className="text-gray-900">{goal.notes}</p>
            </div>
          )}

          {/* Actions */}
          <div className="flex gap-3 justify-between pt-4 border-t">
            <Button
              variant="destructive"
              onClick={() => onDelete(goal.id)}
            >
              <Trash2 className="w-4 h-4 mr-2" />
              Delete Goal
            </Button>
            
            {goal.status !== 'completed' && (
              <Button
                className="bg-green-600 hover:bg-green-700"
                onClick={handleMarkComplete}
              >
                <Check className="w-4 h-4 mr-2" />
                Mark as Completed
              </Button>
            )}
          </div>
        </div>
      </DialogContent>
    </Dialog>
  );
}

import React, { useState, useEffect } from "react";
import { base44 } from "@/api/base44Client";
import { Dialog, DialogContent, DialogHeader, DialogTitle } from "@/components/ui/dialog";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Badge } from "@/components/ui/badge";
import { Progress } from "@/components/ui/progress";
import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";
import { Sparkles, Plus, Trash2, TrendingUp } from "lucide-react";
import { toast } from "sonner";

export default function GoalDetailsModal({ goal, onClose, onUpdate, cryptoHoldings, etoroHoldings }) {
  const [aiRecommendations, setAiRecommendations] = useState(goal.ai_recommendations || []);
  const [loadingAI, setLoadingAI] = useState(false);
  const [selectedCryptos, setSelectedCryptos] = useState(goal.allocated_cryptos || []);
  const [selectedStocks, setSelectedStocks] = useState(goal.allocated_stocks || []);

  const calculateCurrentValue = () => {
    let value = 0;
    
    selectedCryptos.forEach(allocation => {
      const holding = cryptoHoldings.find(h => h.crypto_symbol === allocation.symbol);
      if (holding) value += holding.current_value || 0;
    });

    selectedStocks.forEach(allocation => {
      const holding = etoroHoldings.find(h => h.symbol === allocation.symbol);
      if (holding) value += holding.total_value || 0;
    });

    return value;
  };

  const progress = (calculateCurrentValue() / goal.target_amount) * 100;

  const generateAIRecommendations = async () => {
    setLoadingAI(true);
    try {
      const response = await base44.integrations.Core.InvokeLLM({
        prompt: `As a financial advisor AI, analyze this investment goal and provide actionable recommendations:

Goal Details:
- Name: ${goal.goal_name}
- Type: ${goal.goal_type}
- Target Amount: $${goal.target_amount}
- Current Progress: ${progress.toFixed(1)}%
- Target Date: ${goal.target_date}
- Days Remaining: ${Math.ceil((new Date(goal.target_date) - new Date()) / (1000 * 60 * 60 * 24))}
- Monthly Contribution: $${goal.monthly_contribution || 0}

Current Allocations:
- Cryptocurrencies: ${selectedCryptos.map(c => c.symbol).join(', ') || 'None'}
- Stocks: ${selectedStocks.map(s => s.symbol).join(', ') || 'None'}

Provide 5 specific, actionable recommendations to help achieve this goal faster. Consider:
1. Portfolio diversification
2. Monthly contribution adjustments
3. Asset rebalancing opportunities
4. Risk management
5. Time-based strategies

Format as JSON array of strings.`,
        add_context_from_internet: true,
        response_json_schema: {
          type: "object",
          properties: {
            recommendations: {
              type: "array",
              items: { type: "string" }
            }
          }
        }
      });

      const recommendations = response.recommendations || [];
      setAiRecommendations(recommendations);

      // Update goal with recommendations
      await base44.entities.InvestmentGoal.update(goal.id, {
        ai_recommendations: recommendations
      });

      toast.success("AI recommendations generated!");
    } catch (error) {
      console.error("Error generating recommendations:", error);
      toast.error("Failed to generate recommendations");
    } finally {
      setLoadingAI(false);
    }
  };

  const allocateCrypto = async (symbol) => {
    const holding = cryptoHoldings.find(h => h.crypto_symbol === symbol);
    if (!holding) return;

    const newAllocation = {
      symbol: holding.crypto_symbol,
      quantity: holding.quantity,
      value_usd: holding.current_value
    };

    const updated = [...selectedCryptos, newAllocation];
    setSelectedCryptos(updated);

    try {
      await base44.entities.InvestmentGoal.update(goal.id, {
        allocated_cryptos: updated
      });
      toast.success(`${symbol} allocated to goal`);
    } catch (error) {
      console.error("Error allocating crypto:", error);
      toast.error("Failed to allocate");
    }
  };

  const allocateStock = async (symbol) => {
    const holding = etoroHoldings.find(h => h.symbol === symbol);
    if (!holding) return;

    const newAllocation = {
      symbol: holding.symbol,
      quantity: holding.quantity,
      value_usd: holding.total_value
    };

    const updated = [...selectedStocks, newAllocation];
    setSelectedStocks(updated);

    try {
      await base44.entities.InvestmentGoal.update(goal.id, {
        allocated_stocks: updated
      });
      toast.success(`${symbol} allocated to goal`);
    } catch (error) {
      console.error("Error allocating stock:", error);
      toast.error("Failed to allocate");
    }
  };

  const removeAllocation = async (symbol, type) => {
    if (type === 'crypto') {
      const updated = selectedCryptos.filter(c => c.symbol !== symbol);
      setSelectedCryptos(updated);
      await base44.entities.InvestmentGoal.update(goal.id, { allocated_cryptos: updated });
    } else {
      const updated = selectedStocks.filter(s => s.symbol !== symbol);
      setSelectedStocks(updated);
      await base44.entities.InvestmentGoal.update(goal.id, { allocated_stocks: updated });
    }
    toast.success("Allocation removed");
  };

  return (
    <Dialog open onOpenChange={onClose}>
      <DialogContent className="max-w-3xl max-h-[90vh] overflow-y-auto">
        <DialogHeader>
          <DialogTitle className="flex items-center justify-between">
            <span>{goal.goal_name}</span>
            <Badge>{goal.status}</Badge>
          </DialogTitle>
        </DialogHeader>

        <div className="space-y-6 mt-4">
          {/* Progress Overview */}
          <div className="p-4 bg-gradient-to-br from-blue-50 to-indigo-50 rounded-lg">
            <div className="flex items-center justify-between mb-2">
              <p className="text-sm text-gray-600">Progress</p>
              <p className="text-2xl font-bold text-gray-900">
                ${calculateCurrentValue().toLocaleString()}
              </p>
            </div>
            <Progress value={progress} className="mb-2" />
            <div className="flex items-center justify-between text-sm">
              <span className="text-gray-600">{progress.toFixed(1)}% Complete</span>
              <span className="text-green-600">
                Target: ${goal.target_amount.toLocaleString()}
              </span>
            </div>
          </div>

          <Tabs defaultValue="allocations">
            <TabsList className="w-full">
              <TabsTrigger value="allocations" className="flex-1">Allocations</TabsTrigger>
              <TabsTrigger value="recommendations" className="flex-1">AI Insights</TabsTrigger>
            </TabsList>

            <TabsContent value="allocations" className="space-y-4">
              {/* Crypto Allocations */}
              <div>
                <Label className="mb-2 block">Allocated Cryptocurrencies</Label>
                <div className="space-y-2">
                  {selectedCryptos.map((allocation) => (
                    <div key={allocation.symbol} className="flex items-center justify-between p-3 bg-gray-50 rounded-lg">
                      <div>
                        <p className="font-semibold">{allocation.symbol}</p>
                        <p className="text-sm text-gray-600">
                          {allocation.quantity} units • ${allocation.value_usd?.toLocaleString()}
                        </p>
                      </div>
                      <Button
                        size="sm"
                        variant="ghost"
                        onClick={() => removeAllocation(allocation.symbol, 'crypto')}
                        className="text-red-600"
                      >
                        <Trash2 className="w-4 h-4" />
                      </Button>
                    </div>
                  ))}
                  
                  {cryptoHoldings.filter(h => !selectedCryptos.find(c => c.symbol === h.crypto_symbol)).length > 0 && (
                    <div className="pt-2">
                      <Label className="text-xs text-gray-600">Available to allocate:</Label>
                      <div className="flex flex-wrap gap-2 mt-2">
                        {cryptoHoldings
                          .filter(h => !selectedCryptos.find(c => c.symbol === h.crypto_symbol))
                          .map((holding) => (
                            <Button
                              key={holding.crypto_symbol}
                              size="sm"
                              variant="outline"
                              onClick={() => allocateCrypto(holding.crypto_symbol)}
                            >
                              <Plus className="w-3 h-3 mr-1" />
                              {holding.crypto_symbol}
                            </Button>
                          ))}
                      </div>
                    </div>
                  )}
                </div>
              </div>

              {/* Stock Allocations */}
              <div>
                <Label className="mb-2 block">Allocated Stocks</Label>
                <div className="space-y-2">
                  {selectedStocks.map((allocation) => (
                    <div key={allocation.symbol} className="flex items-center justify-between p-3 bg-gray-50 rounded-lg">
                      <div>
                        <p className="font-semibold">{allocation.symbol}</p>

  const [overallAdvice, setOverallAdvice] = useState(null);

    setIsAnalyzing(true);
    setRecommendations(null);
    setOverallAdvice(null);

    try {
      const safeTransactions = Array.isArray(transactions) ? transactions : [];

      const last30Days = safeTransactions.filter(t => {
        const txDate = new Date(t.created_date);
        const daysAgo = (Date.now() - txDate.getTime()) / (1000 * 60 * 60 * 24);
        return daysAgo <= 30;
      });

      const monthlyIncome = last30Days
        .filter(t => ['credit', 'transfer_in'].includes(t.type))
        .reduce((sum, t) => sum + t.amount, 0);

      const monthlyExpenses = last30Days
        .filter(t => ['debit', 'transfer_out'].includes(t.type))
        .reduce((sum, t) => sum + Math.abs(t.amount), 0);

      const currentSavings = account?.balance || 0;
      const existingGoalsCount = goals.length;

      const prompt = `You are a financial advisor helping someone set realistic savings goals.

USER'S FINANCIAL SITUATION:
- Monthly Income: $${monthlyIncome.toFixed(2)}
- Monthly Expenses: $${monthlyExpenses.toFixed(2)}
- Current Savings: $${currentSavings.toFixed(2)}
- Existing Goals: ${existingGoalsCount}

Based on their finances, recommend 3-5 specific, achievable savings goals they should consider.

For each recommended goal:
- goal_name: specific name (e.g., "Emergency Fund", "Hawaii Vacation")
- target_amount: realistic dollar amount
- suggested_monthly_contribution: what they can afford monthly
- priority: high/medium/low
- timeframe_months: how many months to reach goal
- reasoning: why this goal matters
- category: emergency/vacation/house/car/education/retirement/other

Also provide overall_advice: general strategy for their situation.

Be realistic about what they can afford based on their income minus expenses.`;

        prompt: prompt,

            recommendations: {
              type: "array",
              items: {
                type: "object",
                properties: {
                  goal_name: { type: "string" },

                  suggested_monthly_contribution: { type: "number" },
                  priority: { type: "string" },

                  reasoning: { type: "string" },
                  category: { type: "string" }

          }
        }

      setOverallAdvice(response.overall_advice);

      alert("Failed to generate recommendations. Please try again.");

      setIsAnalyzing(false);
    }
  };

      <CardHeader>

                      <div className="flex items-center justify-between mb-1">

import React, { useState, useEffect } from "react";

export default function InvestmentGoalTracker() {
  const [goals, setGoals] = useState([]);
  const [cryptoHoldings, setCryptoHoldings] = useState([]);
  const [etoroHoldings, setEtoroHoldings] = useState([]);
  const [loading, setLoading] = useState(true);
  const [showCreateModal, setShowCreateModal] = useState(false);
  const [selectedGoal, setSelectedGoal] = useState(null);

  useEffect(() => {
    loadData();
  }, []);

  const loadData = async () => {
    try {
      const user = await base44.auth.me();
      
      const [goalsData, cryptoData, etoroData] = await Promise.all([
        base44.entities.InvestmentGoal.filter({ created_by: user.email }),
        base44.entities.CryptoHolding.filter({ created_by: user.email }),
        base44.entities.EToroHolding.filter({ created_by: user.email })
      ]);

      setGoals(goalsData);
      setCryptoHoldings(cryptoData);
      setEtoroHoldings(etoroData);
    } catch (error) {
      console.error("Error loading investment goals:", error);
    } finally {
      setLoading(false);
    }
  };

  const calculateProgress = (goal) => {
    let currentValue = 0;

    // Calculate value from allocated cryptos
    if (goal.allocated_cryptos && goal.allocated_cryptos.length > 0) {
      goal.allocated_cryptos.forEach(allocation => {
        const holding = cryptoHoldings.find(h => h.crypto_symbol === allocation.symbol);
        if (holding) {
          currentValue += holding.current_value || 0;
        }
      });
    }

    // Calculate value from allocated stocks
    if (goal.allocated_stocks && goal.allocated_stocks.length > 0) {
      goal.allocated_stocks.forEach(allocation => {
        const holding = etoroHoldings.find(h => h.symbol === allocation.symbol);
        if (holding) {
          currentValue += holding.total_value || 0;
        }
      });
    }

    const progress = (currentValue / goal.target_amount) * 100;
    return { currentValue, progress: Math.min(progress, 100) };
  };

  const getPriorityColor = (priority) => {
    switch (priority) {
      case "critical": return "bg-red-600";
      case "high": return "bg-orange-600";
      case "medium": return "bg-yellow-600";
      case "low": return "bg-green-600";
      default: return "bg-gray-600";
    }
  };

  const getStatusColor = (status) => {
    switch (status) {
      case "completed": return "bg-green-600";
      case "active": return "bg-blue-600";
      case "paused": return "bg-gray-600";
      default: return "bg-gray-600";
    }
  };

  const deleteGoal = async (goalId) => {
    if (!confirm("Are you sure you want to delete this goal?")) return;
    
    try {
      await base44.entities.InvestmentGoal.delete(goalId);
      setGoals(goals.filter(g => g.id !== goalId));
    } catch (error) {
      console.error("Error deleting goal:", error);
    }
  };

  const totalTargetAmount = goals.reduce((sum, g) => sum + (g.target_amount || 0), 0);
  const totalCurrentAmount = goals.reduce((sum, g) => {
    const { currentValue } = calculateProgress(g);
    return sum + currentValue;
  }, 0);
  const overallProgress = totalTargetAmount > 0 ? (totalCurrentAmount / totalTargetAmount) * 100 : 0;

  return (
    <>
      <Card className="shadow-lg">
        <CardHeader>
          <div className="flex items-center justify-between">
            <CardTitle className="flex items-center gap-2">
              <Target className="w-6 h-6 text-blue-600" />
              Investment Goals
            </CardTitle>
            <Button onClick={() => setShowCreateModal(true)} size="sm">
              <Plus className="w-4 h-4 mr-2" />
              New Goal
            </Button>
          </div>
        </CardHeader>
        <CardContent className="space-y-6">
          {/* Overall Progress */}
          <div className="p-4 bg-gradient-to-br from-blue-50 to-indigo-50 rounded-lg">
            <div className="flex items-center justify-between mb-2">
              <p className="text-sm text-gray-600">Total Goals Progress</p>
              <p className="text-2xl font-bold text-gray-900">
                ${totalCurrentAmount.toLocaleString()}
              </p>
            </div>
            <Progress value={overallProgress} className="mb-2" />
            <div className="flex items-center justify-between text-sm">
              <span className="text-gray-600">
                {goals.length} {goals.length === 1 ? 'goal' : 'goals'} active
              </span>
              <span className="text-green-600">
                Target: ${totalTargetAmount.toLocaleString()}
              </span>
            </div>
          </div>

          {/* Individual Goals */}
          {loading ? (
            <div className="flex items-center justify-center py-8">
              <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-blue-600"></div>
            </div>
          ) : goals.length === 0 ? (
            <div className="text-center py-8">
              <Target className="w-12 h-12 text-gray-400 mx-auto mb-3" />
              <p className="text-gray-600 mb-4">No investment goals yet</p>
              <Button onClick={() => setShowCreateModal(true)} size="sm">
                Create Your First Goal
              </Button>
            </div>
          ) : (
            <div className="space-y-4">
              {goals.map((goal) => {
                const { currentValue, progress } = calculateProgress(goal);
                const daysRemaining = Math.ceil(
                  (new Date(goal.target_date) - new Date()) / (1000 * 60 * 60 * 24)
                );

                return (
                  <div
                    key={goal.id}
                    className="p-4 rounded-lg border bg-white hover:shadow-md transition-shadow cursor-pointer"
                    onClick={() => setSelectedGoal(goal)}
                  >
                    <div className="flex items-start justify-between mb-3">
                      <div className="flex-1">
                        <div className="flex items-center gap-2 mb-1">
                          <h4 className="font-semibold text-gray-900">{goal.goal_name}</h4>
                          <Badge className={getPriorityColor(goal.priority)}>
                            {goal.priority}
                          </Badge>
                          <Badge className={getStatusColor(goal.status)}>
                            {goal.status}
                          </Badge>
                        </div>
                        <p className="text-xs text-gray-600 capitalize">{goal.goal_type.replace('_', ' ')}</p>
                      </div>
                      <div className="flex items-center gap-2">
                        <Button
                          size="sm"
                          variant="ghost"
                          onClick={(e) => {
                            e.stopPropagation();
                            setSelectedGoal(goal);
                          }}
                        >
                          <Edit2 className="w-4 h-4" />
                        </Button>
                        <Button
                          size="sm"
                          variant="ghost"
                          onClick={(e) => {
                            e.stopPropagation();
                            deleteGoal(goal.id);
                          }}
                          className="text-red-600"
                        >
                          <Trash2 className="w-4 h-4" />
                        </Button>
                      </div>
                    </div>

                    <Progress value={progress} className="mb-2" />
                    
                    <div className="flex items-center justify-between text-sm mb-2">
                      <span className="text-gray-600">
                        ${currentValue.toLocaleString()} / ${goal.target_amount.toLocaleString()}
                      </span>
                      <span className="font-semibold text-blue-600">
                        {progress.toFixed(1)}%
                      </span>
                    </div>

                    <div className="flex items-center justify-between text-xs text-gray-500">
                      <span>{daysRemaining} days remaining</span>
                      {goal.monthly_contribution > 0 && (
                        <span className="flex items-center gap-1">
                          <TrendingUp className="w-3 h-3" />
                          ${goal.monthly_contribution}/mo
                        </span>
                      )}
                    </div>

                    {goal.ai_recommendations && goal.ai_recommendations.length > 0 && (
                      <div className="mt-3 p-2 bg-purple-50 rounded-lg border border-purple-200">
                        <div className="flex items-start gap-2">
                          <Sparkles className="w-4 h-4 text-purple-600 mt-0.5 flex-shrink-0" />
                          <p className="text-xs text-purple-800">
                            {goal.ai_recommendations[0]}
                          </p>
                        </div>
                      </div>
                    )}
                  </div>
                );
              })}
            </div>
          )}
        </CardContent>
      </Card>

      {showCreateModal && (
        <CreateInvestmentGoalModal
          onClose={() => setShowCreateModal(false)}
          onSuccess={() => {
            loadData();
            setShowCreateModal(false);
          }}
          cryptoHoldings={cryptoHoldings}
          etoroHoldings={etoroHoldings}
        />
      )}

      {selectedGoal && (
        <GoalDetailsModal
          goal={selectedGoal}
          onClose={() => setSelectedGoal(null)}
          onUpdate={() => {
            loadData();
            setSelectedGoal(null);
          }}
          cryptoHoldings={cryptoHoldings}
          etoroHoldings={etoroHoldings}
        />
      )}
    </>
  );
}

  })();
  
  const isError = results && (
    (typeof results === 'string' && /error|failed/i.test(results)) ||
    (parsedResults?.success === false)
  );
  
  const statusConfig = {
    pending: { icon: Clock, color: 'text-slate-400', text: 'Pending' },
    running: { icon: Loader2, color: 'text-slate-500', text: 'Running...', spin: true },
    in_progress: { icon: Loader2, color: 'text-slate-500', text: 'Running...', spin: true },
    completed: isError ? 
      { icon: AlertCircle, color: 'text-red-500', text: 'Failed' } : 
      { icon: CheckCircle2, color: 'text-green-600', text: 'Success' },
    success: { icon: CheckCircle2, color: 'text-green-600', text: 'Success' },
    failed: { icon: AlertCircle, color: 'text-red-500', text: 'Failed' },
    error: { icon: AlertCircle, color: 'text-red-500', text: 'Failed' }
  }[status] || { icon: Zap, color: 'text-slate-500', text: '' };
  
  const Icon = statusConfig.icon;
  const formattedName = name.split('.').reverse().join(' ').toLowerCase();
  
  return (
    <div className="mt-2 text-xs">
      <button
        onClick={() => setExpanded(!expanded)}
        className={cn(
          "flex items-center gap-2 px-3 py-1.5 rounded-lg border transition-all",
          "hover:bg-slate-50",
          expanded ? "bg-slate-50 border-slate-300" : "bg-white border-slate-200"
        )}
      >
        <Icon className={cn("h-3 w-3", statusConfig.color, statusConfig.spin && "animate-spin")} />
        <span className="text-slate-700">{formattedName}</span>
        {statusConfig.text && (
          <span className={cn("text-slate-500", isError && "text-red-600")}>
            • {statusConfig.text}
          </span>
        )}
        {!statusConfig.spin && (toolCall.arguments_string || results) && (
          <ChevronRight className={cn("h-3 w-3 text-slate-400 transition-transform ml-auto", 
            expanded && "rotate-90")} />
        )}
      </button>
      
      {expanded && !statusConfig.spin && (
        <div className="mt-1.5 ml-3 pl-3 border-l-2 border-slate-200 space-y-2">
          {toolCall.arguments_string && (
            <div>
              <div className="text-xs text-slate-500 mb-1">Parameters:</div>
              <pre className="bg-slate-50 rounded-md p-2 text-xs text-slate-600 whitespace-pre-wrap">
                {(() => {
                  try {
                    return JSON.stringify(JSON.parse(toolCall.arguments_string), null, 2);
                  } catch {
                    return toolCall.arguments_string;
                  }
                })()}
              </pre>
            </div>
          )}
          {parsedResults && (
            <div>
              <div className="text-xs text-slate-500 mb-1">Result:</div>
              <pre className="bg-slate-50 rounded-md p-2 text-xs text-slate-600 whitespace-pre-wrap max-h-48 overflow-auto">
                {typeof parsedResults === 'object' ? 
                  JSON.stringify(parsedResults, null, 2) : parsedResults}
              </pre>
            </div>
          )}
        </div>
      )}
    </div>
  );
};

export default function MessageBubble({ message }) {
  const isUser = message.role === 'user';
  
  return (
    <div className={cn("flex gap-3", isUser ? "justify-end" : "justify-start")}>
      {!isUser && (
        <div className="h-7 w-7 rounded-lg bg-gradient-to-br from-blue-600 to-indigo-600 flex items-center justify-center mt-0.5">
          <div className="text-white text-xs font-bold">J</div>
        </div>
      )}
      <div className={cn("max-w-[85%]", isUser && "flex flex-col items-end")}>
        {message.content && (
          <div className={cn(
            "rounded-2xl px-4 py-2.5",
            isUser ? "bg-slate-800 text-white" : "bg-white border border-slate-200"
          )}>
            {isUser ? (
              <p className="text-sm leading-relaxed">{message.content}</p>
            ) : (
              <ReactMarkdown 
                className="text-sm prose prose-sm prose-slate max-w-none [&>*:first-child]:mt-0 [&>*:last-child]:mb-0"
                components={{
                  code: ({ inline, className, children, ...props }) => {
                    const match = /language-(\w+)/.exec(className || '');
                    return !inline && match ? (
                      <div className="relative group/code">
                        <pre className="bg-slate-900 text-slate-100 rounded-lg p-3 overflow-x-auto my-2">
                          <code className={className} {...props}>{children}</code>
                        </pre>
                        <Button
                          size="icon"
                          variant="ghost"
                          className="absolute top-2 right-2 h-6 w-6 opacity-0 group-hover/code:opacity-100 bg-slate-800 hover:bg-slate-700"
                          onClick={() => {
                            navigator.clipboard.writeText(String(children).replace(/\n$/, ''));
                          }}
                        >
                          <Copy className="h-3 w-3 text-slate-400" />
                        </Button>
                      </div>
                    ) : (
                      <code className="px-1 py-0.5 rounded bg-slate-100 text-slate-700 text-xs">
                        {children}
                      </code>
                    );
                  },
                  a: ({ children, ...props }) => (
                    <a {...props} target="_blank" rel="noopener noreferrer">{children}</a>
                  ),
                  p: ({ children }) => <p className="my-1 leading-relaxed">{children}</p>,
                  ul: ({ children }) => <ul className="my-1 ml-4 list-disc">{children}</ul>,
                  ol: ({ children }) => <ol className="my-1 ml-4 list-decimal">{children}</ol>,
                  li: ({ children }) => <li className="my-0.5">{children}</li>,
                  h1: ({ children }) => <h1 className="text-lg font-semibold my-2">{children}</h1>,
                  h2: ({ children }) => <h2 className="text-base font-semibold my-2">{children}</h2>,
                  h3: ({ children }) => <h3 className="text-sm font-semibold my-2">{children}</h3>,
                  blockquote: ({ children }) => (
                    <blockquote className="border-l-2 border-slate-300 pl-3 my-2 text-slate-600">
                      {children}
                    </blockquote>
                  ),
                }}
              >
                {message.content}
              </ReactMarkdown>
            )}
          </div>
        )}
        
        {message.tool_calls?.length > 0 && (
          <div className="space-y-1">
            {message.tool_calls.map((toolCall, idx) => (
              <FunctionDisplay key={idx} toolCall={toolCall} />
            ))}
          </div>
        )}
      </div>
    </div>
  );
}
