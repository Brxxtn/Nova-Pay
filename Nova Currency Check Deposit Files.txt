import React, { useState, useRef } from "react";
import { base44 } from "@/api/base44Client";
import { Dialog, DialogContent, DialogHeader, DialogTitle } from "@/components/ui/dialog";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Camera, Upload, CheckCircle2, AlertCircle, Loader2 } from "lucide-react";
import { Card, CardContent } from "@/components/ui/card";

export default function CheckCapture({ open, onClose, onDeposit, account }) {
  const [step, setStep] = useState(1);
  const [frontImage, setFrontImage] = useState(null);
  const [backImage, setBackImage] = useState(null);
  const [frontImageUrl, setFrontImageUrl] = useState("");
  const [backImageUrl, setBackImageUrl] = useState("");
  const [isProcessing, setIsProcessing] = useState(false);
  const [extractedData, setExtractedData] = useState(null);
  const [imageQuality, setImageQuality] = useState({ front: null, back: null });
  const [depositSuccess, setDepositSuccess] = useState(false);
  
  const [checkData, setCheckData] = useState({
    check_amount: "",
    check_number: "",
    memo: ""
  });

  const frontInputRef = useRef(null);
  const backInputRef = useRef(null);

  const analyzeImageQuality = (file) => {
    return new Promise((resolve) => {
      const img = new Image();
      const reader = new FileReader();
      
      reader.onload = (e) => {
        img.onload = () => {
          const quality = {
            width: img.width,
            height: img.height,
            size: file.size,
            score: 0,
            issues: []
          };

          // Check resolution
          if (img.width < 800 || img.height < 600) {
            quality.issues.push("Low resolution - use higher quality camera");
            quality.score += 20;
          } else if (img.width >= 1920 && img.height >= 1080) {
            quality.score += 40;
          } else {
            quality.score += 30;
          }

          // Check file size (indicator of compression/quality)
          if (file.size < 200000) {
            quality.issues.push("Image may be too compressed");
            quality.score += 10;
          } else if (file.size > 5000000) {
            quality.issues.push("Large file size - may take longer to process");
            quality.score += 30;
          } else {
            quality.score += 40;
          }

          // Basic brightness check would require canvas analysis
          quality.score += 20; // Baseline score

          if (quality.score >= 80) {
            quality.rating = "Excellent";
          } else if (quality.score >= 60) {
            quality.rating = "Good";
          } else {
            quality.rating = "Poor";
          }

          resolve(quality);
        };
        img.src = e.target.result;
      };
      reader.readAsDataURL(file);
    });
  };

  const handleImageUpload = async (file, side) => {
    if (!file) return;

    setIsProcessing(true);
    try {
      // Analyze image quality first
      const quality = await analyzeImageQuality(file);
      if (side === 'front') {
        setImageQuality({ ...imageQuality, front: quality });
      } else {
        setImageQuality({ ...imageQuality, back: quality });
      }

      // Upload image to server
      const { file_url } = await base44.integrations.Core.UploadFile({ file });

      if (side === 'front') {
        setFrontImage(file);
        setFrontImageUrl(file_url);
        
        // Try to extract data from front of check
        try {
          const extractResult = await base44.integrations.Core.ExtractDataFromUploadedFile({
            file_url: file_url,
            json_schema: {
              type: "object",
              properties: {
                check_amount: { type: "number" },
                check_number: { type: "string" },
                routing_number: { type: "string" },
                account_number: { type: "string" },
                payee_name: { type: "string" },
                memo: { type: "string" }
              }
            }
          });

          if (extractResult.status === "success" && extractResult.output) {
            setExtractedData(extractResult.output);
            setCheckData({
              check_amount: extractResult.output.check_amount || "",
              check_number: extractResult.output.check_number || "",
              memo: extractResult.output.memo || ""
            });
          }
        } catch (error) {
          console.log("Could not extract check data automatically");
        }
        
        setStep(2);
      } else {
        setBackImage(file);
        setBackImageUrl(file_url);
        
        // Try to extract endorsement data from back
        try {
          const backExtractResult = await base44.integrations.Core.ExtractDataFromUploadedFile({
            file_url: file_url,
            json_schema: {
              type: "object",
              properties: {
                endorsement_present: { type: "boolean" },
                endorsement_text: { type: "string" }
              }
            }
          });

          if (backExtractResult.status === "success" && backExtractResult.output) {
            console.log("Back endorsement detected:", backExtractResult.output);
          }
        } catch (error) {
          console.log("Could not analyze check back");
        }
        
        setStep(3);
      }
    } catch (error) {
      console.error("Error uploading image:", error);
      alert("Failed to upload image. Please try again.");
    } finally {
      setIsProcessing(false);
    }
  };

  const handleSubmit = async () => {
    if (!frontImageUrl) {
      alert("Please capture the front of the check");
      return;
    }

    if (!checkData.check_amount || parseFloat(checkData.check_amount) <= 0) {
      alert("Please enter a valid check amount");
      return;
    }

    setIsProcessing(true);
    
    try {
      const imageQualityScore = imageQuality.front?.score || 85;

      await onDeposit({
        check_amount: parseFloat(checkData.check_amount),
        check_number: checkData.check_number,
        front_image_url: frontImageUrl,
        back_image_url: backImageUrl,
        routing_number: extractedData?.routing_number || "",
        account_number: extractedData?.account_number || "",
        payee_name: extractedData?.payee_name || account?.created_by || "",
        memo: checkData.memo,
        image_quality_score: imageQualityScore
      });

      // Show success confirmation
      setDepositSuccess(true);
      setStep(4);
      
      // Auto-close after 3 seconds
      setTimeout(() => {
        onClose();
        // Reset form
        setStep(1);
        setFrontImage(null);
        setBackImage(null);
        setFrontImageUrl("");
        setBackImageUrl("");
        setCheckData({ check_amount: "", check_number: "", memo: "" });
        setExtractedData(null);
        setImageQuality({ front: null, back: null });
        setDepositSuccess(false);
      }, 3000);
    } catch (error) {
      console.error("Error submitting deposit:", error);
      alert("Failed to submit deposit. Please try again.");
    } finally {
      setIsProcessing(false);
    }
  };

  return (
    <Dialog open={open} onOpenChange={(isOpen) => {
      if (!isOpen && !isProcessing) {
        onClose();
      }
    }}>
      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto" onInteractOutside={(e) => {
        if (isProcessing) {
          e.preventDefault();
        }
      }}>
        <DialogHeader>
          <DialogTitle className="text-2xl">Mobile Check Deposit</DialogTitle>
        </DialogHeader>

        <div className="space-y-6">
          {/* Progress Steps */}
          <div className="flex items-center justify-between">
            <div className={`flex items-center gap-2 ${step >= 1 ? 'text-blue-600' : 'text-gray-400'}`}>
              <div className={`w-8 h-8 rounded-full flex items-center justify-center ${
                step >= 1 ? 'bg-blue-600 text-white' : 'bg-gray-200'
              }`}>
                {frontImageUrl ? <CheckCircle2 className="w-5 h-5" /> : '1'}
              </div>
              <span className="text-sm font-medium">Front</span>
            </div>
            <div className="flex-1 h-1 bg-gray-200 mx-4" />
            <div className={`flex items-center gap-2 ${step >= 2 ? 'text-blue-600' : 'text-gray-400'}`}>
              <div className={`w-8 h-8 rounded-full flex items-center justify-center ${
                step >= 2 ? 'bg-blue-600 text-white' : 'bg-gray-200'
              }`}>
                {backImageUrl ? <CheckCircle2 className="w-5 h-5" /> : '2'}
              </div>
              <span className="text-sm font-medium">Back</span>
            </div>
            <div className="flex-1 h-1 bg-gray-200 mx-4" />
            <div className={`flex items-center gap-2 ${step >= 3 ? 'text-blue-600' : 'text-gray-400'}`}>
              <div className={`w-8 h-8 rounded-full flex items-center justify-center ${
                step >= 3 ? 'bg-blue-600 text-white' : 'bg-gray-200'
              }`}>
                3
              </div>
              <span className="text-sm font-medium">Confirm</span>
            </div>
          </div>

          {/* Step 1: Front Image */}
          {step === 1 && (
            <Card className="border-2 border-dashed border-blue-300">
              <CardContent className="p-8">
                <div className="text-center space-y-4">
                  <Camera className="w-16 h-16 text-blue-600 mx-auto" />
                  <div>
                    <h3 className="text-lg font-bold mb-2">Capture Front of Check</h3>
                    <p className="text-sm text-gray-600">
                      Ensure all four corners are visible and the check is on a dark surface
                    </p>
                  </div>
                  
                  {frontImage && (
                    <div className="mt-4 space-y-3">
                      <img 
                        src={URL.createObjectURL(frontImage)} 
                        alt="Front of check"
                        className="max-w-full h-auto rounded-lg border-2 border-green-500"
                      />
                      {imageQuality.front && (
                        <Card className={`${
                          imageQuality.front.score >= 80 ? 'bg-green-50 border-green-300' :
                          imageQuality.front.score >= 60 ? 'bg-yellow-50 border-yellow-300' :
                          'bg-red-50 border-red-300'
                        }`}>
                          <CardContent className="p-3">
                            <div className="flex items-center justify-between mb-2">
                              <span className="text-sm font-semibold">Image Quality: {imageQuality.front.rating}</span>
                              <span className="text-sm font-bold">{imageQuality.front.score}/100</span>
                            </div>
                            {imageQuality.front.issues.length > 0 && (
                              <ul className="text-xs space-y-1">
                                {imageQuality.front.issues.map((issue, idx) => (
                                  <li key={idx}>• {issue}</li>
                                ))}
                              </ul>
                            )}
                          </CardContent>
                        </Card>
                      )}
                    </div>
                  )}

                  <input
                    ref={frontInputRef}
                    type="file"
                    accept="image/*"
                    capture="environment"
                    className="hidden"
                    onChange={(e) => handleImageUpload(e.target.files[0], 'front')}
                  />

                  <Button
                    onClick={() => frontInputRef.current?.click()}
                    disabled={isProcessing}
                    className="w-full bg-blue-600"
                  >
                    {isProcessing ? (
                      <>
                        <Loader2 className="w-4 h-4 mr-2 animate-spin" />
                        Processing...
                      </>
                    ) : frontImage ? (
                      <>
                        <Camera className="w-4 h-4 mr-2" />
                        Retake Photo
                      </>
                    ) : (
                      <>
                        <Camera className="w-4 h-4 mr-2" />
                        Capture Front
                      </>
                    )}
                  </Button>
                </div>
              </CardContent>
            </Card>
          )}

          {/* Step 2: Back Image */}
          {step === 2 && (
            <Card className="border-2 border-dashed border-blue-300">
              <CardContent className="p-8">
                <div className="text-center space-y-4">
                  <Upload className="w-16 h-16 text-blue-600 mx-auto" />
                  <div>
                    <h3 className="text-lg font-bold mb-2">Capture Back of Check</h3>
                    <p className="text-sm text-gray-600">
                      Make sure you've endorsed the check with your signature
                    </p>
                  </div>

                  {backImage && (
                    <div className="mt-4 space-y-3">
                      <img 
                        src={URL.createObjectURL(backImage)} 
                        alt="Back of check"
                        className="max-w-full h-auto rounded-lg border-2 border-green-500"
                      />
                      {imageQuality.back && (
                        <Card className={`${
                          imageQuality.back.score >= 80 ? 'bg-green-50 border-green-300' :
                          imageQuality.back.score >= 60 ? 'bg-yellow-50 border-yellow-300' :
                          'bg-red-50 border-red-300'
                        }`}>
                          <CardContent className="p-3">
                            <div className="flex items-center justify-between mb-2">
                              <span className="text-sm font-semibold">Image Quality: {imageQuality.back.rating}</span>
                              <span className="text-sm font-bold">{imageQuality.back.score}/100</span>
                            </div>
                            {imageQuality.back.issues.length > 0 && (
                              <ul className="text-xs space-y-1">
                                {imageQuality.back.issues.map((issue, idx) => (
                                  <li key={idx}>• {issue}</li>
                                ))}
                              </ul>
                            )}
                          </CardContent>
                        </Card>
                      )}
                    </div>
                  )}

                  <input
                    ref={backInputRef}
                    type="file"
                    accept="image/*"
                    capture="environment"
                    className="hidden"
                    onChange={(e) => handleImageUpload(e.target.files[0], 'back')}
                  />

                  <div className="flex gap-3">
                    <Button onClick={() => setStep(1)} variant="outline" className="flex-1">
                      Back to Front
                    </Button>
                    <Button
                      onClick={() => backInputRef.current?.click()}
                      disabled={isProcessing}
                      className="flex-1 bg-blue-600"
                    >
                      {isProcessing ? (
                        <>
                          <Loader2 className="w-4 h-4 mr-2 animate-spin" />
                          Processing...
                        </>
                      ) : backImage ? (
                        <>
                          <Camera className="w-4 h-4 mr-2" />
                          Retake Photo
                        </>
                      ) : (
                        <>
                          <Camera className="w-4 h-4 mr-2" />
                          Capture Back
                        </>
                      )}
                    </Button>
                  </div>
                </div>
              </CardContent>
            </Card>
          )}

          {/* Step 3: Confirm Details */}
          {step === 3 && (
            <div className="space-y-4">
              <Card className="bg-blue-50 border-blue-200">
                <CardContent className="p-4">
                  <div className="flex items-start gap-3">
                    <AlertCircle className="w-5 h-5 text-blue-600 flex-shrink-0 mt-0.5" />
                    <div className="text-sm text-blue-800">
                      <p className="font-semibold mb-1">Review Check Details</p>
                      <p>Verify all information is correct before submitting your deposit.</p>
                    </div>
                  </div>
                </CardContent>
              </Card>

              <div className="space-y-4">
                <div>
                  <Label>Check Amount *</Label>
                  <Input
                    type="number"
                    step="0.01"
                    placeholder="0.00"
                    value={checkData.check_amount}
                    onChange={(e) => setCheckData({...checkData, check_amount: e.target.value})}
                  />
                </div>

                <div>
                  <Label>Check Number</Label>
                  <Input
                    placeholder="e.g., 1001"
                    value={checkData.check_number}
                    onChange={(e) => setCheckData({...checkData, check_number: e.target.value})}
                  />
                </div>

                <div>
                  <Label>Memo (Optional)</Label>
                  <Input
                    placeholder="Note about this deposit"
                    value={checkData.memo}
                    onChange={(e) => setCheckData({...checkData, memo: e.target.value})}
                  />
                </div>

                {extractedData && (
                  <Card className="bg-green-50 border-green-200">
                    <CardContent className="p-4">
                      <div className="flex items-start gap-3">
                        <CheckCircle2 className="w-5 h-5 text-green-600 flex-shrink-0" />
                        <div className="text-sm text-green-800">
                          <p className="font-semibold">Auto-detected Information</p>
                          {extractedData.routing_number && (
                            <p>Routing: {extractedData.routing_number}</p>
                          )}
                          {extractedData.account_number && (
                            <p>Account: •••• {extractedData.account_number.slice(-4)}</p>
                          )}
                        </div>
                      </div>
                    </CardContent>
                  </Card>
                )}
              </div>

              <div className="flex gap-3 pt-4">
                <Button onClick={() => setStep(2)} variant="outline" className="flex-1" disabled={isProcessing}>
                  Back
                </Button>
                <Button onClick={handleSubmit} className="flex-1 bg-green-600" disabled={isProcessing}>
                  {isProcessing ? (
                    <>
                      <Loader2 className="w-4 h-4 mr-2 animate-spin" />
                      Processing...
                    </>
                  ) : (
                    <>
                      <CheckCircle2 className="w-4 h-4 mr-2" />
                      Submit Deposit
                    </>
                  )}
                </Button>
              </div>
            </div>
          )}

          {/* Step 4: Success Confirmation */}
          {step === 4 && depositSuccess && (
            <div className="text-center space-y-6 py-8">
              <div className="w-24 h-24 bg-green-500 rounded-full flex items-center justify-center mx-auto animate-pulse">
                <CheckCircle2 className="w-12 h-12 text-white" />
              </div>
              <div>
                <h3 className="text-2xl font-bold text-green-900 mb-2">Deposit Submitted!</h3>
                <p className="text-gray-600 mb-4">
                  Your check for ${parseFloat(checkData.check_amount).toLocaleString()} has been submitted for processing.
                </p>
                <Card className="bg-blue-50 border-blue-200 text-left">
                  <CardContent className="p-4">
                    <p className="text-sm text-blue-800">
                      <strong>What's Next:</strong><br/>
                      • Funds are being verified and will be available within 1-2 business days<br/>
                      • You'll receive a notification once the deposit clears<br/>
                      • Keep the original check for 14 days
                    </p>
                  </CardContent>
                </Card>
              </div>
            </div>
          )}
        </div>
      </DialogContent>
    </Dialog>
  );
}

import React from "react";
import { Card, CardContent } from "@/components/ui/card";
import { Badge } from "@/components/ui/badge";
import { CheckCircle2, Clock, XCircle, AlertCircle, DollarSign } from "lucide-react";

export default function DepositHistory({ deposits }) {
  const getStatusIcon = (status) => {
    switch (status) {
      case 'cleared':
        return <CheckCircle2 className="w-5 h-5 text-green-600" />;
      case 'processing':
      case 'pending':
        return <Clock className="w-5 h-5 text-yellow-600" />;
      case 'rejected':
        return <XCircle className="w-5 h-5 text-red-600" />;
      case 'on_hold':
        return <AlertCircle className="w-5 h-5 text-orange-600" />;
      default:
        return <Clock className="w-5 h-5 text-gray-600" />;
    }
  };

  const getStatusColor = (status) => {
    switch (status) {
      case 'cleared':
        return 'bg-green-100 text-green-800';
      case 'processing':
      case 'pending':
        return 'bg-yellow-100 text-yellow-800';
      case 'rejected':
        return 'bg-red-100 text-red-800';
      case 'on_hold':
        return 'bg-orange-100 text-orange-800';
      default:
        return 'bg-gray-100 text-gray-800';
    }
  };

  if (deposits.length === 0) {
    return (
      <Card>
        <CardContent className="p-12 text-center">
          <DollarSign className="w-16 h-16 text-gray-300 mx-auto mb-4" />
          <h3 className="text-lg font-semibold text-gray-900 mb-2">No Deposits Yet</h3>
          <p className="text-gray-600">
            Start by depositing your first check using the camera button above.
          </p>
        </CardContent>
      </Card>
    );
  }

  return (
    <div className="space-y-4">
      {deposits.map((deposit) => (
        <Card key={deposit.id} className="hover:shadow-lg transition-shadow">
          <CardContent className="p-6">
            <div className="flex items-start justify-between">
              <div className="flex items-start gap-4 flex-1">
                <div className="mt-1">
                  {getStatusIcon(deposit.status)}
                </div>
                <div className="flex-1">
                  <div className="flex items-center gap-3 mb-2">
                    <h3 className="font-bold text-lg">
                      ${deposit.check_amount?.toFixed(2)}
                    </h3>
                    <Badge className={getStatusColor(deposit.status)}>
                      {deposit.status}
                    </Badge>
                  </div>
                  
                  <div className="grid grid-cols-2 gap-2 text-sm text-gray-600">
                    {deposit.check_number && (
                      <p>Check #{deposit.check_number}</p>
                    )}
                    <p>Deposited: {new Date(deposit.deposit_date).toLocaleDateString()}</p>
                    
                    {deposit.cleared_date && (
                      <p>Cleared: {new Date(deposit.cleared_date).toLocaleDateString()}</p>
                    )}
                    
                    {deposit.available_amount > 0 && (
                      <p className="text-green-600 font-semibold">
                        Available: ${deposit.available_amount.toFixed(2)}
                      </p>
                    )}
                    
                    {deposit.hold_amount > 0 && (
                      <p className="text-yellow-600 font-semibold">
                        On Hold: ${deposit.hold_amount.toFixed(2)}
                      </p>
                    )}
                  </div>

                  {deposit.memo && (
                    <p className="text-sm text-gray-500 mt-2 italic">
                      Memo: {deposit.memo}
                    </p>
                  )}

                  {deposit.rejection_reason && (
                    <div className="mt-3 p-3 bg-red-50 rounded-lg">
                      <p className="text-sm text-red-800">
                        <strong>Rejection Reason:</strong> {deposit.rejection_reason}
                      </p>
                    </div>
                  )}

                  {deposit.image_quality_score && deposit.image_quality_score < 70 && (
                    <div className="mt-2">
                      <p className="text-xs text-orange-600">
                        ⚠️ Image quality: {deposit.image_quality_score}/100
                      </p>
                    </div>
                  )}
                </div>
              </div>

              {deposit.front_image_url && (
                <div className="ml-4">
                  <img
                    src={deposit.front_image_url}
                    alt="Check front"
                    className="w-32 h-20 object-cover rounded-lg border-2 border-gray-200"
                  />
                </div>
              )}
            </div>
          </CardContent>
        </Card>
      ))}
    </div>
  );
}

import React from "react";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Progress } from "@/components/ui/progress";
import { Shield, TrendingUp, Calendar, AlertCircle } from "lucide-react";

export default function DepositLimits({ deposits }) {
  const dailyLimit = 5000;
  const monthlyLimit = 20000;

  const today = new Date();
  today.setHours(0, 0, 0, 0);
  const todayDeposits = deposits.filter(d => {
    const depositDate = new Date(d.deposit_date);
    depositDate.setHours(0, 0, 0, 0);
    return depositDate.getTime() === today.getTime();
  });
  const dailyTotal = todayDeposits.reduce((sum, d) => sum + (d.check_amount || 0), 0);

  const thisMonth = deposits.filter(d => {
    const depositDate = new Date(d.deposit_date);
    return depositDate.getMonth() === today.getMonth() && 
           depositDate.getFullYear() === today.getFullYear();
  });
  const monthlyTotal = thisMonth.reduce((sum, d) => sum + (d.check_amount || 0), 0);

  const dailyPercentage = (dailyTotal / dailyLimit) * 100;
  const monthlyPercentage = (monthlyTotal / monthlyLimit) * 100;

  return (
    <div className="space-y-6">
      <Card>
        <CardHeader>
          <CardTitle className="flex items-center gap-2">
            <TrendingUp className="w-5 h-5" />
            Deposit Limits
          </CardTitle>
        </CardHeader>
        <CardContent className="space-y-6">
          <div>
            <div className="flex items-center justify-between mb-2">
              <span className="text-sm font-medium">Daily Limit</span>
              <span className="text-sm text-gray-600">
                ${dailyTotal.toLocaleString()} / ${dailyLimit.toLocaleString()}
              </span>
            </div>
            <Progress value={dailyPercentage} className="h-3" />
            <p className="text-xs text-gray-500 mt-1">
              ${(dailyLimit - dailyTotal).toLocaleString()} remaining today
            </p>
          </div>

          <div>
            <div className="flex items-center justify-between mb-2">
              <span className="text-sm font-medium">Monthly Limit</span>
              <span className="text-sm text-gray-600">
                ${monthlyTotal.toLocaleString()} / ${monthlyLimit.toLocaleString()}
              </span>
            </div>
            <Progress value={monthlyPercentage} className="h-3" />
            <p className="text-xs text-gray-500 mt-1">
              ${(monthlyLimit - monthlyTotal).toLocaleString()} remaining this month
            </p>
          </div>
        </CardContent>
      </Card>

      <Card>
        <CardHeader>
          <CardTitle className="flex items-center gap-2">
            <Calendar className="w-5 h-5" />
            Funds Availability
          </CardTitle>
