
import React, { useState } from "react";
import { Button } from "@/components/ui/button";
import { Download, Loader2 } from "lucide-react";

export const generatePayStubPDF = (payStub) => {
  const totalDeductions = Object.values(payStub.deductions || {}).reduce((sum, val) => sum + (val || 0), 0);
  
  const formatDate = (dateValue) => {
    if (!dateValue) return 'N/A';
    try {
      const date = new Date(dateValue);
      if (isNaN(date.getTime())) return 'N/A';
      return date.toLocaleDateString('en-US', { year: 'numeric', month: '2-digit', day: '2-digit' });
    } catch (error) {
      return 'N/A';
    }
  };

  const htmlContent = `
<!DOCTYPE html>
<html>
<head>
  <meta charset="UTF-8">
  <title>Pay Stub - ${payStub.employee_name}</title>
  <style>
    body {
      font-family: 'Arial', sans-serif;
      margin: 0;
      padding: 40px;
      color: #333;
    }
    .container {
      max-width: 800px;
      margin: 0 auto;
      border: 2px solid #1e40af;
      padding: 40px;
    }
    .header {
      text-align: center;
      border-bottom: 3px solid #1e40af;
      padding-bottom: 20px;
      margin-bottom: 30px;
    }
    .company-name {
      font-size: 28px;
      font-weight: bold;
      color: #1e40af;
      margin-bottom: 10px;
    }
    .subtitle {
      color: #666;
      font-size: 14px;
    }
    .info-grid {
      display: grid;
      grid-template-columns: 1fr 1fr;
      gap: 30px;
      margin-bottom: 30px;
    }
    .info-section {
      background: #f8f9fa;
      padding: 15px;
      border-radius: 8px;
    }
    .info-title {
      font-weight: bold;
      margin-bottom: 10px;
      color: #1e40af;
      font-size: 16px;
    }
    .info-row {
      display: flex;
      justify-content: space-between;
      margin: 8px 0;
      font-size: 14px;
    }
    .earnings-deductions {
      display: grid;
      grid-template-columns: 1fr 1fr;
      gap: 20px;
      margin-bottom: 30px;
    }
    .box {
      border: 1px solid #ddd;
      padding: 20px;
      border-radius: 8px;
    }
    .box-title {
      font-weight: bold;
      margin-bottom: 15px;
      font-size: 16px;
      color: #1e40af;
    }
    .amount-row {
      display: flex;
      justify-content: space-between;
      margin: 10px 0;
      font-size: 14px;
    }
    .total-row {
      border-top: 2px solid #333;
      padding-top: 10px;
      margin-top: 15px;
      font-weight: bold;
    }
    .net-pay {
      background: linear-gradient(135deg, #10b981 0%, #059669 100%);
      color: white;
      padding: 25px;
      border-radius: 12px;
      text-align: center;
      margin: 30px 0;
    }
    .net-pay-label {
      font-size: 14px;
      opacity: 0.9;
      margin-bottom: 8px;
    }
    .net-pay-amount {
      font-size: 36px;
      font-weight: bold;
    }
    .ytd-section {
      border-top: 2px solid #ddd;
      padding-top: 20px;
    }
    .ytd-grid {
      display: grid;
      grid-template-columns: 1fr 1fr;
      gap: 20px;
    }
    @media print {
      body { padding: 0; }
      .container { border: none; }
    }
  </style>
</head>
<body>
  <div class="container">
    <div class="header">
      <div class="company-name">${payStub.employer_name}</div>
      <div class="subtitle">Employee Pay Statement</div>
    </div>

    <div class="info-grid">
      <div class="info-section">
        <div class="info-title">Employee Information</div>
        <div class="info-row">
          <span>Name:</span>
          <span><strong>${payStub.employee_name}</strong></span>
        </div>
        ${payStub.employee_id ? `
        <div class="info-row">
          <span>Employee ID:</span>
          <span><strong>${payStub.employee_id}</strong></span>
        </div>
        ` : ''}
      </div>

      <div class="info-section">
        <div class="info-title">Pay Period</div>
        <div class="info-row">
          <span>Period:</span>
          <span><strong>${formatDate(payStub.pay_period_start)} - ${formatDate(payStub.pay_period_end)}</strong></span>
        </div>
        <div class="info-row">
          <span>Pay Date:</span>
          <span><strong>${formatDate(payStub.pay_date)}</strong></span>
        </div>
      </div>
    </div>

    <div class="earnings-deductions">
      <div class="box">
        <div class="box-title">Earnings</div>
        <div class="amount-row">
          <span>Gross Pay:</span>
          <span><strong>$${(payStub.gross_pay || 0).toFixed(2)}</strong></span>
        </div>
        ${payStub.hourly_rate && payStub.hours_worked ? `

          <span>Rate:</span>
          <span>$${parseFloat(payStub.hourly_rate).toFixed(2)}/hr</span>
        </div>
        <div class="amount-row">
          <span>Hours:</span>
          <span>${parseFloat(payStub.hours_worked).toFixed(2)}</span>
        </div>
        ` : ''}
      </div>

      <div class="box">
        <div class="box-title">Deductions</div>
        ${Object.entries(payStub.deductions || {}).map(([key, value]) => {
          if (!value || value === 0) return '';
          const label = key.replace(/_/g, ' ').replace(/\b\w/g, l => l.toUpperCase());
          return `
          <div class="amount-row">
            <span>${label}:</span>
            <span>$${parseFloat(value).toFixed(2)}</span>
          </div>
          `;
        }).join('')}
        <div class="amount-row total-row">
          <span>Total Deductions:</span>
          <span>$${totalDeductions.toFixed(2)}</span>
        </div>
      </div>
    </div>

    <div class="net-pay">
      <div class="net-pay-label">Net Pay</div>
      <div class="net-pay-amount">$${(payStub.net_pay || 0).toFixed(2)}</div>
      <div class="net-pay-label" style="margin-top: 8px;">Amount deposited to your account</div>
    </div>

    <div class="ytd-section">
      <div class="info-title">Year-to-Date Summary</div>
      <div class="ytd-grid">
        <div class="amount-row">
          <span>YTD Gross:</span>
          <span><strong>$${(payStub.ytd_gross || 0).toFixed(2)}</strong></span>
        </div>
        <div class="amount-row">
          <span>YTD Net:</span>
          <span><strong>$${(payStub.ytd_net || 0).toFixed(2)}</strong></span>
        </div>
      </div>
    </div>
  </div>
</body>
</html>
  `;

  // Create and download as HTML (which can be printed to PDF)
  const blob = new Blob([htmlContent], { type: 'text/html' });
  const url = window.URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `paystub_${payStub.employee_name.replace(/\s+/g, '_')}_${formatDate(payStub.pay_date)}.html`;
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  window.URL.revokeObjectURL(url);

  // Open in new window for printing
  const printWindow = window.open('', '_blank');
  printWindow.document.write(htmlContent);
  printWindow.document.close();
  
  setTimeout(() => {
    printWindow.print();
  }, 500);
};

export default function PDFDownloadButton({ type, data, children }) {
  const [isGenerating, setIsGenerating] = useState(false);

  const handleDownload = () => {
    setIsGenerating(true);
    try {
      if (type === 'paystub') {
        generatePayStubPDF(data);
      } else if (type === 'tax') {
        generateTaxDocumentPDF(data);
      }
    } catch (error) {
      console.error('Error generating PDF:', error);
      alert('Failed to generate PDF. Please try again.');
    } finally {
      setTimeout(() => setIsGenerating(false), 1000);
    }
  };

  return (
    <Button onClick={handleDownload} disabled={isGenerating} className="gap-2">
      {isGenerating ? (
        <>
          <Loader2 className="w-4 h-4 animate-spin" />
          Generating...
        </>
      ) : (
        <>
          <Download className="w-4 h-4" />
          {children || 'Download PDF'}
        </>
      )}
    </Button>
  );
}
