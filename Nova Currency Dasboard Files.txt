import React from "react";

export default function CryptoOverview({ holdings }) {
  const totalValue = holdings.reduce((sum, holding) => sum + (holding.current_value || 0), 0);

  return (
    <Card className="bg-white/80 backdrop-blur-sm border-0 shadow-xl">

        <div className="flex items-center justify-between">
          <h3 className="text-xl font-bold text-gray-900">Crypto Portfolio</h3>
          <Link to={createPageUrl("Crypto")}>
            <Button variant="ghost" size="sm">View All</Button>
          </Link>
        </div>
      </CardHeader>
      <CardContent>
        {holdings.length > 0 ? (
          <div className="space-y-4">
            <div className="p-4 bg-gradient-to-r from-orange-50 to-yellow-50 rounded-xl">
              <div className="flex items-center gap-3">
                <Bitcoin className="w-8 h-8 text-orange-600" />
                <div className="flex-1">
                  <p className="text-sm text-gray-600">Total Portfolio Value</p>
                  <p className="text-2xl font-bold text-gray-900">${totalValue.toFixed(2)}</p>
                </div>
                <Badge className="bg-green-100 text-green-800">
                  <TrendingUp className="w-3 h-3 mr-1" />
                  +12.5%
                </Badge>
              </div>
            </div>
            
            <div className="space-y-3">
              {holdings.slice(0, 3).map((holding) => (
                <div key={holding.id} className="flex items-center justify-between">
                  <div className="flex items-center gap-3">
                    <div className="w-8 h-8 bg-gray-200 rounded-full flex items-center justify-center">

                    </div>
                    <div>
                      <p className="font-medium">{holding.crypto_symbol}</p>

                    </div>
                  </div>
                  <div className="text-right">
                    <p className="font-bold">${holding.current_value?.toFixed(2)}</p>
                  </div>
                </div>
              ))}
            </div>
          </div>
        ) : (
          <div className="text-center py-8">
            <Bitcoin className="w-12 h-12 text-gray-400 mx-auto mb-4" />
            <p className="text-gray-600 mb-4">Start your crypto journey</p>

import React from "react";
import { ArrowUpRight, ArrowDownRight, Bitcoin, ShoppingBag, Car, Coffee } from "lucide-react";
import { Card, CardContent, CardHeader } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { Badge } from "@/components/ui/badge";
import { Link } from "react-router-dom";
import { createPageUrl } from "@/utils";
import { format } from "date-fns";

const getCategoryIcon = (category) => {
  switch (category) {
    case 'crypto': return Bitcoin;
    case 'shopping': return ShoppingBag;
    case 'gas': return Car;
    case 'food': return Coffee;
    default: return ArrowUpRight;
  }
};

export default function RecentTransactions({ transactions }) {
  return (
    <Card className="bg-white/80 backdrop-blur-sm border-0 shadow-xl">
      <CardHeader>
        <div className="flex items-center justify-between">
          <h3 className="text-xl font-bold text-gray-900">Recent Transactions</h3>
          <Link to={createPageUrl("Transactions")}>
            <Button variant="ghost" size="sm">View All</Button>
          </Link>
        </div>
      </CardHeader>
      <CardContent>
        {transactions.length > 0 ? (
          <div className="space-y-4">
            {transactions.slice(0, 5).map((transaction) => {
              const CategoryIcon = getCategoryIcon(transaction.category);
              const isCredit = transaction.type === 'credit' || transaction.type === 'transfer_in';
              
              return (
                <div key={transaction.id} className="flex items-center justify-between p-4 bg-gray-50 rounded-xl hover:bg-gray-100 transition-colors">
                  <div className="flex items-center gap-4">
                    <div className={`w-12 h-12 rounded-full flex items-center justify-center ${
                      isCredit ? 'bg-green-100' : 'bg-red-100'
                    }`}>
                      <CategoryIcon className={`w-6 h-6 ${
                        isCredit ? 'text-green-600' : 'text-red-600'
                      }`} />
                    </div>
                    <div>
                      <p className="font-medium text-gray-900">{transaction.description}</p>
                      <p className="text-sm text-gray-500">
                        {transaction.merchant && `${transaction.merchant} • `}
                        {format(new Date(transaction.created_date), 'MMM d, h:mm a')}
                      </p>
                    </div>
                  </div>
                  <div className="text-right">
                    <p className={`font-bold ${
                      isCredit ? 'text-green-600' : 'text-red-600'
                    }`}>
                      {isCredit ? '+' : '-'}${Math.abs(transaction.amount).toFixed(2)}
                    </p>
                    <Badge variant={transaction.status === 'completed' ? 'default' : 'secondary'} className="text-xs">
                      {transaction.status}
                    </Badge>
                  </div>
                </div>
              );
            })}
          </div>
        ) : (
          <div className="text-center py-8">
            <ArrowUpRight className="w-12 h-12 text-gray-400 mx-auto mb-4" />
            <p className="text-gray-600">No recent transactions</p>
          </div>
        )}
      </CardContent>
    </Card>
  );
}
